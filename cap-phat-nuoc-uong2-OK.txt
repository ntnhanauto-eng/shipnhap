function renderPaginationBar($currentPage, $totalPages, $paramName) {
    if ($totalPages <= 1) return;
    $queryParams = $_GET;

    echo '<div class="pagination-wrapper">';
    if ($currentPage > 1) {
        $queryParams[$paramName] = $currentPage - 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&laquo;</a>';
    } else {
        echo '<span class="page-link disabled">&laquo;</span>';
    }

    $start = max(1, $currentPage - 1);
    $end = min($totalPages, $currentPage + 1);
    if ($currentPage == 1) $end = min($totalPages, $start + 2);
    elseif ($currentPage == $totalPages) $start = max(1, $end - 2);

    for ($i = $start; $i <= $end; $i++) {
        $queryParams[$paramName] = $i;
        $activeClass = ($i == $currentPage) ? 'active' : '';
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link ' . $activeClass . '">' . $i . '</a>';
    }

    if ($currentPage < $totalPages) {
        $queryParams[$paramName] = $currentPage + 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&raquo;</a>';
    } else {
        echo '<span class="page-link disabled">&raquo;</span>';
    }
    echo '</div>';
}

function generateUniqueWaterCode($pdo, $table, $timeStr, $excludeId = 0) {
    $ts = strtotime($timeStr) ?: time();
    $baseCode = date('dmYHis', $ts);

    $sql = "SELECT request_code FROM `{$table}` WHERE request_code LIKE :b AND id != :exId";
    $stmt = $pdo->prepare($sql);
    $stmt->execute([':b' => $baseCode . '%', ':exId' => $excludeId]);
    $existing = $stmt->fetchAll(PDO::FETCH_COLUMN);

    if (empty($existing) || !in_array($baseCode, $existing)) {
        return $baseCode;
    }

    $suffixes = range('A', 'Z');
    foreach ($suffixes as $char) {
        $candidate = $baseCode . $char;
        if (!in_array($candidate, $existing)) {
            return $candidate;
        }
    }
    return $baseCode . '_' . rand(10, 99);
}

function getDisplayCode($row) {
    if (!empty($row['request_code'])) {
        return $row['request_code'];
    }
    return date('dmYHis', strtotime($row['created_at']));
}

function is19LBottle($name, $spec = '') {
    $str = mb_strtolower($name . ' ' . $spec, 'UTF-8');
    return (strpos($str, '19l') !== false || strpos($str, '19 l') !== false || strpos($str, '19 lít') !== false || strpos($str, '20l') !== false);
}

// Hàm hỗ trợ đọc file Excel (.xlsx) thuần PHP không cần thư viện ngoài
function parseXlsxToArray($filepath) {
    if (!class_exists('ZipArchive')) return null;
    $zip = new ZipArchive();
    if ($zip->open($filepath) !== true) return null;

    $strings = [];
    $sharedStrXml = $zip->getFromName('xl/sharedStrings.xml');
    if ($sharedStrXml) {
        $xml = simplexml_load_string($sharedStrXml);
        if ($xml) {
            foreach ($xml->si as $si) {
                if (isset($si->t)) {
                    $strings[] = (string)$si->t;
                } elseif (isset($si->r)) {
                    $t = '';
                    foreach ($si->r as $r) { $t .= (string)$r->t; }
                    $strings[] = $t;
                } else {
                    $strings[] = '';
                }
            }
        }
    }

    $sheetXml = $zip->getFromName('xl/worksheets/sheet1.xml');
    if (!$sheetXml) {
        $zip->close();
        return null;
    }

    $xmlSheet = simplexml_load_string($sheetXml);
    $rows = [];
    if ($xmlSheet && isset($xmlSheet->sheetData->row)) {
        foreach ($xmlSheet->sheetData->row as $r) {
            $rowValues = [];
            foreach ($r->c as $c) {
                $val = (string)$c->v;
                $type = (string)$c['t'];
                if ($type === 's' && isset($strings[intval($val)])) {
                    $val = $strings[intval($val)];
                }
                $rowValues[] = trim($val);
            }
            if (!empty(array_filter($rowValues, fn($v) => $v !== ''))) {
                $rows[] = $rowValues;
            }
        }
    }
    $zip->close();
    return $rows;
}

        $fileName = $_FILES['excel_file']['name'];
        $ext = strtolower(pathinfo($fileName, PATHINFO_EXTENSION));

        $rows = [];
        if ($ext === 'csv') {
            if (($handle = fopen($tmpName, 'r')) !== false) {
                while (($data = fgetcsv($handle, 2000, ',')) !== false) {
                    $rows[] = array_map('trim', $data);
                }
                fclose($handle);
            }
        } elseif ($ext === 'xlsx') {
            $rows = parseXlsxToArray($tmpName);
            if ($rows === null) {
                $error = "Máy chủ PHP cần hỗ trợ ZipArchive để đọc file .xlsx hoặc hãy lưu file dạng CSV/XLS!";
            }
        } elseif ($ext === 'xls') {
            $content = file_get_contents($tmpName);
            if (strpos($content, '<table') !== false) {
                $dom = new DOMDocument();
                @$dom->loadHTML(mb_convert_encoding($content, 'HTML-ENTITIES', 'UTF-8'));
                $trs = $dom->getElementsByTagName('tr');
                foreach ($trs as $tr) {
                    $rowValues = [];
                    $tds = $tr->getElementsByTagName('td');
                    if ($tds->length === 0) $tds = $tr->getElementsByTagName('th');
                    foreach ($tds as $td) {
                        $rowValues[] = trim($td->textContent);
                    }
                    if (!empty(array_filter($rowValues))) $rows[] = $rowValues;
                }
            }
        }

        if (!empty($rows)) {
            $allItems = $pdo->query("SELECT id, item_code, item_name FROM water_items")->fetchAll();
            $itemMapCode = [];
            $itemMapName = [];
            foreach ($allItems as $it) {
                if (!empty($it['item_code'])) $itemMapCode[mb_strtolower(trim($it['item_code']), 'UTF-8')] = $it['id'];
                $itemMapName[mb_strtolower(trim($it['item_name']), 'UTF-8')] = $it['id'];
            }

            $header = array_shift($rows);
            $idxDate = -1; $idxSup = -1; $idxCode = -1; $idxName = -1; $idxQty = -1; $idxInv = -1; $idxNote = -1;

            foreach ($header as $colIdx => $colName) {
                $c = mb_strtolower(trim($colName), 'UTF-8');
                if (strpos($c, 'ngày') !== false || strpos($c, 'date') !== false) $idxDate = $colIdx;
                elseif (strpos($c, 'nhà cung cấp') !== false || strpos($c, 'ncc') !== false || strpos($c, 'supplier') !== false) $idxSup = $colIdx;
                elseif (strpos($c, 'mã sp') !== false || strpos($c, 'mã') !== false || strpos($c, 'code') !== false) $idxCode = $colIdx;
                elseif (strpos($c, 'tên') !== false || strpos($c, 'loại nước') !== false || strpos($c, 'item') !== false) $idxName = $colIdx;
                elseif (strpos($c, 'số lượng') !== false || strpos($c, 'qty') !== false || strpos($c, 'sl') !== false) $idxQty = $colIdx;
                elseif (strpos($c, 'hóa đơn') !== false || strpos($c, 'chứng từ') !== false || strpos($c, 'invoice') !== false) $idxInv = $colIdx;
                elseif (strpos($c, 'ghi chú') !== false || strpos($c, 'note') !== false) $idxNote = $colIdx;
            }

            if ($idxSup === -1) $idxSup = 1;
            if ($idxDate === -1) $idxDate = 0;
            if ($idxCode === -1) $idxCode = 2;
            if ($idxName === -1) $idxName = 3;
            if ($idxQty === -1) $idxQty = 4;
            if ($idxInv === -1) $idxInv = 5;
            if ($idxNote === -1) $idxNote = 6;

            $successCount = 0;
            $failRows = [];

            try {
                $pdo->beginTransaction();
                $stmtIns = $pdo->prepare("
                    INSERT INTO water_imports (invoice_no, supplier_name, import_date, item_id, quantity, notes) 
                    VALUES (:inv, :sup, :idate, :it, :qty, :n)
                ");
                $stmtStock = $pdo->prepare("
                    UPDATE water_items 
                    SET stock_qty = stock_qty + :q1, total_imported = total_imported + :q2 
                    WHERE id = :id
                ");

                foreach ($rows as $lineIndex => $row) {
                    $rowNum = $lineIndex + 2;
                    $supplier = trim($row[$idxSup] ?? '');
                    $qty = intval(preg_replace('/[^0-9]/', '', $row[$idxQty] ?? 0));
                    $rawDate = trim($row[$idxDate] ?? '');
                    $invoice = trim($row[$idxInv] ?? '');
                    $note = trim($row[$idxNote] ?? '');

                    $parsedDate = date('Y-m-d', strtotime($rawDate));
                    if (!$parsedDate || $parsedDate === '1970-01-01') $parsedDate = $today;

                    $pCode = mb_strtolower(trim($row[$idxCode] ?? ''), 'UTF-8');
                    $pName = mb_strtolower(trim($row[$idxName] ?? ''), 'UTF-8');

                    $matchedItemId = null;
                    if ($pCode !== '' && isset($itemMapCode[$pCode])) {
                        $matchedItemId = $itemMapCode[$pCode];
                    } elseif ($pName !== '' && isset($itemMapName[$pName])) {
                        $matchedItemId = $itemMapName[$pName];
                    }

                    if (empty($supplier) || $qty <= 0 || !$matchedItemId) {
                        $failRows[] = "Dòng $rowNum: Không tìm thấy sản phẩm hợp lệ hoặc thiếu tên NCC / số lượng.";
                        continue;
                    }

                    $stmtIns->execute([
                        ':inv'   => $invoice,
                        ':sup'   => $supplier,
                        ':idate' => $parsedDate,
                        ':it'    => $matchedItemId,
                        ':qty'   => $qty,
                        ':n'     => $note
                    ]);
                    $stmtStock->execute([':q1' => $qty, ':q2' => $qty, ':id' => $matchedItemId]);
                    $successCount++;
                }

                $pdo->commit();

                if ($successCount > 0) {
                    $message = "Đã nhập thành công {$successCount} phiếu nhập kho từ file Excel!";
                }
                if (!empty($failRows)) {
                    $error = "Bỏ qua " . count($failRows) . " dòng do dữ liệu không khớp: " . implode('<br>', array_slice($failRows, 0, 5));
                    if (count($failRows) > 5) $error .= "<br>...và " . (count($failRows) - 5) . " dòng lỗi khác.";
                }
            } catch (Exception $e) {
                if ($pdo->inTransaction()) $pdo->rollBack();
                $error = "Lỗi khi nhập dữ liệu: " . $e->getMessage();
            }
        } elseif (empty($error)) {
            $error = "File không có dữ liệu hoặc định dạng file không đúng!";
        }
    } else {
        $error = "Vui lòng chọn file Excel hợp lệ để tải lên!";
    }
}

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'update_request') {
    $req_id   = intval($_POST['request_id']);
    $new_date = $_POST['request_date'];
    $new_dept = trim($_POST['department']);
    $new_rec  = trim($_POST['receiver_name'] ?? '');
    $new_item = intval($_POST['item_id']);
    $new_qty  = intval($_POST['quantity']);
    $new_loc  = trim($_POST['delivery_location'] ?? '');
    $new_notes= trim($_POST['notes'] ?? '');

    try {
        $pdo->beginTransaction();
        $stmtOld = $pdo->prepare("SELECT * FROM water_requests WHERE id = :id FOR UPDATE");
        $stmtOld->execute([':id' => $req_id]);
        $oldReq = $stmtOld->fetch();

        if (!$oldReq) throw new Exception("Bản ghi cấp phát không tồn tại!");

        $stmtIt = $pdo->prepare("SELECT item_name, specification, stock_qty, unit FROM water_items WHERE id = :id FOR UPDATE");
        $stmtIt->execute([':id' => $new_item]);
        $it = $stmtIt->fetch();
        if (!$it) throw new Exception("Sản phẩm nước không tồn tại!");

        if (intval($oldReq['item_id']) === $new_item) {
            $diff = $new_qty - intval($oldReq['quantity']);
            if ($diff > 0 && $diff > $it['stock_qty']) {
                throw new Exception("Không đủ nước trong kho để tăng số lượng cấp! Còn thiếu {$diff} {$it['unit']}.");
            }
            $stmtUpStock = $pdo->prepare("UPDATE water_items SET stock_qty = GREATEST(0, stock_qty - :diff) WHERE id = :id");
            $stmtUpStock->execute([':diff' => $diff, ':id' => $new_item]);
        } else {
            $stmtBack = $pdo->prepare("UPDATE water_items SET stock_qty = stock_qty + :q WHERE id = :id");
            $stmtBack->execute([':q' => $oldReq['quantity'], ':id' => $oldReq['item_id']]);

            if ($new_qty > $it['stock_qty']) {
                throw new Exception("Không đủ tồn kho mặt hàng mới để cấp! Tồn: {$it['stock_qty']} {$it['unit']}.");
            }
            $stmtTake = $pdo->prepare("UPDATE water_items SET stock_qty = stock_qty - :q WHERE id = :id");
            $stmtTake->execute([':q' => $new_qty, ':id' => $new_item]);
        }

        $is19L = is19LBottle($it['item_name'], $it['specification']);
        $ret_bottles = $is19L ? max(0, intval($_POST['return_bottle_count'] ?? 0)) : 0;

        $stmtUpReq = $pdo->prepare("
            UPDATE water_requests 
            SET request_date = :r_date, department = :dept, receiver_name = :rec, item_id = :item_id, 
                quantity = :qty, has_bottle_return = :has_ret, delivery_location = :loc, notes = :notes 
            WHERE id = :id
        ");
        $stmtUpReq->execute([
            ':r_date'  => $new_date, ':dept'    => $new_dept, ':rec'     => $new_rec,
            ':item_id' => $new_item, ':qty'     => $new_qty,  ':has_ret' => $ret_bottles,
            ':loc'     => $new_loc,  ':notes'   => $new_notes, ':id'     => $req_id
        ]);

        $pdo->commit();
        $message = "Đã cập nhật bản ghi lịch sử cấp phát nước thành công!";
    } catch (Exception $e) {
        if ($pdo->inTransaction()) $pdo->rollBack();
        $error = "Lỗi cập nhật: " . $e->getMessage();
    }
}

function openModal(id) { document.getElementById(id).style.display = 'flex'; }
function closeModal(id) { document.getElementById(id).style.display = 'none'; }
function toggleMoreMenu(e) { if(e) e.stopPropagation(); closeAllMenus(); document.getElementById('more_dropdown_menu').classList.toggle('show'); }

// Bật/tắt menu dropdown cho nút Nhập kho
function toggleImportMenu(e) {
    if (e) e.stopPropagation();
    const btn = e.currentTarget;
    const parent = btn.closest('.import-btn-group');
    const menu = parent ? parent.querySelector('.import-menu') : null;
    const isShown = menu && menu.classList.contains('show');
    closeAllMenus();
    if (menu && !isShown) {
        menu.classList.add('show');
    }
}

function closeAllMenus() {
    document.querySelectorAll('.import-menu').forEach(el => el.classList.remove('show'));
    const moreMenu = document.getElementById('more_dropdown_menu');
    if (moreMenu) moreMenu.classList.remove('show');
}

// Bật/tắt menu Thao tác trong bảng Đề Xuất
function togglePropDropdown(e, id) {
    if (e) e.stopPropagation();
    const currentMenu = document.getElementById('prop_dropdown_' + id);
    if (!currentMenu) return;
    const isShown = currentMenu.classList.contains('show');
    closeAllPropDropdowns();
    closeAllPencilDropdowns();
    if (!isShown) {
        currentMenu.classList.add('show');
    }
}

function toggleNotePopover(e, btn) {
    if (e) e.stopPropagation();
    const container = btn.closest('.note-interactive-container');
    if (!container) return;
    const pop = container.querySelector('.note-popover');

window.onclick = function(e) {
    if (e.target.classList.contains('modal-overlay')) e.target.style.display = 'none';
    if (!e.target.closest('.import-btn-group')) closeAllMenus();
    if (!e.target.closest('.more-dropdown')) {
        const moreMenu = document.getElementById('more_dropdown_menu');
        if (moreMenu) moreMenu.classList.remove('show');
    }
    if (!e.target.closest('.prop-dropdown')) closeAllPropDropdowns();
    if (!e.target.closest('.prop-pencil-dropdown')) closeAllPencilDropdowns();
    if (!e.target.closest('.note-interactive-container')) closeAllNotePopovers();
}

// Mở modal Phê Duyệt đề xuất (Có thời gian gửi trên cùng, mã yêu cầu, Người yêu cầu & Phòng ban)
function openApproveModal(p, code) {
    document.getElementById('ap_proposal_id').value = p.id;
    document.getElementById('ap_time').innerText = p.created_at || '';
    document.getElementById('ap_request_code').value = code || p.request_code || '';
    document.getElementById('ap_receiver_name').value = p.receiver_name;
    document.getElementById('ap_department').value = p.department;
    document.getElementById('ap_item_name').value = p.item_name + ' (' + (p.specification || '—') + ')';
    document.getElementById('ap_req_quantity').value = p.quantity;
    document.getElementById('ap_appr_quantity').value = (p.approved_quantity !== null && parseInt(p.approved_quantity) > 0) ? p.approved_quantity : p.quantity;
    document.getElementById('ap_approval_note').value = p.approval_note || '';

    const str = ((p.item_name || '') + ' ' + (p.specification || '')).toLowerCase();
    const is19L = (str.includes('19l') || str.includes('19 l') || str.includes('19 lít') || str.includes('20l'));
    const bottleWrap = document.getElementById('ap_bottle_wrapper');
    const inpBottle = document.getElementById('ap_return_bottle_count');

    if (is19L) {
        bottleWrap.style.display = 'block';
        inpBottle.value = parseInt(p.return_bottle_count) || 0;
    } else {
        bottleWrap.style.display = 'none';
        inpBottle.value = 0;
    }

    openModal('modal-approve-proposal');
}

// Mở modal Chỉnh sửa nội dung đề xuất (Hiển thị đầy đủ thông tin trước/sau duyệt)
function openEditProposalModal(p) {
    document.getElementById('ep_proposal_id').value = p.id;
    document.getElementById('ep_receiver_name').value = p.receiver_name;
    document.getElementById('ep_department').value = p.department;
    document.getElementById('ep_item_id').value = p.item_id;
    document.getElementById('ep_quantity').value = p.quantity;
    document.getElementById('ep_delivery_location').value = p.delivery_location || '';

    // Xử lý thông tin sau duyệt: Chỉ hiển thị khi trạng thái là Đã duyệt hoặc Đã cấp phát
    const approvedGroup = document.getElementById('ep_approved_group');
    const inpApprQty = document.getElementById('ep_approved_quantity');
    const inpApprNote = document.getElementById('ep_approval_note');

    if (p.status === 'Đã duyệt' || p.status === 'Đã cấp phát') {
        approvedGroup.style.display = 'flex';
        inpApprQty.value = p.approved_quantity !== null ? p.approved_quantity : p.quantity;
        inpApprNote.value = p.approval_note || '';
    } else {
        approvedGroup.style.display = 'none';
        inpApprQty.value = '';
        inpApprNote.value = '';
    }

    const str = ((p.item_name || '') + ' ' + (p.specification || '')).toLowerCase();
    const is19L = (str.includes('19l') || str.includes('19 l') || str.includes('19 lít') || str.includes('20l'));
    const box = document.getElementById('ep_bottle_wrapper');
    const inp = document.getElementById('ep_return_bottle_count');

    if (is19L) {
        box.style.display = 'block';
        inp.value = parseInt(p.return_bottle_count) || 0;
    } else {
        box.style.display = 'none';
        inp.value = 0;
    }

    openModal('modal-edit-proposal');
}
