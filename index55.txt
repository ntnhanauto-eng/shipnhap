require_once 'db_connect.php';

$today = date('Y-m-d');

// 1. Thống kê hợp đồng (Sắp hết hạn trong 30 ngày & Đã quá hạn)
$contract_expiring_count = 0;
$contract_expired_count = 0;
try {
    $contract_expiring_count = $pdo->query("SELECT COUNT(*) FROM contracts WHERE status = 'Hiệu lực' AND contract_type != 'Không xác định thời hạn' AND end_date >= '$today' AND end_date <= DATE_ADD('$today', INTERVAL 30 DAY)")->fetchColumn();
    $contract_expired_count = $pdo->query("SELECT COUNT(*) FROM contracts WHERE status = 'Hiệu lực' AND contract_type != 'Không xác định thời hạn' AND end_date < '$today'")->fetchColumn();
} catch (PDOException $e) {}

// 2. Thống kê chấm công hôm nay (Đi làm, Nghỉ phép, Về sớm, Tăng ca)
$att_working_count = 0;
$att_leave_count = 0;
$att_early_count = 0;
$att_ot_count = 0;
try {
    $att_working_count = $pdo->query("SELECT COUNT(*) FROM attendance_details WHERE work_date = '$today' AND shift_status IN ('1.0', '0.5', 'CT')")->fetchColumn();
    $att_leave_count = $pdo->query("SELECT COUNT(*) FROM attendance_details WHERE work_date = '$today' AND shift_status IN ('PN', 'Ô', 'KP')")->fetchColumn();
    $att_early_count = $pdo->query("SELECT COUNT(*) FROM attendance_details WHERE work_date = '$today' AND early_minutes > 0")->fetchColumn();
    $att_ot_count = $pdo->query("SELECT COUNT(*) FROM attendance_details WHERE work_date = '$today' AND (ot_normal > 0 OR ot_weekend > 0)")->fetchColumn();
} catch (PDOException $e) {}

// 3. Thống kê số lượng suất ăn hôm nay (Đồng bộ chuẩn 100% theo logic của quan-ly-bep.php)
$today_lunch_count = 0;
try {
    $stmtLunch = $pdo->prepare("
        SELECT COUNT(*) FROM employees e
        JOIN attendance_details a ON e.id = a.employee_id
        WHERE a.work_date = :today
        AND a.shift_status IN ('1.0', '0.5', 'CT')
        AND (a.skip_lunch IS NULL OR a.skip_lunch = 0)
    ");
    $stmtLunch->execute([':today' => $today]);
    $today_lunch_count = $stmtLunch->fetchColumn();
} catch (PDOException $e) {
    $today_lunch_count = 0;
}

// 4. Thống kê đề xuất nước uống trong tuần hiện tại (Thứ 2 đến Chủ nhật)
$cur_week_code = date('Y-\WW');
$week_start_date = date('Y-m-d 00:00:00', strtotime('monday this week'));
$week_end_date   = date('Y-m-d 23:59:59', strtotime('sunday this week'));

$water_week_total = 0;
$water_week_wait  = 0;
$water_week_appr  = 0;
$water_week_rej   = 0;

try {
    $stmtWaterWeek = $pdo->prepare("
        SELECT 
            COUNT(*) AS total_req,
            COUNT(CASE WHEN status = 'Chờ duyệt' THEN 1 ELSE NULL END) AS total_wait,
            COUNT(CASE WHEN status IN ('Đã duyệt', 'Đã cấp phát', 'Chờ cấp phát') THEN 1 ELSE NULL END) AS total_appr,
            COUNT(CASE WHEN status = 'Từ chối' THEN 1 ELSE NULL END) AS total_rej
        FROM water_proposals 
        WHERE created_at BETWEEN :ws AND :we
    ");
    $stmtWaterWeek->execute([':ws' => $week_start_date, ':we' => $week_end_date]);
    $weekWaterData = $stmtWaterWeek->fetch(PDO::FETCH_ASSOC);

    if ($weekWaterData) {
        $water_week_total = intval($weekWaterData['total_req'] ?? 0);
        $water_week_wait  = intval($weekWaterData['total_wait'] ?? 0);
        $water_week_appr  = intval($weekWaterData['total_appr'] ?? 0);
        $water_week_rej   = intval($weekWaterData['total_rej'] ?? 0);
    }
} catch (PDOException $e) {}

// Lấy danh sách việc cần lưu ý gấp từ bộ phận nhân sự (nếu có bảng hr_tasks)
$hr_urgent_tasks = [];
try {
    $stmt = $pdo->query("SELECT task_name, task_url, is_urgent FROM hr_tasks WHERE status = 'pending' AND is_urgent = 1 ORDER BY id DESC LIMIT 3");
    $hr_urgent_tasks = $stmt->fetchAll();
} catch (PDOException $e) {
    $hr_urgent_tasks = [];
}

<script>
// Tự động đồng bộ ngầm toàn bộ 4 chỉ số đề xuất nước uống trong tuần mỗi 5 giây mà không cần F5
function autoCheckNewWaterProposals() {
    fetch('api-check-water.php?mode=week_stats')
        .then(response => response.json())
        .then(data => {
            if (data && data.success) {
                const elTotal = document.getElementById('water_stat_total');
                const elWait  = document.getElementById('water_stat_wait');
                const elAppr  = document.getElementById('water_stat_appr');
                const elRej   = document.getElementById('water_stat_rej');
                const box     = document.getElementById('water_notice_box');

                let hasChanged = false;

                // Cập nhật số lượng Tổng
                if (elTotal && data.water_week_total !== undefined) {
                    if (parseInt(elTotal.innerText) !== parseInt(data.water_week_total)) {
                        elTotal.innerText = data.water_week_total;
                        hasChanged = true;
                    }
                }
                // Cập nhật số lượng Chờ duyệt
                if (elWait && data.water_week_wait !== undefined) {
                    if (parseInt(elWait.innerText) !== parseInt(data.water_week_wait)) {
                        elWait.innerText = data.water_week_wait;
                        hasChanged = true;
                    }
                } else if (elWait && data.pending_water_count !== undefined) {
                    // Tương thích ngược nếu API trả về pending_water_count
                    if (parseInt(elWait.innerText) !== parseInt(data.pending_water_count)) {
                        elWait.innerText = data.pending_water_count;
                        hasChanged = true;
                    }
                }
                // Cập nhật số lượng Đã duyệt
                if (elAppr && data.water_week_appr !== undefined) {
                    if (parseInt(elAppr.innerText) !== parseInt(data.water_week_appr)) {
                        elAppr.innerText = data.water_week_appr;
                        hasChanged = true;
                    }
                }
                // Cập nhật số lượng Từ chối
                if (elRej && data.water_week_rej !== undefined) {
                    if (parseInt(elRej.innerText) !== parseInt(data.water_week_rej)) {
                        elRej.innerText = data.water_week_rej;
                        hasChanged = true;
                    }
                }

                // Hiệu ứng nhấp nháy báo hiệu khi dữ liệu có sự biến động
                if (hasChanged && box) {
                    box.style.transition = 'background-color 0.4s';
                    box.style.backgroundColor = '#e0f2fe';
                    setTimeout(() => {
                        box.style.backgroundColor = '#f8fafc';
                    }, 1000);
                }
            }
        })
        .catch(error => {
            // Không làm gián đoạn trải nghiệm người dùng khi mạng chập chờn
        });
}

setInterval(autoCheckNewWaterProposals, 5000);
</script>
