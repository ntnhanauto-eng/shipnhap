<script src="https://www.youtube.com/iframe_api"></script>
<script>
    // Nhận dữ liệu phụ đề từ PHP chuyển sang JSON trong JavaScript
    const subtitles = <?php echo $subtitles_json; ?>;
    let player;

    // Hàm này tự động chạy khi thư viện YouTube API tải xong
    function onYouTubeIframeAPIReady() {
        if ("<?php echo $video_id; ?>" !== "") {
            player = new YT.Player('player', {
                height: '100%',
                width: '100%',
                videoId: '<?php echo $video_id; ?>',
                playerVars: {
                    'playsinline': 1,
                    'controls': 1,
                    'cc_load_policy': 0 // Tắt phụ đề gốc của YouTube để không bị đè chữ
                },
                events: {
                    'onStateChange': onPlayerStateChange
                }
            });
        }
    }

    let subtitleInterval;

    function onPlayerStateChange(event) {
        // Nếu video đang phát (State = 1), kiểm tra thời gian mỗi 200ms để cập nhật phụ đề
        if (event.data == YT.PlayerState.PLAYING) {
            subtitleInterval = setInterval(updateSubtitles, 200);
        } else {
            clearInterval(subtitleInterval);
        }
    }

    function updateSubtitles() {
        if (!player) return;
        
        // Lấy thời gian hiện tại của video (tính bằng giây)
        const currentTime = player.getCurrentTime();
        
        // Tìm xem giây hiện tại có trùng với đoạn phụ đề nào không
        const currentSub = subtitles.find(sub => currentTime >= sub.start && currentTime <= (sub.start + sub.duration));
        
        if (currentSub) {
            document.getElementById('en-text').innerText = currentSub.en;
            document.getElementById('vi-text').innerText = currentSub.vi;
        } else {
            // Nếu không có thoại, xóa trắng hộp phụ đề
            document.getElementById('en-text').innerText = "";
            document.getElementById('vi-text').innerText = "";
        }
    }
</script>
