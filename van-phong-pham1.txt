function renderPaginationBar($currentPage, $totalPages, $paramName) {
    if ($totalPages <= 1) return;
    $queryParams = $_GET;

    echo '<div class="pagination-wrapper">';
    if ($currentPage > 1) {
        $queryParams[$paramName] = $currentPage - 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&laquo;</a>';
    } else {
        echo '<span class="page-link disabled">&laquo;</span>';
    }

    $start = max(1, $currentPage - 1);
    $end = min($totalPages, $currentPage + 1);
    if ($currentPage == 1) $end = min($totalPages, $start + 2);
    elseif ($currentPage == $totalPages) $start = max(1, $end - 2);

    for ($i = $start; $i <= $end; $i++) {
        $queryParams[$paramName] = $i;
        $activeClass = ($i == $currentPage) ? 'active' : '';
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link ' . $activeClass . '">' . $i . '</a>';
    }

    if ($currentPage < $totalPages) {
        $queryParams[$paramName] = $currentPage + 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&raquo;</a>';
    } else {
        echo '<span class="page-link disabled">&raquo;</span>';
    }
    echo '</div>';
}

function generateUniqueVppCode($pdo, $table, $timeStr, $excludeId = 0) {
    $ts = strtotime($timeStr) ?: time();
    $baseCode = date('dmYHis', $ts);

    $sql = "SELECT request_code FROM `{$table}` WHERE request_code LIKE :b AND id != :exId";
    $stmt = $pdo->prepare($sql);
    $stmt->execute([':b' => $baseCode . '%', ':exId' => $excludeId]);
    $existing = $stmt->fetchAll(PDO::FETCH_COLUMN);

    if (empty($existing) || !in_array($baseCode, $existing)) {
        return $baseCode;
    }

    $suffixes = range('A', 'Z');
    foreach ($suffixes as $char) {
        $candidate = $baseCode . $char;
        if (!in_array($candidate, $existing)) {
            return $candidate;
        }
    }
    return $baseCode . '_' . rand(10, 99);
}

function normalizeFastText($str) {
    $str = str_replace("\xC2\xA0", ' ', (string)$str);
    $str = mb_strtolower(trim($str), 'UTF-8');
    $str = preg_replace('/[áàảãạăắằẳẵặâấầẩẫậ]/u', 'a', $str);
    $str = preg_replace('/[éèẻẽẹêếềểễệ]/u', 'e', $str);
    $str = preg_replace('/[íìỉĩị]/u', 'i', $str);
    $str = preg_replace('/[óòỏõọôốồổỗộơớờởỡợ]/u', 'o', $str);
    $str = preg_replace('/[úùủũụưứừửữự]/u', 'u', $str);
    $str = preg_replace('/[ýỳỷỹỵ]/u', 'y', $str);
    $str = preg_replace('/đ/u', 'd', $str);
    return preg_replace('/[^a-z0-9]/', '', $str);
}

function parseExcelDate($rawDate, $defaultToday) {
    $rawDate = trim(str_replace("\xC2\xA0", ' ', (string)$rawDate));
    if (empty($rawDate)) return $defaultToday;

    if (is_numeric($rawDate) && intval($rawDate) > 30000 && intval($rawDate) < 60000) {
        $days = intval($rawDate) - 25569;
        return gmdate('Y-m-d', $days * 86400);
    }

    $rawDate = str_replace('/', '-', $rawDate);
    $ts = strtotime($rawDate);
    if ($ts && date('Y', $ts) > 2000) {
        return date('Y-m-d', $ts);
    }
    return $defaultToday;
}

function parseXlsxFastStream($filepath) {
    if (!class_exists('ZipArchive')) return null;
    $zip = new ZipArchive();
    if ($zip->open($filepath) !== true) return null;

    $strings = [];
    $sharedStrXml = $zip->getFromName('xl/sharedStrings.xml');
    if ($sharedStrXml) {
        $reader = new XMLReader();
        $reader->XML($sharedStrXml);
        $currText = '';
        $inT = false;
        while ($reader->read()) {
            if ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 'si') {
                $currText = '';
            } elseif ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 't') {
                $inT = true;
            } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 't') {
                $inT = false;
            } elseif ($inT && ($reader->nodeType === XMLReader::TEXT || $reader->nodeType === XMLReader::CDATA)) {
                $currText .= $reader->value;
            } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 'si') {
                $strings[] = $currText;
            }
        }
        $reader->close();
    }

    $sheetXml = $zip->getFromName('xl/worksheets/sheet1.xml');
    if (!$sheetXml) {
        $zip->close();
        return null;
    }

    $rows = [];
    $reader = new XMLReader();
    $reader->XML($sheetXml);

    $currentRow = [];
    $currentVal = '';
    $cellType = '';
    $inVal = false;

    while ($reader->read()) {
        if ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 'row') {
            $currentRow = [];
        } elseif ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 'c') {
            $cellType = $reader->getAttribute('t') ?? '';
            $currentVal = '';
        } elseif ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 'v') {
            $inVal = true;
        } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 'v') {
            $inVal = false;
        } elseif ($inVal && ($reader->nodeType === XMLReader::TEXT || $reader->nodeType === XMLReader::CDATA)) {
            $currentVal .= $reader->value;
        } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 'c') {
            $val = trim($currentVal);
            if ($cellType === 's' && isset($strings[intval($val)])) {
                $val = trim($strings[intval($val)]);
            }
            $currentRow[] = $val;
        } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 'row') {
            if (!empty(array_filter($currentRow, fn($v) => $v !== ''))) {
                $rows[] = $currentRow;
            }
        }
    }
    $reader->close();
    $zip->close();
    return $rows;
}

// 1.4 Cấp phát theo mã yêu cầu
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'deliver_group_proposals') {
    $req_code      = trim($_POST['request_code'] ?? '');
    $receiver_name = trim($_POST['receiver_name'] ?? '');
    $department    = trim($_POST['department'] ?? '');
    $delivery_loc  = trim($_POST['delivery_location'] ?? '');
    $custom_notes  = trim($_POST['notes'] ?? '');
    $deliver_items = $_POST['deliver_quantities'] ?? [];

    try {
        if (empty($req_code)) throw new Exception("Không xác định được mã yêu cầu đề xuất!");
        $pdo->beginTransaction();

        $stmtProps = $pdo->prepare("
            SELECT p.*, i.item_name, i.stock_qty, i.unit 
            FROM vpp_proposals p 
            JOIN vpp_items i ON p.item_id = i.id 
            WHERE p.request_code = :code AND p.status IN ('Đã duyệt', 'Chờ cấp phát')
            FOR UPDATE
        ");
        $stmtProps->execute([':code' => $req_code]);
        $rows = $stmtProps->fetchAll();

        if (empty($rows)) {
            throw new Exception("Yêu cầu đề xuất chưa được duyệt hoặc đã được cấp phát từ trước!");
        }

        $stmtSub = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty - :qty WHERE id = :id");
        $stmtReq = $pdo->prepare("
            INSERT INTO vpp_requests (request_code, request_date, department, receiver_name, item_id, quantity, delivery_location, notes)
            VALUES (:code, :r_date, :dept, :rec, :item_id, :qty, :loc, :notes)
        ");
        $stmtUpProp = $pdo->prepare("
            UPDATE vpp_proposals 
            SET status = 'Đã cấp phát', approved_quantity = :appr_q 
            WHERE id = :id
        ");

        $itemDeliveredCount = 0;
        foreach ($rows as $p) {
            $propId = $p['id'];
            $giveQty = isset($deliver_items[$propId]) ? max(0, intval($deliver_items[$propId])) : intval($p['approved_quantity'] !== null ? $p['approved_quantity'] : $p['quantity']);

            if ($giveQty <= 0) {
                $stmtUpProp->execute([':appr_q' => 0, ':id' => $propId]);
                continue;
            }

            if ($p['stock_qty'] < $giveQty) {
                throw new Exception("Tồn kho không đủ để cấp mặt hàng [{$p['item_name']}]! Tồn kho hiện có: {$p['stock_qty']} {$p['unit']}, cần cấp: {$giveQty} {$p['unit']}.");
            }

            $stmtSub->execute([':qty' => $giveQty, ':id' => $p['item_id']]);

            $stmtReq->execute([
                ':code'    => $req_code,
                ':r_date'  => date('Y-m-d'),
                ':dept'    => $department ?: $p['department'],
                ':rec'     => $receiver_name ?: $p['receiver_name'],
                ':item_id' => $p['item_id'],
                ':qty'     => $giveQty,
                ':loc'     => $delivery_loc ?: $p['delivery_location'],
                ':notes'   => $custom_notes
            ]);

            $stmtUpProp->execute([':appr_q' => $giveQty, ':id' => $propId]);
            $itemDeliveredCount++;
        }

        $pdo->commit();
        $message = "Đã xác nhận cấp phát thành công {$itemDeliveredCount} mặt hàng cho yêu cầu [{$req_code}]!";
    } catch (Exception $e) {
        if ($pdo->inTransaction()) $pdo->rollBack();
        $error = "Lỗi cấp phát: " . $e->getMessage();
    }
}

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'update_request') {
    $req_id   = intval($_POST['request_id']);
    $new_date = $_POST['request_date'];
    $new_dept = trim($_POST['department']);
    $new_rec  = trim($_POST['receiver_name'] ?? '');
    $new_item = intval($_POST['item_id']);
    $new_qty  = intval($_POST['quantity']);
    $new_loc  = trim($_POST['delivery_location'] ?? '');
    $new_notes= trim($_POST['notes'] ?? '');

    try {
        $pdo->beginTransaction();
        $stmtOld = $pdo->prepare("SELECT * FROM vpp_requests WHERE id = :id FOR UPDATE");
        $stmtOld->execute([':id' => $req_id]);
        $oldReq = $stmtOld->fetch();

        if (!$oldReq) throw new Exception("Bản ghi cấp phát không tồn tại!");

        $stmtIt = $pdo->prepare("SELECT item_name, specification, stock_qty, unit FROM vpp_items WHERE id = :id FOR UPDATE");
        $stmtIt->execute([':id' => $new_item]);
        $it = $stmtIt->fetch();
        if (!$it) throw new Exception("Mặt hàng VPP không tồn tại!");

        if (intval($oldReq['item_id']) === $new_item) {
            $diff = $new_qty - intval($oldReq['quantity']);
            if ($diff > 0 && $diff > $it['stock_qty']) {
                throw new Exception("Không đủ hàng trong kho để tăng số lượng cấp! Còn thiếu {$diff} {$it['unit']}.");
            }
            $stmtUpStock = $pdo->prepare("UPDATE vpp_items SET stock_qty = GREATEST(0, stock_qty - :diff) WHERE id = :id");
            $stmtUpStock->execute([':diff' => $diff, ':id' => $new_item]);
        } else {
            $stmtBack = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty + :q WHERE id = :id");
            $stmtBack->execute([':q' => $oldReq['quantity'], ':id' => $oldReq['item_id']]);

            if ($new_qty > $it['stock_qty']) {
                throw new Exception("Không đủ tồn kho mặt hàng mới để cấp! Tồn: {$it['stock_qty']} {$it['unit']}.");
            }
            $stmtTake = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty - :q WHERE id = :id");
            $stmtTake->execute([':q' => $new_qty, ':id' => $new_item]);
        }

        $stmtUpReq = $pdo->prepare("
            UPDATE vpp_requests 
            SET request_date = :r_date, department = :dept, receiver_name = :rec, item_id = :item_id, 
                quantity = :qty, delivery_location = :loc, notes = :notes 
            WHERE id = :id
        ");
        $stmtUpReq->execute([
            ':r_date'  => $new_date, ':dept'    => $new_dept, ':rec'     => $new_rec,
            ':item_id' => $new_item, ':qty'     => $new_qty,
            ':loc'     => $new_loc,  ':notes'   => $new_notes, ':id'     => $req_id
        ]);

        $pdo->commit();
        $message = "Đã cập nhật bản ghi lịch sử cấp phát VPP thành công!";
    } catch (Exception $e) {
        if ($pdo->inTransaction()) $pdo->rollBack();
        $error = "Lỗi cập nhật: " . $e->getMessage();
    }
}

        COUNT(DISTINCT CASE WHEN status = 'Từ chối' THEN request_code ELSE NULL END) AS count_rej_week
    FROM vpp_proposals 
    WHERE created_at BETWEEN :ws AND :we
");
$weekPropStmt->execute([':ws' => $week_start_date, ':we' => $week_end_date]);

$paramsP = [];

if (!empty($f_prop_search)) {
    $whereP[] = "(p.request_code LIKE :ps1 
                 OR DATE_FORMAT(p.created_at, '%d%m%Y%H%i%s') LIKE :ps2 
                 OR p.receiver_name LIKE :ps3 
                 OR i.item_name LIKE :ps4)";
    $paramsP[':ps1'] = "%$f_prop_search%";
    $paramsP[':ps2'] = "%$f_prop_search%";
    $paramsP[':ps3'] = "%$f_prop_search%";
    $paramsP[':ps4'] = "%$f_prop_search%";
}

if (!empty($f_prop_dept)) {
    $whereP[] = "p.department = :pdept";
    $paramsP[':pdept'] = $f_prop_dept;
}

if (!empty($f_prop_status)) {
    if ($f_prop_status === 'Chờ duyệt') {
        $whereP[] = "p.status = 'Chờ duyệt'";
    } elseif ($f_prop_status === 'Đã sơ duyệt') {
        $whereP[] = "p.status = 'Đã sơ duyệt'";
    } elseif ($f_prop_status === 'Đã duyệt') {
        $whereP[] = "p.status IN ('Đã duyệt', 'Đã cấp phát', 'Chờ cấp phát')";
    } elseif ($f_prop_status === 'Từ chối') {
        $whereP[] = "p.status = 'Từ chối'";
    }
}

if ($f_prop_mode === 'day') {
    $whereP[] = "DATE(p.created_at) = :pd";
    $paramsP[':pd'] = $f_prop_date;
} elseif ($f_prop_mode === 'week') {
    $yw = substr($f_prop_week, 0, 4);
    $ww = substr($f_prop_week, 6);
    $dto = new DateTime();
    $dto->setISODate(intval($yw), intval($ww));
    $ws = $dto->format('Y-m-d 00:00:00');
    $dto->modify('+6 days');
    $we = $dto->format('Y-m-d 23:59:59');
    $whereP[] = "p.created_at BETWEEN :pws AND :pwe";
    $paramsP[':pws'] = $ws;
    $paramsP[':pwe'] = $we;
} elseif ($f_prop_mode === 'year') {
    $whereP[] = "DATE_FORMAT(p.created_at, '%Y') = :py";
    $paramsP[':py'] = $f_prop_year;
} else {
    $whereP[] = "DATE_FORMAT(p.created_at, '%Y-%m') = :pm";

    JOIN vpp_items i ON p.item_id = i.id
    WHERE " . implode(' AND ', $whereP) . "
    GROUP BY p.request_code, p.employee_id, p.receiver_name, p.department, p.delivery_location, p.status, p.reject_reason, p.created_at
    ORDER BY (p.status = 'Chờ duyệt') DESC, (p.status = 'Đã sơ duyệt') DESC, (p.status = 'Đã duyệt' OR p.status = 'Chờ cấp phát') DESC, MAX(p.id) DESC

<script>
function openModal(id) { document.getElementById(id).style.display = 'flex'; }
function closeModal(id) { document.getElementById(id).style.display = 'none'; }

function togglePropDropdown(e, code) {
    if (e) e.stopPropagation();
    const currentMenu = document.getElementById('prop_dropdown_' + code);
    if (!currentMenu) return;
    const isShown = currentMenu.classList.contains('show');
    closeAllPropDropdowns();
    closeAllPencilDropdowns();
    if (!isShown) currentMenu.classList.add('show');
}
function closeAllPropDropdowns() {
    document.querySelectorAll('.prop-dropdown-menu').forEach(el => el.classList.remove('show'));
}

function togglePropPencilDropdown(e, code) {
    if (e) e.stopPropagation();
    const currentMenu = document.getElementById('prop_pencil_menu_' + code);
    if (!currentMenu) return;
    const isShown = currentMenu.classList.contains('show');
    closeAllPropDropdowns();
    closeAllPencilDropdowns();
    if (!isShown) currentMenu.classList.add('show');
}
function closeAllPencilDropdowns() {
    document.querySelectorAll('.prop-pencil-menu').forEach(el => el.classList.remove('show'));
}

window.onclick = function(e) {
    if (e.target.classList.contains('modal-overlay')) e.target.style.display = 'none';
    if (!e.target.closest('.prop-dropdown')) closeAllPropDropdowns();
    if (!e.target.closest('.prop-pencil-dropdown')) closeAllPencilDropdowns();
}

// 2. MỞ CỬA SỔ XEM & IN BIỂU MẪU TRÌNH KÝ
function openPrintDocModal(p) {
    document.getElementById('prn_so_phieu').innerText = p.request_code;
    document.getElementById('prn_receiver_name').innerText = p.receiver_name;
    document.getElementById('prn_sign_receiver').innerText = p.receiver_name;
    
    const empInfo = p.employee_id ? `#${p.employee_id} / ${p.department}` : p.department;
    document.getElementById('prn_dept_info').innerText = empInfo;

    const tbody = document.getElementById('prn_items_tbody');
    tbody.innerHTML = '';
    const items = JSON.parse(p.items_json || '[]');
    let totalQty = 0;

    items.forEach((it, idx) => {
        const qtyToIssue = (it.pre_approved_quantity !== null) ? parseInt(it.pre_approved_quantity) : parseInt(it.quantity);
        totalQty += qtyToIssue;
        const stockLeft = Math.max(0, parseInt(it.stock_qty) - qtyToIssue);

        const tr = document.createElement('tr');
        tr.innerHTML = `
            <td>${idx + 1}</td>
            <td>${it.item_code || '—'}</td>
            <td class="text-left"><b>${it.item_name}</b> ${it.specification ? '(' + it.specification + ')' : ''}</td>
            <td>${it.unit}</td>
            <td style="font-weight:bold;">${qtyToIssue}</td>
            <td>${stockLeft}</td>
            <td></td>
        `;
        tbody.appendChild(tr);
    });

    // Điền thêm các dòng trống để biểu mẫu đầy đặn (tối thiểu 6 dòng như mẫu)
    const emptyRowsCount = Math.max(0, 6 - items.length);
    for (let i = 0; i < emptyRowsCount; i++) {
        const tr = document.createElement('tr');
        tr.innerHTML = `
            <td>${items.length + i + 1}</td>
            <td></td><td></td><td></td><td></td><td></td><td></td>
        `;
        tbody.appendChild(tr);
    }

    document.getElementById('prn_total_qty').innerText = totalQty;
    openModal('modal-print-view');
}

// 4. MỞ MODAL XEM CHI TIẾT CÁC MẶT HÀNG
function openProposalDetailsModal(p) {
    document.getElementById('vd_req_code').innerText = p.request_code;
    document.getElementById('vd_receiver').innerText = p.receiver_name + (p.employee_id ? ' (#' + p.employee_id + ')' : '');
    document.getElementById('vd_dept').innerText = p.department;

    const tbody = document.getElementById('vd_items_tbody');
    tbody.innerHTML = '';
    const items = JSON.parse(p.items_json || '[]');

    items.forEach((it, idx) => {
        const tr = document.createElement('tr');
        const apprText = (it.approved_quantity !== null) ? it.approved_quantity : ((it.pre_approved_quantity !== null) ? it.pre_approved_quantity + ' (Sơ duyệt)' : '--');
        tr.innerHTML = `
            <td style="text-align:center; color:#64748b; font-weight:bold;">${idx + 1}</td>
            <td><span style="background:#f1f5f9; padding:2px 5px; border-radius:4px; font-weight:600;">${it.item_code || '—'}</span></td>
            <td><b>${it.item_name}</b> ${it.specification ? '<br><span style="font-size:0.75rem; color:#64748b;">(' + it.specification + ')</span>' : ''}</td>
            <td style="text-align:center; font-weight:600;">${it.unit}</td>
            <td style="text-align:center; font-weight:bold; color:#0284c7;">${it.quantity}</td>
            <td style="text-align:center; font-weight:bold; color:#15803d;">${apprText}</td>
            <td style="text-align:center; font-weight:bold; ${it.stock_qty > 0 ? 'color:#15803d;' : 'color:#dc2626;'}">${it.stock_qty}</td>
        `;
        tbody.appendChild(tr);
    });

    openModal('modal-view-details');
}

// 5. CẤP PHÁT & TỪ CHỐI
function openDeliverGroupModal(p) {
    document.getElementById('dlg_request_code').value = p.request_code;
    document.getElementById('dlg_receiver').value = p.receiver_name;
    document.getElementById('dlg_department').value = p.department;

    const tbody = document.getElementById('dlg_items_tbody');
    tbody.innerHTML = '';
    const items = JSON.parse(p.items_json || '[]');

    items.forEach(it => {
        const tr = document.createElement('tr');
        const defaultQty = it.approved_quantity !== null ? it.approved_quantity : it.quantity;
        tr.innerHTML = `
            <td><b>${it.item_name}</b> (${it.unit})</td>
            <td style="text-align:center; font-weight:bold;">${it.stock_qty}</td>
            <td style="text-align:center; color:#15803d; font-weight:bold;">${defaultQty}</td>
            <td style="text-align:center;">
                <input type="number" name="deliver_quantities[${it.proposal_id}]" min="0" value="${defaultQty}" required
                       style="width:75px; padding:4px 6px; border:1.5px solid #10b981; border-radius:4px; text-align:center; font-weight:bold;">
            </td>
        `;
        tbody.appendChild(tr);
    });

    openModal('modal-deliver-group');
}
