// Khởi tạo bảng đề xuất mua VPP nếu chưa tồn tại
try {
    $pdo->exec("
    CREATE TABLE IF NOT EXISTS `vpp_purchase_proposals` (
        `id` INT AUTO_INCREMENT PRIMARY KEY,
        `request_code` VARCHAR(60) DEFAULT '',
        `employee_id` INT DEFAULT NULL,
        `receiver_name` VARCHAR(150) NOT NULL,
        `department` VARCHAR(100) NOT NULL,
        `item_id` INT DEFAULT NULL,
        `custom_item_name` VARCHAR(150) DEFAULT '',
        `specification` VARCHAR(100) DEFAULT '',
        `unit` VARCHAR(50) DEFAULT 'Cái',
        `quantity` INT NOT NULL DEFAULT 1,
        `pre_approved_quantity` INT DEFAULT NULL,
        `approved_quantity` INT DEFAULT NULL,
        `approval_note` VARCHAR(255) DEFAULT '',
        `status` VARCHAR(50) DEFAULT 'Chờ duyệt',
        `reject_reason` VARCHAR(255) DEFAULT '',
        `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
        INDEX (`item_id`),
        INDEX (`status`),
        INDEX (`request_code`)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");
} catch (PDOException $e) {}

$allItems = $pdo->query("SELECT * FROM vpp_items ORDER BY item_name ASC")->fetchAll();

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'submit_purchase_proposal') {
    $employee_id = !empty($_POST['employee_id']) ? intval($_POST['employee_id']) : null;
    $receiver_name = trim($_POST['receiver_name'] ?? '');
    $department = trim($_POST['department'] ?? '');
    $items_data = $_POST['items'] ?? [];

    try {
        if (empty($receiver_name) || empty($department)) {
            throw new Exception("Vui lòng điền đầy đủ Họ tên người đề xuất và Bộ phận / Phòng ban!");
        }
        if (empty($items_data)) {
            throw new Exception("Vui lòng chọn hoặc điền ít nhất một mặt hàng cần đề xuất mua!");
        }

        $baseCode = 'MUA-' . date('dmYHis');
        $pdo->beginTransaction();

        $stmtIns = $pdo->prepare("
            INSERT INTO vpp_purchase_proposals 
            (request_code, employee_id, receiver_name, department, item_id, custom_item_name, specification, unit, quantity, status)
            VALUES (:code, :empid, :rec, :dept, :it_id, :c_name, :spec, :unit, :qty, 'Chờ duyệt')
        ");

        $insertedCount = 0;
        foreach ($items_data as $row) {
            $itemId = intval($row['item_id'] ?? 0);
            $qty = intval($row['quantity'] ?? 0);
            if ($qty <= 0) continue;

            $customName = trim($row['custom_item_name'] ?? '');
            $spec = trim($row['specification'] ?? '');
            $unit = trim($row['unit'] ?? 'Cái');

            $stmtIns->execute([
                ':code' => $baseCode,
                ':empid' => $employee_id,
                ':rec' => $receiver_name,
                ':dept' => $department,
                ':it_id' => ($itemId > 0 ? $itemId : null),
                ':c_name' => ($itemId > 0 ? '' : $customName),
                ':spec' => $spec,
                ':unit' => $unit,
                ':qty' => $qty
            ]);
            $insertedCount++;
        }

        if ($insertedCount === 0) {
            throw new Exception("Số lượng đề xuất của các mặt hàng không hợp lệ!");
        }

        $pdo->commit();
        $message = "Gửi đề xuất mua văn phòng phẩm thành công! Mã phiếu của bạn là: <strong>{$baseCode}</strong>";
    } catch (Exception $e) {
        if ($pdo->inTransaction()) $pdo->rollBack();
        $error = "Lỗi gửi đề xuất: " . $e->getMessage();
    }
}

<script>
let rowIndex = 0;
const allItemsData = <?= json_encode($allItems) ?>;

function handleItemSelect(select) {
const row = select.closest('tr');
const customInput = row.querySelector('input[name*="[custom_item_name]"]');
const specInput = row.querySelector('input[name*="[specification]"]');
const unitInput = row.querySelector('input[name*="[unit]"]');

if (select.value === "0") {
customInput.style.display = "block";
customInput.value = "";
customInput.required = true;
specInput.value = "";
unitInput.value = "Cái";
} else {
customInput.style.display = "none";
customInput.required = false;
const selectedOption = select.options[select.selectedIndex];
customInput.value = selectedOption.getAttribute('data-name');
specInput.value = selectedOption.getAttribute('data-spec') || '';
unitInput.value = selectedOption.getAttribute('data-unit') || 'Cái';
}
}

function addItemRow() {
rowIndex++;
const tbody = document.getElementById('items_tbody');
const tr = document.createElement('tr');
let optionsHtml = '<option value="0">-- Nhập mặt hàng mới bên ngoài --</option>';
allItemsData.forEach(it => {
optionsHtml += `<option value="${it.id}" data-name="${it.item_name}" data-spec="${it.specification || ''}" data-unit="${it.unit || 'Cái'}">${it.item_name} (Tồn kho: ${it.stock_qty} ${it.unit})</option>`;
});

tr.innerHTML = `
<td>
<select name="items[${rowIndex}][item_id]" onchange="handleItemSelect(this)">
${optionsHtml}
</select>
<input type="text" name="items[${rowIndex}][custom_item_name]" placeholder="Nhập tên mặt hàng..." style="margin-top: 5px;">
</td>
<td><input type="text" name="items[${rowIndex}][specification]" placeholder="VD: Ram 500 tờ..."></td>
<td><input type="text" name="items[${rowIndex}][unit]" value="Cái" placeholder="ĐVT..."></td>
<td style="text-align: center;"><input type="number" name="items[${rowIndex}][quantity]" min="1" value="1" required style="text-align: center;"></td>
<td style="text-align: center;"><button type="button" class="btn-del" onclick="removeRow(this)">✕</button></td>
`;
tbody.appendChild(tr);
handleItemSelect(tr.querySelector('select'));
}

function removeRow(btn) {
const tbody = document.getElementById('items_tbody');
if (tbody.rows.length > 1) {
btn.closest('tr').remove();
} else {
alert("Cần phải có ít nhất một mặt hàng đề xuất!");
}
}

document.addEventListener("DOMContentLoaded", () => {
document.querySelectorAll('#items_tbody tr').forEach(tr => {
const select = tr.querySelector('select');
if(select) handleItemSelect(select);
});
});
