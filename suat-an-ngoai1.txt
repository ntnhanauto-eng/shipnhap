        $pdo->exec("ALTER TABLE `meal_attendance` DROP INDEX `uniq_emp_date`");
    }
    if (!in_array('uniq_emp_record', $idxNames)) {
        $pdo->exec("ALTER TABLE `meal_attendance` ADD UNIQUE KEY `uniq_emp_record` (`record_date`, `emp_no`)");
    }

    // Tự động kiểm tra bổ sung cột
    $cols = $pdo->query("SHOW COLUMNS FROM `meal_attendance`")->fetchAll(PDO::FETCH_COLUMN);
    if (!in_array('first_name', $cols))  $pdo->exec("ALTER TABLE `meal_attendance` ADD COLUMN `first_name` VARCHAR(50) DEFAULT '' AFTER `full_name`");
    if (!in_array('last_name', $cols))   $pdo->exec("ALTER TABLE `meal_attendance` ADD COLUMN `last_name` VARCHAR(100) DEFAULT '' AFTER `first_name`");
    if (!in_array('emp_no', $cols))      $pdo->exec("ALTER TABLE `meal_attendance` ADD COLUMN `emp_no` VARCHAR(50) NOT NULL DEFAULT '' AFTER `last_name`");
    if (!in_array('state', $cols))       $pdo->exec("ALTER TABLE `meal_attendance` ADD COLUMN `state` VARCHAR(50) DEFAULT '' AFTER `check_time`");
    if (!in_array('new_state', $cols))   $pdo->exec("ALTER TABLE `meal_attendance` ADD COLUMN `new_state` VARCHAR(50) DEFAULT '' AFTER `state`");
    if (!in_array('exception', $cols))   $pdo->exec("ALTER TABLE `meal_attendance` ADD COLUMN `exception` VARCHAR(100) DEFAULT '' AFTER `new_state`");
    if (!in_array('operation', $cols))   $pdo->exec("ALTER TABLE `meal_attendance` ADD COLUMN `operation` VARCHAR(100) DEFAULT '' AFTER `exception`");

$message = '';
$error = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $action = $_POST['action'] ?? '';

    // 1. Lưu danh sách sau khi xem trước (Gộp 1 ID nhiều lần quẹt thành 1 người)
    if ($action === 'confirm_import_excel') {
        $importDate = $_POST['import_date'] ?? $today;
        $jsonData = $_POST['excel_rows_json'] ?? '';
        $rows = json_decode($jsonData, true);

        if (!empty($rows) && is_array($rows)) {
            try {
                // Gom nhóm theo emp_no (ID): Quẹt nhiều lần trong ngày chỉ lấy 1 người
                $dedupRows = [];
                foreach ($rows as $r) {
                    $no = trim((string)($r['no'] ?? ''));
                    if ($no === '') continue;

                    if (!isset($dedupRows[$no])) {
                        $dedupRows[$no] = $r;
                    } else {
                        // Nếu dòng sau có giờ quẹt sớm hơn thì cập nhật giờ sớm nhất
                        if (!empty($r['time']) && (empty($dedupRows[$no]['time']) || $r['time'] < $dedupRows[$no]['time'])) {
                            $dedupRows[$no]['time'] = $r['time'];
                        }
                    }
                }

                $pdo->beginTransaction();

                // Dùng ON DUPLICATE KEY UPDATE để không bao giờ bị lỗi trùng khóa 1062
                $stmtUpsert = $pdo->prepare("
                    INSERT INTO meal_attendance 
                    (record_date, department, full_name, first_name, last_name, emp_no, check_time, state, new_state, exception, operation, is_canceled, cancel_reason)
                    VALUES 
                    (:d, :dept, :fn, :fname, :lname, :no, :t, :st, :nst, :exp, :op, 0, '')
                    ON DUPLICATE KEY UPDATE
                        department = VALUES(department),
                        full_name = VALUES(full_name),
                        first_name = VALUES(first_name),
                        last_name = VALUES(last_name),
                        check_time = VALUES(check_time),
                        state = VALUES(state),
                        new_state = VALUES(new_state),
                        exception = VALUES(exception),
                        operation = VALUES(operation)
                ");

                $countProcessed = 0;

                foreach ($dedupRows as $no => $r) {
                    $dept  = trim((string)($r['department'] ?? ''));
                    $fn    = trim((string)($r['name'] ?? ''));
                    $t     = trim((string)($r['time'] ?? ''));
                    $st    = trim((string)($r['state'] ?? ''));
                    $nst   = trim((string)($r['new_state'] ?? ''));
                    $exp   = trim((string)($r['exception'] ?? ''));
                    $op    = trim((string)($r['operation'] ?? ''));

                    if (empty($no) || empty($fn)) continue;

                    // Tách tên
                    $parts = preg_split('/\s+/', $fn);
                    $fname = count($parts) > 0 ? end($parts) : $fn;
                    $lname = count($parts) > 1 ? implode(' ', array_slice($parts, 0, -1)) : '';

                    $stmtUpsert->execute([
                        ':d'     => $importDate,
                        ':dept'  => $dept,
                        ':fn'    => $fn,
                        ':fname' => $fname,
                        ':lname' => $lname,
                        ':no'    => $no,
                        ':t'     => $t,
                        ':st'    => $st,
                        ':nst'   => $nst,
                        ':exp'   => $exp,
                        ':op'    => $op
                    ]);
                    $stmtUpsert->closeCursor();
                    $countProcessed++;
                }

                $pdo->commit();
                $message = "Đã nhập thành công dữ liệu ngày [{$importDate}]: Ghi nhận chính xác {$countProcessed} nhân viên đi làm (các lượt quẹt trùng lặp đã tự động gộp thành 1)!";
            } catch (Exception $e) {
                if ($pdo->inTransaction()) {
                    $pdo->rollBack();
                }
                $error = "Lỗi khi lưu dữ liệu: " . $e->getMessage();
            }
        } else {
            $error = "Không có dữ liệu hợp lệ để nhập!";
        }
    }

    // 2. Cập nhật sửa dòng
    if ($action === 'update_row') {
        $id = intval($_POST['id'] ?? 0);
        $dept = trim($_POST['department'] ?? '');
        $fn   = trim($_POST['full_name'] ?? '');
        $no   = trim($_POST['emp_no'] ?? '');
        $t    = trim($_POST['check_time'] ?? '');
        $st   = trim($_POST['state'] ?? '');
        $nst  = trim($_POST['new_state'] ?? '');
        $exp  = trim($_POST['exception'] ?? '');
        $op   = trim($_POST['operation'] ?? '');
        $is_cancel = isset($_POST['is_canceled']) ? 1 : 0;
        $reason = trim($_POST['cancel_reason'] ?? '');

        if ($id > 0 && !empty($fn)) {
            $parts = preg_split('/\s+/', $fn);
            $fname = count($parts) > 0 ? end($parts) : $fn;
            $lname = count($parts) > 1 ? implode(' ', array_slice($parts, 0, -1)) : '';

            $stmt = $pdo->prepare("
                UPDATE meal_attendance 
                SET department = :dept, full_name = :fn, first_name = :fname, last_name = :lname, emp_no = :no, check_time = :t,
                    state = :st, new_state = :nst, exception = :exp, operation = :op, is_canceled = :ic, cancel_reason = :cr
                WHERE id = :id
            ");
            $stmt->execute([
                ':dept' => $dept, ':fn' => $fn, ':fname' => $fname, ':lname' => $lname, ':no' => $no,
                ':t' => $t, ':st' => $st, ':nst' => $nst, ':exp' => $exp, ':op' => $op,
                ':ic' => $is_cancel, ':cr' => $reason, ':id' => $id
            ]);
            $stmt->closeCursor();
            $message = "Đã cập nhật thông tin nhân viên [{$fn}] thành công!";
        }
    }

    // 3. Xóa dòng
    if ($action === 'delete_row') {
        $id = intval($_POST['id'] ?? 0);
        if ($id > 0) {
            $stmt = $pdo->prepare("DELETE FROM meal_attendance WHERE id = :id");
            $stmt->execute([':id' => $id]);
            $stmt->closeCursor();
            $message = "Đã xóa bản ghi chấm công suất ăn!";
        }
    }
}

    $where[] = "first_name LIKE :fname";

if (!empty($f_last_name)) {
    $where[] = "last_name LIKE :lname";
    $params[':lname'] = "%{$f_last_name}%";
}
if (!empty($f_emp_no)) {
    $where[] = "emp_no LIKE :no";
    $params[':no'] = "%{$f_emp_no}%";
}
if (!empty($f_dept)) {
    $where[] = "department = :dept";
    $params[':dept'] = $f_dept;
}

$whereSql = implode(' AND ', $where);

$stmtFilterSum = $pdo->prepare("
    SELECT 
        COUNT(*) AS total_count,
        SUM(CASE WHEN is_canceled = 1 THEN 1 ELSE 0 END) AS total_canceled,
        SUM(CASE WHEN is_canceled = 0 THEN 1 ELSE 0 END) AS total_active
    FROM meal_attendance 
    WHERE {$whereSql}
");

$filterSumData = $stmtFilterSum->fetch(PDO::FETCH_ASSOC);
$stmtFilterSum->closeCursor();

$total_records   = intval($filterSumData['total_count'] ?? 0);
$total_canceled  = intval($filterSumData['total_canceled'] ?? 0);
$total_active    = intval($filterSumData['total_active'] ?? 0);

$totalPages = ceil($total_records / $limit) ?: 1;
if ($p_meal > $totalPages) $p_meal = $totalPages;
$offset = ($p_meal - 1) * $limit;

// Danh sách trang hiện tại
$stmtList = $pdo->prepare("

    WHERE {$whereSql}
    ORDER BY department ASC, emp_no ASC
    LIMIT {$limit} OFFSET {$offset}
");
$stmtList->execute($params);
$attendanceList = $stmtList->fetchAll(PDO::FETCH_ASSOC);
$stmtList->closeCursor();

// Danh sách phòng ban
$stmtDepts = $pdo->prepare("SELECT DISTINCT department FROM meal_attendance WHERE record_date = :d AND department != '' ORDER BY department ASC");
$stmtDepts->execute([':d' => $f_date]);
$departmentList = $stmtDepts->fetchAll(PDO::FETCH_COLUMN);
$stmtDepts->closeCursor();

// Thống kê phòng ban (Khối 2)
$stmtStat = $pdo->prepare("
    SELECT 
        department,
        COUNT(*) AS total_working,
        SUM(CASE WHEN is_canceled = 1 THEN 1 ELSE 0 END) AS total_canceled,
        SUM(CASE WHEN is_canceled = 0 THEN 1 ELSE 0 END) AS total_meals
    FROM meal_attendance
    WHERE record_date = :d
    GROUP BY department
    ORDER BY department ASC
");
$stmtStat->execute([':d' => $f_date]);
$statsByDept = $stmtStat->fetchAll(PDO::FETCH_ASSOC);
$stmtStat->closeCursor();

$allDeptWorking = 0;

foreach ($statsByDept as $sb) {
    $allDeptWorking += $sb['total_working'];

    $allDeptMeals += $sb['total_meals'];
}

// Xử lý file Excel: Tìm theo ID (No.), 1 người quẹt nhiều lần chỉ tính 1 người
function handleExcelUpload(input) {
    const file = input.files[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = function(e) {
        try {
            const data = new Uint8Array(e.target.result);
            const workbook = XLSX.read(data, { type: 'array' });
            const firstSheetName = workbook.SheetNames[0];
            const worksheet = workbook.Sheets[firstSheetName];
            const jsonSheet = XLSX.utils.sheet_to_json(worksheet, { header: 1, defval: '' });

            if (!jsonSheet || jsonSheet.length < 2) {
                alert('File Excel không có dữ liệu hợp lệ!');
                return;
            }

            let headerRowIndex = 0;
            let colDept = -1, colName = -1, colNo = -1, colDate = -1, colTime = -1;
            let colState = -1, colNewState = -1, colException = -1, colOperation = -1;

            // Quét tìm vị trí các cột
            for (let r = 0; r < Math.min(15, jsonSheet.length); r++) {
                const row = jsonSheet[r];
                for (let c = 0; c < row.length; c++) {
                    const val = String(row[c]).toLowerCase().trim();
                    if (val === 'department' || val.includes('phòng')) colDept = c;
                    else if (val === 'name' || val.includes('tên') || val.includes('họ tên')) colName = c;
                    else if (val === 'no.' || val === 'no' || val.includes('mã nv') || val.includes('user id') || val.includes('id')) colNo = c;
                    else if (val === 'date' || val.includes('ngày')) colDate = c;
                    else if (val === 'time' || val.includes('giờ')) colTime = c;
                    else if (val === 'state') colState = c;
                    else if (val === 'new state' || val === 'newstate') colNewState = c;
                    else if (val === 'exception') colException = c;
                    else if (val === 'operation') colOperation = c;
                }
                if (colName !== -1 && (colNo !== -1 || colTime !== -1)) {
                    headerRowIndex = r;
                    break;
                }
            }

            // Thứ tự cột fallback
            if (colDept === -1)      colDept = 0;
            if (colName === -1)      colName = 1;
            if (colNo === -1)        colNo = 2;
            if (colDate === -1)      colDate = 3;
            if (colTime === -1)      colTime = 4;
            if (colState === -1)     colState = 5;
            if (colNewState === -1)  colNewState = 6;
            if (colException === -1) colException = 7;
            if (colOperation === -1) colOperation = 8;

            // Dùng Map theo No. để 1 người quẹt nhiều lần chỉ tính là 1 người
            const uniqueIdMap = {};
            let detectedDate = '<?= $f_date ?>';

            for (let r = headerRowIndex + 1; r < jsonSheet.length; r++) {
                const row = jsonSheet[r];
                if (!row || row.length === 0) continue;

                const dept      = String(row[colDept] || '').trim();
                const name      = String(row[colName] || '').trim();
                const no        = String(row[colNo] || '').trim();
                const dateVal   = String(row[colDate] || '').trim();
                const timeVal   = String(row[colTime] || '').trim();
                const state     = String(row[colState] || '').trim();
                const newState  = String(row[colNewState] || '').trim();
                const exception = String(row[colException] || '').trim();
                const operation = String(row[colOperation] || '').trim();

                // Điều kiện: Có giờ quẹt (Time khác rỗng) và có No. (ID)
                if (timeVal !== '' && name !== '' && no !== '') {
                    if (!uniqueIdMap[no]) {
                        // Thêm mới nếu chưa có ID này
                        uniqueIdMap[no] = {
                            department: dept,
                            name: name,
                            no: no,
                            date: dateVal,
                            time: timeVal,
                            state: state,
                            new_state: newState,
                            exception: exception,
                            operation: operation
                        };
                    } else {
                        // Nếu ID đã có: So sánh và giữ lại giờ quẹt sớm nhất trong ngày
                        if (timeVal < uniqueIdMap[no].time) {
                            uniqueIdMap[no].time = timeVal;
                            uniqueIdMap[no].state = state;
                        }
                    }

                    if (dateVal && dateVal.includes('-') && dateVal.length >= 8) {
                        detectedDate = dateVal;
                    }
                }
            }

            const validRows = Object.values(uniqueIdMap);

            if (validRows.length === 0) {
                alert('Không tìm thấy dòng nào có giờ chấm công hợp lệ!');
                input.value = '';
                return;
            }

            document.getElementById('preview_import_date_val').value = detectedDate;
            document.getElementById('preview_date_text').innerText = detectedDate;
            document.getElementById('preview_count_text').innerText = validRows.length;
            document.getElementById('preview_json_val').value = JSON.stringify(validRows);

            const tbody = document.getElementById('preview_tbody');
            tbody.innerHTML = '';
            validRows.forEach((item, index) => {
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td style="text-align:center; font-weight:bold; color:#64748b;">${index + 1}</td>
                    <td>${item.department}</td>
                    <td><strong>${item.name}</strong></td>
                    <td><b>${item.no}</b></td>
                    <td style="text-align:center;">${item.date}</td>
                    <td style="text-align:center; color:#0284c7; font-weight:700;">${item.time}</td>
                    <td style="text-align:center;">${item.state}</td>
                    <td style="text-align:center;">${item.new_state}</td>
                    <td style="text-align:center; color:#64748b;">${item.exception}</td>
                    <td style="text-align:center; color:#64748b;">${item.operation}</td>
                `;
                tbody.appendChild(tr);
            });

            openModal('modal-preview-excel');
            input.value = '';
        } catch (err) {
            alert('Lỗi khi đọc file Excel: ' + err.message);
            input.value = '';
        }
    };
    reader.readAsArrayBuffer(file);
}

// Tick cắt cơm qua AJAX
function toggleCancelMeal(id, checkbox) {
    const isChecked = checkbox.checked ? 1 : 0;
    let reason = '';
    
    if (isChecked) {
        reason = prompt('Nhập lý do cắt cơm (nếu có, có thể để trống):', '') || '';
    }

    const formData = new FormData();
    formData.append('ajax_action', 'toggle_cancel');
    formData.append('id', id);
    formData.append('status', isChecked);
    formData.append('reason', reason);

    fetch('suat-an-ngoai.php', {
        method: 'POST',
        body: formData
    })
    .then(res => res.json())
    .then(data => {
        if (data.success) {
            const row = document.getElementById('row_' + id);
            const reasonText = document.getElementById('reason_text_' + id);

            if (isChecked) {
                row.classList.add('row-canceled');
                reasonText.innerText = reason;
            } else {
                row.classList.remove('row-canceled');
                reasonText.innerText = '';
            }
            setTimeout(() => { location.reload(); }, 350);
        } else {
            alert('Có lỗi xảy ra: ' + data.message);
            checkbox.checked = !isChecked;
        }
    })
    .catch(err => {
        alert('Lỗi kết nối máy chủ!');
        checkbox.checked = !isChecked;
    });
}
