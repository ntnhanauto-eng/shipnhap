function parseExcelDate($dateValue, $defaultDate = null) {
    if (empty($dateValue)) return $defaultDate ?: date('Y-m-d');
    $val = trim((string)$dateValue);

    if (is_numeric($val) && $val > 1000) {
        $unixTimestamp = ($val - 25569) * 86400;
        return gmdate('Y-m-d', $unixTimestamp);
    }

    $val = str_replace('/', '-', $val);
    $ts = strtotime($val);
    if ($ts !== false && $ts > 0) {
        return date('Y-m-d', $ts);
    }
    return $defaultDate ?: date('Y-m-d');
}

function parseXlsxFastStream($filePath) {
    $rows = [];
    $zip = new ZipArchive();
    if ($zip->open($filePath) !== TRUE) {
        return $rows;
    }

    $sharedStrings = [];
    $sharedStringsXml = $zip->getFromName('xl/sharedStrings.xml');
    if ($sharedStringsXml !== false) {
        $xml = simplexml_load_string($sharedStringsXml);
        if ($xml) {
            foreach ($xml->si as $si) {
                if (isset($si->t)) {
                    $sharedStrings[] = (string)$si->t;
                } elseif (isset($si->r)) {
                    $t = '';
                    foreach ($si->r as $r) {
                        $t .= (string)$r->t;
                    }
                    $sharedStrings[] = $t;
                } else {
                    $sharedStrings[] = '';
                }
            }
        }
    }

    $sheetXml = $zip->getFromName('xl/worksheets/sheet1.xml');
    if ($sheetXml !== false) {
        $xml = simplexml_load_string($sheetXml);
        if ($xml && isset($xml->sheetData->row)) {
            foreach ($xml->sheetData->row as $row) {
                $currentRow = [];
                $lastColIndex = 0;
                foreach ($row->c as $cell) {
                    $attr = $cell->attributes();
                    $ref = (string)$attr['r'];
                    preg_match('/^([A-Z]+)(\d+)$/', $ref, $matches);
                    $colLetters = $matches[1] ?? 'A';
                    
                    $colIndex = 0;
                    $len = strlen($colLetters);
                    for ($i = 0; $i < $len; $i++) {
                        $colIndex = $colIndex * 26 + (ord($colLetters[$i]) - 64);
                    }
                    $colIndex -= 1;

                    while ($lastColIndex < $colIndex) {
                        $currentRow[] = '';
                        $lastColIndex++;
                    }

                    $type = (string)$attr['t'];
                    $value = (string)$cell->v;

                    if ($type === 's') {
                        $idx = intval($value);
                        $cellVal = $sharedStrings[$idx] ?? '';
                    } elseif ($type === 'inlineStr' && isset($cell->is->t)) {
                        $cellVal = (string)$cell->is->t;
                    } else {
                        $cellVal = $value;
                    }

                    $currentRow[] = trim($cellVal);
                    $lastColIndex = $colIndex + 1;
                }
                if (!empty(array_filter($currentRow, fn($v) => $v !== ''))) {
                    $rows[] = $currentRow;
                }
            }
        }
    }
    $zip->close();
    return $rows;
}

function renderPaginationBar($currentPage, $totalPages, $paramName) {
    if ($totalPages <= 1) return;
    $queryParams = $_GET;

    echo '<div class="pagination-wrapper">';
    if ($currentPage > 1) {
        $queryParams[$paramName] = $currentPage - 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&laquo;</a>';
    } else {
        echo '<span class="page-link disabled">&laquo;</span>';
    }

    $start = max(1, $currentPage - 1);
    $end = min($totalPages, $currentPage + 1);
    if ($currentPage == 1) $end = min($totalPages, $start + 2);
    elseif ($currentPage == $totalPages) $start = max(1, $end - 2);

    for ($i = $start; $i <= $end; $i++) {
        $queryParams[$paramName] = $i;
        $activeClass = ($i == $currentPage) ? 'active' : '';
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link ' . $activeClass . '">' . $i . '</a>';
    }

    if ($currentPage < $totalPages) {
        $queryParams[$paramName] = $currentPage + 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&raquo;</a>';
    } else {
        echo '<span class="page-link disabled">&raquo;</span>';
    }
    echo '</div>';
}

function normalizeFastText($str) {
    $str = str_replace("\xC2\xA0", ' ', (string)$str);
    $str = mb_strtolower(trim($str), 'UTF-8');
    $str = preg_replace('/[áàảãạăắằẳẵặâấầẩẫậ]/u', 'a', $str);
    $str = preg_replace('/[éèẻẽẹêếềểễệ]/u', 'e', $str);
    $str = preg_replace('/[íìỉĩị]/u', 'i', $str);
    $str = preg_replace('/[óòỏõọôốồổỗộơớờởỡợ]/u', 'o', $str);
    $str = preg_replace('/[úùủũụưứừửữự]/u', 'u', $str);
    $str = preg_replace('/[ýỳỷỹỵ]/u', 'y', $str);
    $str = preg_replace('/đ/u', 'd', $str);
    return preg_replace('/[^a-z0-9]/', '', $str);
}

            WHERE p.request_code = :code AND p.status IN ('Đã duyệt', 'Chờ cấp phát')
            FOR UPDATE
        ");
        $stmtProps->execute([':code' => $req_code]);
        $rows = $stmtProps->fetchAll();

        if (empty($rows)) {
            throw new Exception("Yêu cầu đề xuất chưa được duyệt hoặc đã được cấp phát từ trước!");
        }

        $stmtSub = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty - :qty WHERE id = :id");
        $stmtReq = $pdo->prepare("
            INSERT INTO vpp_requests (request_code, request_date, employee_id, department, receiver_name, item_id, quantity, delivery_location, notes)
            VALUES (:code, :r_date, :empid, :dept, :rec, :item_id, :qty, :loc, :notes)

        $stmtUpProp = $pdo->prepare("UPDATE vpp_proposals SET status = 'Đã cấp phát', approved_quantity = :appr_q WHERE id = :id");

        $itemDeliveredCount = 0;
        foreach ($rows as $p) {
            $propId = $p['id'];
            $giveQty = isset($deliver_items[$propId]) ? max(1, intval($deliver_items[$propId])) : max(1, intval($p['approved_quantity'] !== null ? $p['approved_quantity'] : $p['quantity']));

            if ($p['stock_qty'] < $giveQty) {
                throw new Exception("Tồn kho không đủ để cấp mặt hàng [{$p['item_name']}]! Tồn kho hiện có: " . number_format($p['stock_qty']) . " {$p['unit']}, cần cấp: " . number_format($giveQty) . " {$p['unit']}.");

            $stmtSub->execute([':qty' => $giveQty, ':id' => $p['item_id']]);
            $stmtReq->execute([

                ':r_date'  => date('Y-m-d'),

                ':loc'     => $delivery_loc ?: $p['delivery_location'],
                ':notes'   => $custom_notes

            $stmtUpProp->execute([':appr_q' => $giveQty, ':id' => $propId]);
            $itemDeliveredCount++;
        }

        $pdo->commit();
        $message = "Đã xác nhận cấp phát thành công {$itemDeliveredCount} mặt hàng cho yêu cầu [{$req_code}]!";

function openPrintDocModalByCode(code) {
    closeAllPropDropdowns();
    const p = getProposalData(code);
    if (!p) return;

    document.getElementById('prn_receiver_name').innerText = p.receiver_name;
    document.getElementById('prn_sign_receiver').innerText = p.receiver_name;
    
    const empInfo = p.employee_id ? `#${p.employee_id} / ${p.department}` : p.department;
    document.getElementById('prn_dept_info').innerText = empInfo;

    const tbody = document.getElementById('prn_items_tbody');
    tbody.innerHTML = '';
    let totalQty = 0;
    let validRowIndex = 1;

    (p.items || []).forEach(it => {
        const qtyToIssue = (it.pre_approved_quantity !== null && it.pre_approved_quantity !== undefined) ? parseInt(it.pre_approved_quantity) : parseInt(it.quantity);
        if (qtyToIssue <= 0) return;

        totalQty += qtyToIssue;
        const stockLeft = Math.max(0, parseInt(it.stock_qty) - qtyToIssue);

        const tr = document.createElement('tr');
        tr.innerHTML = `
            <td>${validRowIndex++}</td>
            <td>${it.item_code || '—'}</td>
            <td class="text-left"><b>${it.item_name}</b> ${it.specification ? '(' + it.specification + ')' : ''}</td>
            <td>${it.unit}</td>
            <td style="font-weight:bold;">${formatNumber(qtyToIssue)}</td>
            <td>${formatNumber(stockLeft)}</td>
            <td></td>
        `;
        tbody.appendChild(tr);
    });

    const currentRowsCount = validRowIndex - 1;
    const emptyRowsCount = Math.max(0, 6 - currentRowsCount);
    for (let i = 0; i < emptyRowsCount; i++) {
        const tr = document.createElement('tr');
        tr.innerHTML = `<td>${currentRowsCount + i + 1}</td><td></td><td></td><td></td><td></td><td></td><td></td>`;
        tbody.appendChild(tr);
    }

    document.getElementById('prn_total_qty').innerText = formatNumber(totalQty);
    openModal('modal-print-view');
}

function openDeliverGroupModalByCode(code) {
    const p = getProposalData(code);
    if (!p) return;

    document.getElementById('dlg_request_code').value = p.request_code;
    document.getElementById('dlg_code_text').innerText = p.request_code;
    document.getElementById('dlg_receiver').value = p.receiver_name;
    document.getElementById('dlg_receiver_text').innerText = p.receiver_name + (p.employee_id ? ' (Mã NV: #' + p.employee_id + ')' : '');
    document.getElementById('dlg_department').value = p.department;
    document.getElementById('dlg_dept_text').innerText = p.department;
    if (document.getElementById('dlg_location')) {
        document.getElementById('dlg_location').value = p.delivery_location || '';
    }

    const tbody = document.getElementById('dlg_items_tbody');
    tbody.innerHTML = '';

    (p.items || []).forEach((it, idx) => {
        const tr = document.createElement('tr');
        const apprQty = (it.approved_quantity !== null && it.approved_quantity !== undefined) ? it.approved_quantity : ((it.pre_approved_quantity !== null && it.pre_approved_quantity !== undefined) ? it.pre_approved_quantity : it.quantity);
        const defaultDeliver = Math.max(1, parseInt(apprQty) || 1);

        tr.innerHTML = `
            <td style="text-align:center; font-weight:bold; color:#64748b;">${idx + 1}</td>
            <td><span style="background:#f1f5f9; padding:2px 5px; border-radius:4px; font-weight:600;">${it.item_code || '—'}</span></td>
            <td>
                <b>${it.item_name}</b>
                ${it.specification ? '<br><span style="font-size:0.75rem; color:#64748b;">(' + it.specification + ')</span>' : ''}
            </td>
            <td style="text-align:center; font-weight:600;">${it.unit}</td>
            <td style="text-align:center; font-weight:bold; ${it.stock_qty >= defaultDeliver ? 'color:#15803d;' : 'color:#dc2626;'}">
                ${formatNumber(it.stock_qty)}
            </td>
            <td style="text-align:center; color:#0284c7; font-weight:bold;">
                ${formatNumber(it.quantity)}
            </td>
            <td style="text-align:center; color:#15803d; font-weight:bold;">
                ${formatNumber(apprQty)}
            </td>
            <td style="text-align:center;">
                <input type="number" name="deliver_quantities[${it.proposal_id}]" min="1" value="${defaultDeliver}" required
                       style="width:80px; padding:4px 6px; border:1.5px solid #10b981; border-radius:4px; text-align:center; font-weight:bold; color:#0f172a;">
            </td>
        `;
        tbody.appendChild(tr);
    });

    openModal('modal-deliver-group');
}
