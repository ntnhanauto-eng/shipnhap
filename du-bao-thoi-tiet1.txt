<?php
// API Proxy backend xử lý kéo dữ liệu và chạy thuật toán phân tích
if (isset($_GET['action']) && $_GET['action'] === 'get_forecast') {
    header('Content-Type: application/json');
    $lat = filter_input(INPUT_GET, 'lat', FILTER_VALIDATE_FLOAT) ?: 12.6075;
    $lon = filter_input(INPUT_GET, 'lon', FILTER_VALIDATE_FLOAT) ?: 109.1235;

    // 1. Kéo dữ liệu ECMWF (9km) và ICON (13km)
    $url_ecmwf = "https://api.open-meteo.com/v1/ecmwf?latitude={$lat}&longitude={$lon}&hourly=temperature_2m,precipitation,surface_pressure,wind_speed_10m,wind_direction_10m&timezone=Asia%2FBangkok";
    $url_icon  = "https://api.open-meteo.com/v1/dwd-icon?latitude={$lat}&longitude={$lon}&hourly=temperature_2m,precipitation,surface_pressure,wind_speed_10m,wind_direction_10m&timezone=Asia%2FBangkok";

    $fetch = function($url) {
        $ch = curl_init();
        curl_setopt($ch, CURLOPT_URL, $url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_TIMEOUT, 8);
        curl_setopt($ch, CURLOPT_USERAGENT, 'WeatherExpertApp/2.0');
        $res = curl_exec($ch);
        curl_close($ch);
        return $res ? json_decode($res, true) : null;
    };

    $ecmwf = $fetch($url_ecmwf);
    $icon  = $fetch($url_icon);

    if (!$ecmwf || !$icon || !isset($ecmwf['hourly']['time'])) {
        echo json_encode(['error' => 'Không thể tải dữ liệu mô hình khí tượng.']);
        exit;
    }

    $times = $ecmwf['hourly']['time'];
    $processed = [];

    // 2. Chạy thuật toán khí tượng chuyên gia cho từng giờ
    for ($i = 0; $i < count($times); $i++) {
        $time_str = $times[$i];
        $hour = (int)substr($time_str, 11, 2);
        
        $rain_e = (float)($ecmwf['hourly']['precipitation'][$i] ?? 0);
        $temp_e = (float)($ecmwf['hourly']['temperature_2m'][$i] ?? 0);
        
        $rain_i = (float)($icon['hourly']['precipitation'][$i] ?? 0);
        $temp_i = (float)($icon['hourly']['temperature_2m'][$i] ?? 0);
        $wind_spd = (float)($icon['hourly']['wind_speed_10m'][$i] ?? 0);
        $wind_dir = (float)($icon['hourly']['wind_direction_10m'][$i] ?? 0);
        $pres = (float)($icon['hourly']['surface_pressure'][$i] ?? 1010);

        // Thuật toán xác suất mưa
        $prob = 10;
        $desc = "Trời quang, tạnh ráo";
        $warning = "";

        if ($rain_e >= 1.5 && $rain_i >= 1.5) {
            $prob = 85;
            $desc = "Mưa to diện rộng";
        } elseif ($rain_e >= 0.5 && $rain_i >= 0.5) {
            $prob = 70;
            $desc = "Mưa rào vừa";
        } elseif ($rain_e >= 1.5 || $rain_i >= 1.5) {
            $prob = 55;
            $desc = ($rain_e >= 1.5) ? "Mưa cục bộ (ECMWF báo mạnh)" : "Mưa dông nhiệt thoáng qua (ICON báo)";
        } elseif ($rain_e > 0 || $rain_i > 0) {
            $prob = 35;
            $desc = "Mưa nhỏ / Mưa phùn nhẹ";
        }

        // Kiểm tra gió biển Đông (góc 60° đến 150° tại Khánh Hòa)
        $is_sea_wind = ($wind_dir >= 60 && $wind_dir <= 150 && $wind_spd >= 8);
        if ($is_sea_wind && $prob >= 35) {
            $prob += 10;
        }

        // Kiểm tra đối lưu nhiệt buổi chiều (13h - 17h)
        if ($hour >= 13 && $hour <= 17 && $temp_i >= 32.5 && $is_sea_wind) {
            $prob = max($prob, 80);
            $warning = "Nguy cơ dông sét nhiệt cực đoan & gió giật";
        }

        $prob = min(98, max(5, $prob));

        $processed[] = [
            'time' => $time_str,
            'hour' => $hour,
            'date' => substr($time_str, 0, 10),
            'temp' => round(($temp_e + $temp_i) / 2, 1),
            'rain_ecmwf' => $rain_e,
            'rain_icon' => $rain_i,
            'wind_speed' => round($wind_spd, 1),
            'wind_dir' => round($wind_dir),
            'pressure' => round($pres, 1),
            'prob' => $prob,
            'desc' => $desc,
            'warning' => $warning
        ];
    }

    echo json_encode(['success' => true, 'data' => $processed]);
    exit;
}

<script>
    // 1. Khởi tạo Bản đồ Leaflet tập trung tỉnh Khánh Hòa
    const map = L.map('map').setView([12.50, 109.15], 10);
    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
        attribution: '&copy; OpenStreetMap contributors'
    }).addTo(map);

    let currentMarker = L.marker([12.6075, 109.1235]).addTo(map)
        .bindPopup("<b>Điểm chọn</b><br>Ninh Hòa, Khánh Hòa").openPopup();

    let weatherData = [];
    let selectedDate = "";
    let selectedHour = 14;

    // 2. Click bản đồ để lấy tọa độ bất kỳ
    map.on('click', function(e) {
        const lat = e.latlng.lat.toFixed(4);
        const lon = e.latlng.lng.toFixed(4);
        currentMarker.setLatLng(e.latlng)
            .setPopupContent(`<b>Tọa độ mới:</b><br>${lat}, ${lon}`).openPopup();
        document.getElementById('coords-text').innerText = `Vị trí: ${lat}, ${lon}`;
        loadWeather(lat, lon);
    });

    // 3. Gọi Backend PHP lấy dữ liệu
    function loadWeather(lat, lon) {
        document.getElementById('weather-title').innerText = "Đang kéo dữ liệu ECMWF & ICON...";
        fetch(`index.php?action=get_forecast&lat=${lat}&lon=${lon}`)
            .then(res => res.json())
            .then(res => {
                if(res.success) {
                    weatherData = res.data;
                    initDayBar();
                    renderView();
                } else {
                    alert(res.error);
                }
            });
    }

    // 4. Tạo thanh chọn 7 ngày
    function initDayBar() {
        const dayBar = document.getElementById('day-bar');
        dayBar.innerHTML = "";
        const dates = [...new Set(weatherData.map(d => d.date))];
        if(!selectedDate || !dates.includes(selectedDate)) {
            selectedDate = dates[0];
        }

        dates.forEach(d => {
            const btn = document.createElement('button');
            btn.className = `day-btn ${d === selectedDate ? 'active' : ''}`;
            const parts = d.split('-');
            btn.innerText = `${parts[2]}/${parts[1]}`;
            btn.onclick = () => {
                document.querySelectorAll('.day-btn').forEach(b => b.classList.remove('active'));
                btn.classList.add('active');
                selectedDate = d;
                renderView();
            };
            dayBar.appendChild(btn);
        });
    }

    // 5. Cập nhật chi tiết theo giờ
    function renderView() {
        const hourData = weatherData.find(d => d.date === selectedDate && d.hour == selectedHour);
        if(!hourData) return;

        document.getElementById('hour-display').innerText = `${String(selectedHour).padStart(2, '0')}:00`;
        document.getElementById('weather-title').innerText = hourData.desc;
        
        const badge = document.getElementById('prob-badge');
        badge.innerText = `${hourData.prob}% Mưa`;
        badge.className = `metric-badge ${hourData.prob >= 70 ? 'badge-red' : (hourData.prob >= 40 ? 'badge-yellow' : 'badge-green')}`;

        document.getElementById('warning-box').innerText = hourData.warning;
        document.getElementById('val-rain-e').innerText = `${hourData.rain_ecmwf} mm`;
        document.getElementById('val-rain-i').innerText = `${hourData.rain_icon} mm`;
        document.getElementById('val-temp').innerText = `${hourData.temp} °C`;
        document.getElementById('val-wind').innerText = `${hourData.wind_speed} km/h (${hourData.wind_dir}°)`;
        document.getElementById('val-pres').innerText = `${hourData.pressure} hPa`;

        // Cơ chế gió biển
        const isSea = (hourData.wind_dir >= 60 && hourData.wind_dir <= 150);
        document.getElementById('val-wind-mech').innerText = isSea ? "Gió Biển (Cấp ẩm)" : "Gió Đất/Núi (Khô)";

        // Đổ bảng timeline theo ngày
        const tBody = document.getElementById('timeline-body');
        tBody.innerHTML = "";
        const daySlots = weatherData.filter(d => d.date === selectedDate && [7, 11, 14, 17, 20, 23].includes(d.hour));
        daySlots.forEach(slot => {
            const tr = document.createElement('tr');
            tr.innerHTML = `
                <td><b>${slot.hour}h</b></td>
                <td>${slot.rain_ecmwf} mm</td>
                <td>${slot.rain_icon} mm</td>
                <td><span style="color:${slot.prob>=70?'#f87171':(slot.prob>=40?'#fbbf24':'#34d399')}; font-weight:bold;">${slot.prob}%</span></td>
                <td>${slot.desc}</td>
            `;
            tBody.appendChild(tr);
        });
    }

    // Sự kiện kéo thanh trượt giờ
    document.getElementById('hour-slider').addEventListener('input', function(e) {
        selectedHour = parseInt(e.target.value);
        renderView();
    });

    // Chạy mặc định khi tải trang
    loadWeather(12.6075, 109.1235);
</script>
