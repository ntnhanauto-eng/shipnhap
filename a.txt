// Hàm kiểm tra sản phẩm có phải loại bình 19L/20L đổi trả vỏ hay không
function is19LBottle($name, $spec = '') {
    $str = mb_strtolower($name . ' ' . $spec, 'UTF-8');
    return (strpos($str, '19l') !== false || strpos($str, '19 l') !== false || strpos($str, '19 lít') !== false || strpos($str, '20l') !== false);
}

// ==================== 1. XỬ LÝ CẤP PHÁT TỪ ĐỀ XUẤT ====================
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'approve_proposal') {
    $prop_id       = intval($_POST['proposal_id']);
    $receiver_name = trim($_POST['receiver_name'] ?? '');
    $department    = trim($_POST['department'] ?? '');
    $item_id       = intval($_POST['item_id'] ?? 0);
    $quantity      = intval($_POST['quantity'] ?? 1);
    $delivery_loc  = trim($_POST['delivery_location'] ?? '');
    $custom_notes  = trim($_POST['notes'] ?? '');

    try {
        $pdo->beginTransaction();
        $stmtProp = $pdo->prepare("SELECT * FROM water_proposals WHERE id = :id AND status = 'Chờ cấp phát' FOR UPDATE");
        $stmtProp->execute([':id' => $prop_id]);
        $prop = $stmtProp->fetch();

        if (!$prop) {
            throw new Exception("Yêu cầu đề xuất không tồn tại hoặc đã được xử lý từ trước!");
        }

        // Kiểm tra tồn kho nước
        $stmtCheck = $pdo->prepare("SELECT item_name, specification, stock_qty, unit FROM water_items WHERE id = :id FOR UPDATE");
        $stmtCheck->execute([':id' => $item_id]);
        $it = $stmtCheck->fetch();

        if (!$it) {
            throw new Exception("Sản phẩm nước không tồn tại trong kho!");
        }
        if ($it['stock_qty'] < $quantity) {
            throw new Exception("Tồn kho hiện tại không đủ để cấp! [{$it['item_name']}] chỉ còn {$it['stock_qty']} {$it['unit']}.");
        }

        // Kiểm tra: Chỉ bình 19L mới tính số vỏ gửi trả
        $is19L = is19LBottle($it['item_name'], $it['specification']);
        $ret_bottles = $is19L ? max(0, intval($_POST['return_bottle_count'] ?? 0)) : 0;

        // 1. Trừ tồn kho
        $stmtSub = $pdo->prepare("UPDATE water_items SET stock_qty = stock_qty - :qty WHERE id = :id");
        $stmtSub->execute([':qty' => $quantity, ':id' => $item_id]);

        // 2. Ghi vào lịch sử cấp phát
        $final_notes = "Cấp theo đề xuất #" . $prop['id'] . ($custom_notes ? " - " . $custom_notes : "");

        $stmtReq = $pdo->prepare("
            INSERT INTO water_requests (request_date, department, receiver_name, item_id, quantity, has_bottle_return, delivery_location, notes)
            VALUES (:r_date, :dept, :rec, :item_id, :qty, :has_ret, :loc, :notes)
        ");
        $stmtReq->execute([
            ':r_date'  => date('Y-m-d'),
            ':dept'    => $department,
            ':rec'     => $receiver_name,
            ':item_id' => $item_id,
            ':qty'     => $quantity,
            ':has_ret' => $ret_bottles,
            ':loc'     => $delivery_loc,
            ':notes'   => $final_notes
        ]);

        // 3. Đánh dấu đề xuất là đã cấp phát
        $stmtUp = $pdo->prepare("UPDATE water_proposals SET status = 'Đã cấp phát' WHERE id = :id");
        $stmtUp->execute([':id' => $prop_id]);

        $pdo->commit();
        $message = "Đã xác nhận cấp phát nước uống thành công cho " . htmlspecialchars($receiver_name) . "!";

// ==================== 2. XỬ LÝ SỬA / XÓA LỊCH SỬ CẤP PHÁT ====================
// Sửa lịch sử cấp phát
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'update_request') {
    $req_id   = intval($_POST['request_id']);
    $new_date = $_POST['request_date'];
    $new_dept = trim($_POST['department']);
    $new_rec  = trim($_POST['receiver_name'] ?? '');
    $new_item = intval($_POST['item_id']);
    $new_qty  = intval($_POST['quantity']);
    $new_loc  = trim($_POST['delivery_location'] ?? '');
    $new_notes= trim($_POST['notes'] ?? '');

    try {
        $pdo->beginTransaction();
        $stmtOld = $pdo->prepare("SELECT * FROM water_requests WHERE id = :id FOR UPDATE");
        $stmtOld->execute([':id' => $req_id]);
        $oldReq = $stmtOld->fetch();

        if (!$oldReq) {
            throw new Exception("Bản ghi cấp phát không tồn tại!");
        }

        $stmtIt = $pdo->prepare("SELECT item_name, specification, stock_qty, unit FROM water_items WHERE id = :id FOR UPDATE");
        $stmtIt->execute([':id' => $new_item]);
        $it = $stmtIt->fetch();

        if (!$it) {
            throw new Exception("Sản phẩm nước không tồn tại!");
        }

        if (intval($oldReq['item_id']) === $new_item) {
            $diff = $new_qty - intval($oldReq['quantity']);
            if ($diff > 0 && $diff > $it['stock_qty']) {
                throw new Exception("Không đủ nước trong kho để tăng số lượng cấp! Còn thiếu {$diff} {$it['unit']}.");
            }
            $stmtUpStock = $pdo->prepare("UPDATE water_items SET stock_qty = GREATEST(0, stock_qty - :diff) WHERE id = :id");
            $stmtUpStock->execute([':diff' => $diff, ':id' => $new_item]);
        } else {
            $stmtBack = $pdo->prepare("UPDATE water_items SET stock_qty = stock_qty + :q WHERE id = :id");
            $stmtBack->execute([':q' => $oldReq['quantity'], ':id' => $oldReq['item_id']]);

            if ($new_qty > $it['stock_qty']) {
                throw new Exception("Không đủ tồn kho mặt hàng mới để cấp! Tồn kho hiện tại: {$it['stock_qty']} {$it['unit']}.");
            }
            $stmtTake = $pdo->prepare("UPDATE water_items SET stock_qty = stock_qty - :q WHERE id = :id");
            $stmtTake->execute([':q' => $new_qty, ':id' => $new_item]);
        }

        $is19L = is19LBottle($it['item_name'], $it['specification']);
        $ret_bottles = $is19L ? max(0, intval($_POST['return_bottle_count'] ?? 0)) : 0;

        $stmtUpReq = $pdo->prepare("
            UPDATE water_requests 
            SET request_date = :r_date, department = :dept, receiver_name = :rec, item_id = :item_id, 
                quantity = :qty, has_bottle_return = :has_ret, delivery_location = :loc, notes = :notes 
            WHERE id = :id
        ");
        $stmtUpReq->execute([
            ':r_date'  => $new_date, 
            ':dept'    => $new_dept, 
            ':rec'     => $new_rec,
            ':item_id' => $new_item,
            ':qty'     => $new_qty, 
            ':has_ret' => $ret_bottles,
            ':loc'     => $new_loc,
            ':notes'   => $new_notes, 
            ':id'      => $req_id
        ]);

        $pdo->commit();
        $message = "Đã cập nhật bản ghi lịch sử cấp phát nước và điều chỉnh kho thành công!";
    } catch (Exception $e) {
        if ($pdo->inTransaction()) $pdo->rollBack();
        $error = "Lỗi cập nhật: " . $e->getMessage();
    }
}

            $stmtDel->execute([':id' => $req_id]);

// ==================== 3. CÁC THAO TÁC CRUD SẢN PHẨM & KHO ====================
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    $act = $_POST['action'];

    // Thêm sản phẩm mới
    if ($act === 'add_item') {
        $code = trim($_POST['item_code'] ?? '');
        $name = trim($_POST['item_name'] ?? '');
        $spec = trim($_POST['specification'] ?? '');
        $unit = trim($_POST['unit'] ?? 'Bình');
        $init = intval($_POST['initial_stock'] ?? 0);
        if ($name) {
            $stmt = $pdo->prepare("INSERT INTO water_items (item_code, item_name, specification, unit, initial_stock, total_imported, stock_qty) VALUES (:c, :n, :s, :u, :i, 0, :st)");
            $stmt->execute([':c'=>$code, ':n'=>$name, ':s'=>$spec, ':u'=>$unit, ':i'=>$init, ':st'=>$init]);
            $message = "Đã thêm loại nước mới vào danh mục!";
        }
    }

    // Sửa thông tin sản phẩm nước
    if ($act === 'update_item') {
        $item_id = intval($_POST['item_id']);
        $code    = trim($_POST['item_code'] ?? '');
        $name    = trim($_POST['item_name'] ?? '');
        $spec    = trim($_POST['specification'] ?? '');
        $unit    = trim($_POST['unit'] ?? 'Bình');
        $init    = intval($_POST['initial_stock'] ?? 0);

        if (!empty($name) && $item_id > 0) {
            try {
                $pdo->beginTransaction();
                $stmtOld = $pdo->prepare("SELECT initial_stock, stock_qty FROM water_items WHERE id = :id FOR UPDATE");
                $stmtOld->execute([':id' => $item_id]);
                $oldItem = $stmtOld->fetch();

                $diffInit = $init - intval($oldItem['initial_stock']);
                $newStock = max(0, intval($oldItem['stock_qty']) + $diffInit);

                $stmt = $pdo->prepare("
                    UPDATE water_items 
                    SET item_code = :code, item_name = :name, specification = :spec, unit = :unit, initial_stock = :init, stock_qty = :stock 
                    WHERE id = :id
                ");
                $stmt->execute([
                    ':code'  => $code,
                    ':name'  => $name,
                    ':spec'  => $spec,
                    ':unit'  => $unit,
                    ':init'  => $init,
                    ':stock' => $newStock,
                    ':id'    => $item_id
                ]);
                $pdo->commit();
                $message = "Đã cập nhật thông tin sản phẩm nước uống thành công!";
            } catch (PDOException $e) {
                if ($pdo->inTransaction()) $pdo->rollBack();
                $error = "Lỗi cập nhật: " . $e->getMessage();
            }
        }
    }

    // Xóa sản phẩm nước
    if ($act === 'delete_item') {
        $item_id = intval($_POST['item_id']);
        try {
            $stmt = $pdo->prepare("DELETE FROM water_items WHERE id = :id");
            $stmt->execute([':id' => $item_id]);
            $message = "Đã xóa loại nước khỏi danh mục kho!";
        } catch (PDOException $e) {
            $error = "Không thể xóa do loại nước này đang có dữ liệu trong lịch sử cấp phát: " . $e->getMessage();
        }
    }

    // Nhập kho
    if ($act === 'import_stock') {
        $item_id = intval($_POST['item_id']);
        $add_qty = intval($_POST['add_qty']);
        if ($item_id > 0 && $add_qty > 0) {
            $stmt = $pdo->prepare("UPDATE water_items SET stock_qty = stock_qty + :q1, total_imported = total_imported + :q2 WHERE id = :id");
            $stmt->execute([':q1'=>$add_qty, ':q2'=>$add_qty, ':id'=>$item_id]);
            $message = "Đã nhập thêm {$add_qty} vào kho thành công!";
        }
    }

    // Cấp phát thủ công
    if ($act === 'add_manual_request') {
        $r_date = $_POST['request_date'];
        $dept   = trim($_POST['department']);
        $rec    = trim($_POST['receiver_name'] ?? '');
        $item_id= intval($_POST['item_id']);
        $qty    = intval($_POST['quantity']);
        $loc    = trim($_POST['delivery_location'] ?? '');
        $notes  = trim($_POST['notes'] ?? '');

        $stmtChk = $pdo->prepare("SELECT item_name, specification FROM water_items WHERE id = :id");
        $stmtChk->execute([':id' => $item_id]);
        $curItem = $stmtChk->fetch();

        $is19L = $curItem ? is19LBottle($curItem['item_name'], $curItem['specification']) : false;
        $ret_bottles = $is19L ? max(0, intval($_POST['return_bottle_count'] ?? 0)) : 0;

        try {
            $pdo->beginTransaction();
            $stmtSub = $pdo->prepare("UPDATE water_items SET stock_qty = GREATEST(0, stock_qty - :q) WHERE id = :id");
            $stmtSub->execute([':q' => $qty, ':id' => $item_id]);

            $stmtReq = $pdo->prepare("
                INSERT INTO water_requests (request_date, department, receiver_name, item_id, quantity, has_bottle_return, delivery_location, notes)
                VALUES (:rd, :dept, :rec, :it, :q, :hr, :loc, :n)
            ");
            $stmtReq->execute([':rd'=>$r_date, ':dept'=>$dept, ':rec'=>$rec, ':it'=>$item_id, ':q'=>$qty, ':hr'=>$ret_bottles, ':loc'=>$loc, ':n'=>$notes]);
            $pdo->commit();
            $message = "Đã cấp phát nước thành công!";
        } catch (Exception $e) { $pdo->rollBack(); $error = "Lỗi: " . $e->getMessage(); }
    }

    // Tùy chỉnh tồn kho thủ công
    if ($act === 'manual_set_stock') {
        $stmt = $pdo->prepare("UPDATE water_items SET stock_qty = :q WHERE id = :id");
        $stmt->execute([':q' => intval($_POST['new_qty']), ':id' => intval($_POST['item_id'])]);
        $message = "Đã cập nhật tồn kho thực tế thành công!";
    }

    // Reset toàn bộ
    if ($act === 'reset_all_data') {
        $pdo->exec("UPDATE water_items SET initial_stock = 0, total_imported = 0, stock_qty = 0");
        $pdo->exec("TRUNCATE TABLE water_requests");
        $pdo->exec("TRUNCATE TABLE water_proposals");
        $message = "Đã reset toàn bộ dữ liệu nước uống về 0!";
    }
}

window.onclick = function(e) {
    if (e.target.classList.contains('modal-overlay')) e.target.style.display = 'none';

// Kiểm tra hiển thị ô trả vỏ trong Modal xác nhận đề xuất
function toggleCfpBottleBox(selectElem) {
    const selectedOpt = selectElem.options[selectElem.selectedIndex];
    const is19L = selectedOpt ? selectedOpt.getAttribute('data-is19l') : '0';
    const wrapper = document.getElementById('cfp_bottle_wrapper');
    if (is19L === '1') {
        wrapper.style.display = 'block';

function toggleMgBottle(selectElem) {
    const selectedOpt = selectElem.options[selectElem.selectedIndex];
    const is19L = selectedOpt ? selectedOpt.getAttribute('data-is19l') : '0';
    const box = document.getElementById('mg_bottle_box');
    if (is19L === '1') {
        box.style.display = 'block';
    } else {

    }
}
