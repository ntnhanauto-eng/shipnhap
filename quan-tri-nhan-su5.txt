// Hàm tính thời gian làm việc an toàn tuyệt đối
function calculateWorkDuration($startDate) {
    if (empty($startDate) || $startDate === '0000-00-00' || strpos($startDate, '0000') !== false) return '—';
    try {
        $time = strtotime($startDate);
        if ($time === false || $time <= 0) return '—';
        $start = new DateTime(date('Y-m-d', $time));
        $now = new DateTime();
        if ($start > $now) return '0 ngày';
        $diff = $start->diff($now);
        $parts = [];
        if ($diff->y > 0) $parts[] = $diff->y . ' năm';
        if ($diff->m > 0) $parts[] = $diff->m . ' tháng';
        if ($diff->d > 0) $parts[] = $diff->d . ' ngày';
        return !empty($parts) ? implode(' ', $parts) : '0 ngày';
    } catch (Exception $e) {
        return '—';
    }
}

        if ($pdo->inTransaction()) $pdo->rollBack();
        $error = "Lỗi tuyển dụng: " . $e->getMessage();

function renderPagination($currentPage, $totalPages, $paramName, $tabName) {
    if ($totalPages <= 1) return;

    $queryParams = $_GET;
    $queryParams['tab'] = $tabName;

    echo '<div class="pagination-wrapper">';
    if ($currentPage > 1) {
        $queryParams[$paramName] = $currentPage - 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&laquo;</a>';
    } else {
        echo '<span class="page-link disabled">&laquo;</span>';
    }

    $start = max(1, $currentPage - 1);
    $end = min($totalPages, $currentPage + 1);

    if ($currentPage == 1) {
        $end = min($totalPages, $start + 2);
    } elseif ($currentPage == $totalPages) {
        $start = max(1, $end - 2);
    }

    for ($i = $start; $i <= $end; $i++) {
        $queryParams[$paramName] = $i;
        $activeClass = ($i == $currentPage) ? 'active' : '';
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link ' . $activeClass . '">' . $i . '</a>';
    }

    if ($currentPage < $totalPages) {
        $queryParams[$paramName] = $currentPage + 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&raquo;</a>';
    } else {
        echo '<span class="page-link disabled">&raquo;</span>';
    }

    echo '</div>';
}

function switchTab(tabId) {
    document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));
    document.querySelectorAll('.tab-btn').forEach(el => el.classList.remove('active'));
    
    let targetTab = document.getElementById(tabId);
    let targetBtn = document.getElementById('btn-' + tabId);
    
    if (targetTab && targetBtn) {
        targetTab.classList.add('active');
        targetBtn.classList.add('active');
        localStorage.setItem('hr_current_tab', tabId);
    }
}

window.addEventListener('DOMContentLoaded', () => {
    let urlParams = new URLSearchParams(window.location.search);
    let urlTab = urlParams.get('tab');
    let serverActiveTab = "<?= htmlspecialchars($active_tab) ?>";
    let savedTab = urlTab || serverActiveTab || localStorage.getItem('hr_current_tab') || 'employees-tab';
    switchTab(savedTab);
});

function uploadAndPreviewCandidateExcel() {
    let fileInput = document.getElementById('cand_excel_file');
    if (!fileInput.files || fileInput.files.length === 0) {
        alert('Vui lòng chọn file Excel!');
        return;
    }

    let formData = new FormData();
    formData.append('action', 'preview_candidate_excel');
    formData.append('excel_file', fileInput.files[0]);

    document.getElementById('cand_upload_loading').style.display = 'block';

    fetch('quan-tri-nhan-su.php', {
        method: 'POST',
        body: formData
    })
    .then(res => res.json())
    .then(data => {
        document.getElementById('cand_upload_loading').style.display = 'none';
        if (data.success) {
            renderCandidatePreviewTable(data.data);
        } else {
            alert(data.message || 'Lỗi khi đọc file Excel!');
        }
    })
    .catch(err => {
        document.getElementById('cand_upload_loading').style.display = 'none';
        alert('Đã xảy ra lỗi khi gửi yêu cầu tới máy chủ!');
    });
}

function renderCandidatePreviewTable(rows) {
    let tbody = document.getElementById('preview_table_body');
    tbody.innerHTML = '';

    rows.forEach((r, idx) => {
        let tr = document.createElement('tr');
        tr.innerHTML = `
            <td style="text-align:center;">${idx + 1}</td>
            <td style="text-align:center;">${r.id ? '<span class="id-tag">#' + escapeHtml(String(r.id)) + '</span>' : '—'}</td>
            <td><strong>${escapeHtml(r.fullname)}</strong></td>
            <td style="text-align:center;">${escapeHtml(r.gender || '')}</td>
            <td style="text-align:center;">${escapeHtml(r.birth_year || '')}</td>
            <td style="text-align:center;">${escapeHtml(r.id_card || '')}</td>
            <td>${escapeHtml(r.address || '')}</td>
            <td style="text-align:center;">${escapeHtml(r.phone || '')}</td>
            <td>${escapeHtml(r.skills_note || '')}</td>
            <td>${escapeHtml(r.applied_position || '')}</td>
            <td style="text-align:center;">${escapeHtml(r.interview_time || '')}</td>
            <td style="text-align:center;">${escapeHtml(r.candidate_status || '')}</td>
            <td style="text-align:center;">${escapeHtml(r.interview_status || '')}</td>
            <td>${escapeHtml(r.docs_needed || '')}</td>
            <td style="text-align:center;">${escapeHtml(r.hire_date || '')}</td>
            <td>${escapeHtml(r.notes || '')}</td>
        `;
        tbody.appendChild(tr);
    });

    document.getElementById('preview_count_info').innerText = `✓ Đã nhận diện được ${rows.length} hồ sơ ứng viên hợp lệ:`;
    document.getElementById('import_data_json_input').value = JSON.stringify(rows);
    document.getElementById('cand-import-step-upload').style.display = 'none';
    document.getElementById('cand-import-step-preview').style.display = 'block';
    document.getElementById('btn_process_cand_excel').style.display = 'none';
    document.getElementById('form_confirm_candidates').style.display = 'inline-block';
}

function escapeHtml(text) {
    if (!text) return '';
    return text.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
}

function updateContractDates() {
    let startDateStr = document.getElementById('hire_start_date').value;
    let months = parseInt(document.getElementById('hire_months').value) || 2;
    if (startDateStr) {
        let startDate = new Date(startDateStr);
        startDate.setMonth(startDate.getMonth() + months);
        let endDateStr = startDate.toISOString().split('T')[0];
        document.getElementById('hire_end_date').value = endDateStr;
    }
}
