function renderPaginationBar($currentPage, $totalPages, $paramName) {
    if ($totalPages <= 1) return;
    $queryParams = $_GET;

    echo '<div class="pagination-wrapper">';
    if ($currentPage > 1) {
        $queryParams[$paramName] = $currentPage - 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&laquo;</a>';
    } else {
        echo '<span class="page-link disabled">&laquo;</span>';
    }

    $start = max(1, $currentPage - 1);
    $end = min($totalPages, $currentPage + 1);
    if ($currentPage == 1) $end = min($totalPages, $start + 2);
    elseif ($currentPage == $totalPages) $start = max(1, $end - 2);

    for ($i = $start; $i <= $end; $i++) {
        $queryParams[$paramName] = $i;
        $activeClass = ($i == $currentPage) ? 'active' : '';
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link ' . $activeClass . '">' . $i . '</a>';
    }

    if ($currentPage < $totalPages) {
        $queryParams[$paramName] = $currentPage + 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&raquo;</a>';
    } else {
        echo '<span class="page-link disabled">&raquo;</span>';
    }
    echo '</div>';
}

function is19LBottle($name, $spec = '') {
    $str = mb_strtolower($name . ' ' . $spec, 'UTF-8');
    return (strpos($str, '19l') !== false || strpos($str, '19 l') !== false || strpos($str, '19 lít') !== false || strpos($str, '19 lit') !== false || strpos($str, '20l') !== false);
}

    } elseif ($f_prop_status === 'Đã duyệt') {
        $whereP[] = "p.status IN ('Đã duyệt', 'Đã cấp phát', 'Chờ cấp phát')";
    } elseif ($f_prop_status === 'Từ chối') {
        $whereP[] = "p.status = 'Từ chối'";
    }
}

if ($f_prop_mode === 'day') {
    $whereP[] = "DATE(p.created_at) = :pd";
    $paramsP[':pd'] = $f_prop_date;
} elseif ($f_prop_mode === 'week') {
    // Chuẩn ISO-8601: Tuần tính chính xác từ Thứ 2 (00:00:00) đến Chủ Nhật (23:59:59)
    $yw = substr($f_prop_week, 0, 4);
    $ww = substr($f_prop_week, 6);
    $dto = new DateTime();
    $dto->setISODate(intval($yw), intval($ww));
    $ws = $dto->format('Y-m-d 00:00:00');
    $dto->modify('+6 days');
    $we = $dto->format('Y-m-d 23:59:59');
    $whereP[] = "p.created_at BETWEEN :pws AND :pwe";
    $paramsP[':pws'] = $ws;
    $paramsP[':pwe'] = $we;
} elseif ($f_prop_mode === 'year') {
    $whereP[] = "DATE_FORMAT(p.created_at, '%Y') = :py";
    $paramsP[':py'] = $f_prop_year;
} else {
    $whereP[] = "DATE_FORMAT(p.created_at, '%Y-%m') = :pm";
    $paramsP[':pm'] = $f_prop_month;
}

    if (proposalFormModal) {

    if (proposalFormModal) {
        proposalFormModal.style.display = 'none';
    }

            div.innerHTML = `<strong>#${emp.id} - ${emp.fullname}</strong> (${emp.department || 'Chưa phân bổ'})`;
            div.onclick = function() {
                inpReceiver.value = emp.fullname;
                hiddenEmpId.value = emp.id;
                if (emp.department) {
                    selDept.value = emp.department;
                }
                suggestBox.style.display = 'none';
            };
            suggestBox.appendChild(div);
        });
        suggestBox.style.display = 'block';
    } else {
        suggestBox.style.display = 'none';
        hiddenEmpId.value = '';
    }
});

document.addEventListener('click', function(e) {
    if (!e.target.closest('#inp_receiver') && !e.target.closest('#suggest_box')) {
        suggestBox.style.display = 'none';
    }
    if (!e.target.closest('.note-interactive-container')) {
        document.querySelectorAll('.note-popover').forEach(el => el.classList.remove('show'));
    }
});

// Tự động phát hiện loại bình 19L để hiển thị ô trả vỏ và cập nhật số lượng
function checkWaterSpec(selectElem) {
    const selectedOpt = selectElem.options[selectElem.selectedIndex];
    const spec = (selectedOpt.getAttribute('data-spec') || '').toLowerCase();
    const name = (selectedOpt.text || '').toLowerCase();
    const boxBottle = document.getElementById('box_bottle_return');

    // Kiểm tra chuẩn loại nước 19L/20L
    const is19L = spec.includes('19l') || spec.includes('19 l') || spec.includes('19 lít') || spec.includes('19 lit') || 
                  name.includes('19l') || name.includes('19 l') || name.includes('19 lít') || name.includes('19 lit') ||
                  spec.includes('20l') || name.includes('20l');

    if (is19L) {
        boxBottle.style.display = 'block';
        syncBottleCount();
    } else {
        boxBottle.style.display = 'none';
        if (inpReturn) inpReturn.value = 0; // Đặt về 0 khi chọn nước chai/ly
    }
}

// Hàm tự động cập nhật số vỏ bình trả bằng với số bình yêu cầu (nếu đang hiển thị ô vỏ bình)
function syncBottleCount() {
    const boxBottle = document.getElementById('box_bottle_return');
    if (boxBottle && boxBottle.style.display === 'block' && inpQty && inpReturn) {
        const val = parseInt(inpQty.value);
        inpReturn.value = (!isNaN(val) && val > 0) ? val : 0;
    } else if (inpReturn) {
        inpReturn.value = 0;
    }
}

});

function togglePropInputs(mode) {
    document.getElementById('prop_input_day').style.display   = (mode === 'day') ? 'inline-block' : 'none';
    document.getElementById('prop_input_week').style.display  = (mode === 'week') ? 'inline-block' : 'none';

    document.getElementById('prop_input_year').style.display  = (mode === 'year') ? 'inline-block' : 'none';
}

function toggleNotePopover(e, btn) {
    if (e) e.stopPropagation();
    const container = btn.closest('.note-interactive-container');
    if (!container) return;
    const pop = container.querySelector('.note-popover');
    if (!pop) return;
    const isShown = pop.classList.contains('show');
    document.querySelectorAll('.note-popover').forEach(el => el.classList.remove('show'));
    if (!isShown) pop.classList.add('show');
}
