// ==================== HÀM NÉN ẢNH DƯỚI 100KB ====================
function compressAndUploadImage($file, $targetDir = 'uploads/vpp/')
{
    if (!isset($file['tmp_name']) || empty($file['tmp_name'])) return '';

    if (!is_dir($targetDir)) {
        mkdir($targetDir, 0777, true);
    }

    $fileInfo = @getimagesize($file['tmp_name']);
    if ($fileInfo === false) return '';

    $mime = $fileInfo['mime'];
    $srcImg = null;

    if ($mime === 'image/jpeg') {
        $srcImg = imagecreatefromjpeg($file['tmp_name']);
    } elseif ($mime === 'image/png') {
        $srcImg = imagecreatefrompng($file['tmp_name']);
    } elseif ($mime === 'image/webp') {
        $srcImg = imagecreatefromwebp($file['tmp_name']);
    }

    if (!$srcImg) return '';

    $ext = pathinfo($file['name'], PATHINFO_EXTENSION);
    $newFileName = 'vpp_mua_' . uniqid() . '.' . ($ext ?: 'jpg');
    $destination = $targetDir . $newFileName;

    $quality = 85;

    if ($mime === 'image/png') {
        imagepng($srcImg, $destination, 8);
    } else {
        imagejpeg($srcImg, $destination, $quality);
    }

    while (filesize($destination) > 100 * 1024 && $quality > 20) {
        $quality -= 15;

        if ($mime === 'image/png') {
            imagepng($srcImg, $destination, 9);
        } else {
            imagejpeg($srcImg, $destination, $quality);
        }
    }

    imagedestroy($srcImg);

    return $destination;
}

// Khởi tạo bảng đề xuất mua VPP và bổ sung cột custom_image nếu chưa có
try {

    $pdo->exec("
        CREATE TABLE IF NOT EXISTS `vpp_purchase_proposals` (
            `id` INT AUTO_INCREMENT PRIMARY KEY,
            `request_code` VARCHAR(60) DEFAULT '',
            `employee_id` INT DEFAULT NULL,
            `receiver_name` VARCHAR(150) NOT NULL,
            `department` VARCHAR(100) NOT NULL,
            `item_id` INT DEFAULT NULL,
            `custom_item_name` VARCHAR(150) DEFAULT '',
            `specification` VARCHAR(100) DEFAULT '',
            `unit` VARCHAR(50) DEFAULT 'Cái',
            `custom_image` VARCHAR(255) DEFAULT '',
            `quantity` INT NOT NULL DEFAULT 1,
            `pre_approved_quantity` INT DEFAULT NULL,
            `approved_quantity` INT DEFAULT NULL,
            `approval_note` VARCHAR(255) DEFAULT '',
            `status` VARCHAR(50) DEFAULT 'Chờ duyệt',
            `reject_reason` VARCHAR(255) DEFAULT '',
            `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            INDEX (`item_id`),
            INDEX (`status`),
            INDEX (`request_code`)
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");

    $colsProp = $pdo
        ->query("SHOW COLUMNS FROM `vpp_purchase_proposals`")
        ->fetchAll(PDO::FETCH_COLUMN);

    if (!in_array('custom_image', $colsProp)) {
        $pdo->exec("
            ALTER TABLE `vpp_purchase_proposals`
            ADD COLUMN `custom_image` VARCHAR(255) DEFAULT ''
            AFTER `unit`
        ");
    }

} catch (PDOException $e) {

if (
    $_SERVER['REQUEST_METHOD'] === 'POST' &&
    isset($_POST['action']) &&
    $_POST['action'] === 'submit_purchase_proposal'
) {

    $employee_id = !empty($_POST['employee_id'])
        ? intval($_POST['employee_id'])
        : null;

    $receiver_name = trim($_POST['receiver_name'] ?? '');
    $department = trim($_POST['department'] ?? '');
    $items_data = $_POST['items'] ?? [];

    try {

        if (empty($receiver_name) || empty($department)) {
            throw new Exception(
                "Vui lòng điền đầy đủ Họ tên/Mã số người đề xuất và Bộ phận / Phòng ban!"
            );
        }

        if (empty($items_data)) {
            throw new Exception(
                "Vui lòng chọn hoặc điền ít nhất một mặt hàng cần đề xuất mua!"
            );
        }

        $baseCode = 'MUA-' . date('dmYHis');

        $pdo->beginTransaction();

        $stmtIns = $pdo->prepare("
            INSERT INTO vpp_purchase_proposals
            (
                request_code,
                employee_id,
                receiver_name,
                department,
                item_id,
                custom_item_name,
                specification,
                unit,
                custom_image,
                quantity,
                status
            )
            VALUES
            (
                :code,
                :empid,
                :rec,
                :dept,
                :it_id,
                :c_name,
                :spec,
                :unit,
                :img,
                :qty,
                'Chờ duyệt'
            )
        ");

        $insertedCount = 0;

        foreach ($items_data as $idx => $row) {

            $itemId = intval($row['item_id'] ?? 0);
            $qty = intval($row['quantity'] ?? 0);

            if ($qty <= 0) continue;

            $customName = trim($row['custom_item_name'] ?? '');
            $spec = trim($row['specification'] ?? '');
            $unit = trim($row['unit'] ?? 'Cái');
            $imgPath = '';

            // Xử lý nén ảnh nếu là mặt hàng mới ngoài kho và có upload file
            if (
                $itemId === 0 &&
                isset($_FILES['items']['name'][$idx]['image']) &&
                $_FILES['items']['error'][$idx]['image'] === UPLOAD_ERR_OK
            ) {

                $fileData = [
                    'name' => $_FILES['items']['name'][$idx]['image'],
                    'tmp_name' => $_FILES['items']['tmp_name'][$idx]['image'],
                    'error' => $_FILES['items']['error'][$idx]['image']
                ];

                $imgPath = compressAndUploadImage($fileData);
            }

            $stmtIns->execute([
                ':code' => $baseCode,
                ':empid' => $employee_id,
                ':rec' => $receiver_name,
                ':dept' => $department,
                ':it_id' => ($itemId > 0 ? $itemId : null),
                ':c_name' => ($itemId > 0 ? '' : $customName),
                ':spec' => $spec,
                ':unit' => $unit,
                ':img' => $imgPath,
                ':qty' => $qty
            ]);

            $insertedCount++;
        }

        if ($insertedCount === 0) {
            throw new Exception(
                "Số lượng đề xuất của các mặt hàng không hợp lệ!"
            );
        }

        $pdo->commit();

        $message =
            "Gửi đề xuất mua văn phòng phẩm thành công! " .
            "Mã phiếu của bạn là: <strong>{$baseCode}</strong>";

    } catch (Exception $e) {

        if ($pdo->inTransaction()) {
            $pdo->rollBack();
        }

        $error = "Lỗi gửi đề xuất: " . $e->getMessage();
    }
}

const inpReceiver = document.getElementById('inp_receiver');
const suggestBox = document.getElementById('suggest_box');
const selDept = document.getElementById('sel_dept');
const hiddenEmpId = document.getElementById('hidden_emp_id');

inpReceiver.addEventListener('input', function() {

    const val = this.value.trim().toLowerCase();

    suggestBox.innerHTML = '';

    if (!val) {

        suggestBox.style.display = 'none';
        hiddenEmpId.value = '';

        return;
    }

    const matched = employees.filter(e =>

        e.fullname.toLowerCase().includes(val) ||

        e.id.toString() === val ||

        ('id:' + e.id).includes(val) ||

        ('#' + e.id).includes(val)

    );

    if (matched.length > 0) {

        matched.slice(0, 8).forEach(emp => {

            const div = document.createElement('div');

            div.className = 'suggest-item';

            div.innerHTML =
                `<strong>#${emp.id} - ${escapeHtml(emp.fullname)}</strong>
                (${escapeHtml(emp.department || 'Chưa phân bổ')})`;

            div.onclick = function() {

                inpReceiver.value = emp.fullname;

                hiddenEmpId.value = emp.id;

                if (emp.department) {
                    selDept.value = emp.department;
                }

                suggestBox.style.display = 'none';
            };

            suggestBox.appendChild(div);

        });

        suggestBox.style.display = 'block';

    } else {

        suggestBox.style.display = 'none';
        hiddenEmpId.value = '';

    }

});


document.addEventListener('click', function(e) {

    if (

    }

});


// =========================================================
// ESCAPE HTML
// =========================================================

function escapeHtml(value) {

    return String(value ?? '')
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');

}

function handleGlobalModeChange(select) {

    const availablePanel =
        document.getElementById('available_items_panel');

    const customPanel =
        document.getElementById('custom_item_panel');

    if (select.value === 'available') {

        availablePanel.style.display = 'block';
        customPanel.style.display = 'none';

        // Khi chuyển sang mặt hàng có sẵn,
        // xóa các mặt hàng mới chưa thêm
        clearCustomPanel();

        updateHiddenItems();

    }

    else if (select.value === 'custom') {

        availablePanel.style.display = 'none';
        customPanel.style.display = 'block';

        // Không xóa dữ liệu backend đã tạo.
        // Chỉ bỏ trạng thái chọn ở giao diện.
        updateHiddenItems();

    }

}

function filterAvailableItems() {

    const keyword =
        document
            .getElementById('available_search')
            .value
            .trim()
            .toLowerCase();

    const rows =
        document.querySelectorAll('.available-product-row');

    rows.forEach(row => {

        const searchText =
            row.getAttribute('data-search') || '';

        row.style.display =
            !keyword || searchText.includes(keyword)
                ? ''
                : 'none';

    });

}

function toggleAvailableItem(checkbox) {

    const row = checkbox.closest('tr');

    const qtyInput =
        row.querySelector('.available-qty-input');

    if (checkbox.checked) {

        qtyInput.disabled = false;

        if (
            !qtyInput.value ||
            parseInt(qtyInput.value, 10) <= 0
        ) {
            qtyInput.value = 1;
        }

        row.classList.add('selected');

    } else {

        qtyInput.disabled = true;
        row.classList.remove('selected');

    }

    updateHiddenItems();

}

function updateHiddenItems() {

    const container =
        document.getElementById('hidden_items_container');

    container.innerHTML = '';

    let index = 0;


    // -----------------------------------------------------
    // MẶT HÀNG CÓ SẴN
    // -----------------------------------------------------

    const checkedItems =
        document.querySelectorAll(
            '.available-item-check:checked'
        );

    checkedItems.forEach(checkbox => {

        const row = checkbox.closest('tr');

        const qtyInput =
            row.querySelector('.available-qty-input');

        const itemId =
            checkbox.getAttribute('data-id');

        const spec =
            checkbox.getAttribute('data-spec') || '';

        const unit =
            checkbox.getAttribute('data-unit') || 'Cái';

        let qty =
            parseInt(qtyInput.value, 10);

        if (!qty || qty <= 0) {
            qty = 1;
        }


        addHiddenInput(
            container,
            `items[${index}][item_id]`,
            itemId
        );

        addHiddenInput(
            container,
            `items[${index}][custom_item_name]`,
            ''
        );

        addHiddenInput(
            container,
            `items[${index}][specification]`,
            spec
        );

        addHiddenInput(
            container,
            `items[${index}][unit]`,
            unit
        );

        addHiddenInput(
            container,
            `items[${index}][quantity]`,
            qty
        );


        index++;

    });


    // -----------------------------------------------------
    // MẶT HÀNG MỚI
    // -----------------------------------------------------

    const customRows =
        document.querySelectorAll('.custom-added-row');

    customRows.forEach(row => {

        const name =
            row.getAttribute('data-name') || '';

        const spec =
            row.getAttribute('data-spec') || '';

        const unit =
            row.getAttribute('data-unit') || 'Cái';

        const qty =
            parseInt(
                row.getAttribute('data-quantity') || '1',
                10
            );

        const fileInput =
            row.querySelector('.custom-row-file');


        addHiddenInput(
            container,
            `items[${index}][item_id]`,
            '0'
        );

        addHiddenInput(
            container,
            `items[${index}][custom_item_name]`,
            name
        );

        addHiddenInput(
            container,
            `items[${index}][specification]`,
            spec
        );

        addHiddenInput(
            container,
            `items[${index}][unit]`,
            unit
        );

        addHiddenInput(
            container,
            `items[${index}][quantity]`,
            qty
        );


        // File không thể chuyển bằng input hidden.
        // Input file thật được đặt trong form với đúng
        // name items[index][image].
        if (fileInput) {
            fileInput.name =
                `items[${index}][image]`;
        }


        index++;

    });


    updateSelectedCount();

}

function addCustomItem() {

    const nameInput =
        document.getElementById('custom_item_name');

    const specInput =
        document.getElementById('custom_specification');

    const unitInput =
        document.getElementById('custom_unit');

    const quantityInput =
        document.getElementById('custom_quantity');

    const fileInput =
        document.getElementById('custom_file');


    const name =
        nameInput.value.trim();

    const spec =
        specInput.value.trim();

    const unit =
        unitInput.value.trim() || 'Cái';

    let quantity =
        parseInt(quantityInput.value, 10);

    if (!name) {

        alert(
            'Vui lòng nhập tên mặt hàng mới!'
        );

        nameInput.focus();

        return;
    }

    if (!quantity || quantity <= 0) {
        quantity = 1;
    }


    const list =
        document.getElementById(
            'custom_items_list'
        );


    const row =
        document.createElement('div');

    row.className =
        'custom-added-row';


    row.setAttribute(
        'data-name',
        name
    );

    row.setAttribute(
        'data-spec',
        spec
    );

    row.setAttribute(
        'data-unit',
        unit
    );

    row.setAttribute(
        'data-quantity',
        quantity
    );


    row.style.cssText = `
        display:grid;
        grid-template-columns:minmax(0,1.5fr)
                           minmax(0,1fr)
                           90px
                           80px
                           170px
                           40px;
        gap:8px;
        align-items:center;
        padding:8px;
        margin-bottom:6px;
        border:1px solid #e2e8f0;
        border-radius:7px;
        background:#f8fafc;
        font-size:0.82rem;
    `;


    let fileHtml = '';


    // -----------------------------------------------------
    // Chuyển file input thật vào dòng sản phẩm
    // -----------------------------------------------------

    if (fileInput.files && fileInput.files[0]) {

        const clonedFile =
            fileInput.cloneNode(true);

        clonedFile.className =
            'custom-row-file';

        clonedFile.style.display =
            'none';

        // Xóa input cũ khỏi form
        // và chuyển bản clone vào row.

        row.appendChild(
            document.createElement('span')
        );

        const fileContainer =
            document.createElement('div');

        fileContainer.style.display =
            'contents';

        fileContainer.appendChild(
            clonedFile
        );

        row.appendChild(fileContainer);

        fileHtml =
            '📷 ' +
            escapeHtml(
                fileInput.files[0].name
            );

    } else {

        fileHtml = 'Không có ảnh';

    }


    // -----------------------------------------------------
    // Tạo giao diện dòng
    // -----------------------------------------------------

    const content =
        document.createElement('div');

    content.style.display =
        'contents';


    const nameDiv =
        document.createElement('div');

    nameDiv.innerHTML =
        '<strong>' +
        escapeHtml(name) +
        '</strong>';


    const specDiv =
        document.createElement('div');

    specDiv.textContent =
        spec || '—';


    const unitDiv =
        document.createElement('div');

    unitDiv.textContent =
        unit;


    const qtyDiv =
        document.createElement('div');

    qtyDiv.textContent =
        quantity;


    const imageDiv =
        document.createElement('div');

    imageDiv.innerHTML =
        fileHtml;


    const deleteDiv =
        document.createElement('div');


    const deleteBtn =
        document.createElement('button');

    deleteBtn.type =
        'button';

    deleteBtn.className =
        'btn-del';

    deleteBtn.textContent =
        '✕';

    deleteBtn.onclick =
        function() {

            row.remove();

            updateHiddenItems();

        };


    deleteDiv.appendChild(
        deleteBtn
    );


    row.appendChild(nameDiv);
    row.appendChild(specDiv);
    row.appendChild(unitDiv);
    row.appendChild(qtyDiv);
    row.appendChild(imageDiv);
    row.appendChild(deleteDiv);


    // Nếu có file thì đưa input file thật
    // vào trong row.
    if (fileInput.files && fileInput.files[0]) {

        const realFileInput =
            fileInput.cloneNode(true);

        realFileInput.className =
            'custom-row-file';

        realFileInput.style.display =
            'none';

        // Clone file input bằng cloneNode
        // không giữ được FileList trong một số trình duyệt.
        // Vì vậy sử dụng DataTransfer để sao chép file.
        try {

            const dt =
                new DataTransfer();

            dt.items.add(
                fileInput.files[0]
            );

            realFileInput.files =
                dt.files;

        } catch (e) {
        }


        realFileInput.name =
            'items[temp][image]';

        row.appendChild(
            realFileInput
        );

    }


    list.appendChild(row);


    // -----------------------------------------------------
    // Reset form nhập
    // -----------------------------------------------------

    nameInput.value = '';
    specInput.value = '';
    unitInput.value = 'Cái';
    quantityInput.value = '1';


    removeCustomImage();


    updateHiddenItems();

// =========================================================
// PREVIEW ẢNH MẶT HÀNG MỚI
// =========================================================

function handleCustomImageChange(input) {

    if (
        input.files &&
        input.files[0]
    ) {

        const reader =
            new FileReader();

        reader.onload =
            function(e) {

                const thumb =
                    document.getElementById(
                        'custom_thumb'
                    );

                const btnWrap =
                    document.getElementById(
                        'custom_upload_btn_wrap'
                    );

                const previewWrap =
                    document.getElementById(
                        'custom_preview_wrap'
                    );


                if (thumb) {
                    thumb.src =
                        e.target.result;
                }

                if (btnWrap) {
                    btnWrap.style.display =
                        'none';
                }

                if (previewWrap) {
                    previewWrap.style.display =
                        'flex';
                }

            };

        reader.readAsDataURL(
            input.files[0]
        );

    }

}
