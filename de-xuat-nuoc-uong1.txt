try {
    $pdo->exec("
        CREATE TABLE IF NOT EXISTS `water_proposals` (
            `id` INT AUTO_INCREMENT PRIMARY KEY,
            `request_code` VARCHAR(50) DEFAULT NULL,
            `employee_id` INT DEFAULT NULL,
            `receiver_name` VARCHAR(150) NOT NULL,
            `department` VARCHAR(100) NOT NULL,
            `item_id` INT NOT NULL,
            `quantity` INT NOT NULL DEFAULT 1,
            `approved_quantity` INT DEFAULT NULL,
            `return_bottle_count` INT NOT NULL DEFAULT 0,
            `delivery_location` TEXT DEFAULT NULL,
            `approval_note` TEXT DEFAULT NULL,
            `status` VARCHAR(50) DEFAULT 'Chờ duyệt',
            `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            INDEX (`item_id`),
            INDEX (`status`),
            INDEX (`created_at`)
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");

    $cols = $pdo->query("SHOW COLUMNS FROM water_proposals")->fetchAll(PDO::FETCH_COLUMN);
    if (!in_array('request_code', $cols)) {
        $pdo->exec("ALTER TABLE water_proposals ADD COLUMN `request_code` VARCHAR(50) DEFAULT NULL AFTER `id`");
    }
    if (!in_array('approved_quantity', $cols) && !in_array('approved_qty', $cols)) {
        $pdo->exec("ALTER TABLE water_proposals ADD COLUMN `approved_quantity` INT DEFAULT NULL AFTER `quantity`");
    }
    if (!in_array('approval_note', $cols) && !in_array('note', $cols)) {
        $pdo->exec("ALTER TABLE water_proposals ADD COLUMN `approval_note` TEXT DEFAULT NULL AFTER `delivery_location`");
    }
} catch (PDOException $e) {}

// 2. Xử lý gửi đề xuất
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'submit_proposal') {
    $receiver_name  = trim($_POST['receiver_name'] ?? '');
    $department     = trim($_POST['department'] ?? '');

    $return_bottles = intval($_POST['return_bottle_count'] ?? 0);
    $delivery_loc   = trim($_POST['delivery_location'] ?? '');
    $employee_id    = !empty($_POST['employee_id']) ? intval($_POST['employee_id']) : null;

    if (empty($receiver_name) || empty($department) || $item_id <= 0 || $quantity <= 0) {
        $error = "Vui lòng điền đầy đủ họ tên, phòng ban và chọn loại nước cần đề xuất!";

            // Mã yêu cầu ghép trực tiếp từ Ngày Tháng Năm Giờ Phút Giây chuẩn theo cap-phat-nuoc-uong.php
            $requestCode = date('YmdHis');

            // Kiểm tra cột lưu mã trong DB để lưu đúng
            $cols = $pdo->query("SHOW COLUMNS FROM water_proposals")->fetchAll(PDO::FETCH_COLUMN);
            $reqCol = in_array('request_code', $cols) ? 'request_code' : (in_array('code', $cols) ? 'code' : null);

            if ($reqCol) {
                $stmt = $pdo->prepare("
                    INSERT INTO water_proposals ({$reqCol}, employee_id, receiver_name, department, item_id, quantity, return_bottle_count, delivery_location, status)
                    VALUES (:req_code, :emp_id, :rec_name, :dept, :item_id, :qty, :ret_b, :loc, 'Chờ duyệt')
                ");
                $stmt->execute([
                    ':req_code' => $requestCode,
                    ':emp_id'   => $employee_id,
                    ':rec_name' => $receiver_name,
                    ':dept'     => $department,
                    ':item_id'  => $item_id,
                    ':qty'      => $quantity,
                    ':ret_b'    => max(0, $return_bottles),
                    ':loc'      => $delivery_loc
                ]);
            } else {
                $stmt = $pdo->prepare("
                    INSERT INTO water_proposals (employee_id, receiver_name, department, item_id, quantity, return_bottle_count, delivery_location, status)
                    VALUES (:emp_id, :rec_name, :dept, :item_id, :qty, :ret_b, :loc, 'Chờ duyệt')
                ");
                $stmt->execute([
                    ':emp_id'   => $employee_id,
                    ':rec_name' => $receiver_name,
                    ':dept'     => $department,
                    ':item_id'  => $item_id,
                    ':qty'      => $quantity,
                    ':ret_b'    => max(0, $return_bottles),
                    ':loc'      => $delivery_loc
                ]);

            $error = "Lỗi gửi yêu cầu: " . $e->getMessage();

// 1. Quản lý Modal Phiếu Đề Xuất
const btnOpenModal   = document.getElementById('btnOpenModal');
const btnCloseModal  = document.getElementById('btnCloseModal');
const proposalModal  = document.getElementById('proposalModal');

btnOpenModal.addEventListener('click', () => {
    proposalModal.style.display = 'flex';
});

btnCloseModal.addEventListener('click', () => {
    proposalModal.style.display = 'none';
});

window.addEventListener('click', (e) => {
    if (e.target === proposalModal) {
        proposalModal.style.display = 'none';
    }
});

// 2. Gợi ý nhân viên
const employees   = <?= json_encode($employees) ?>;
const inpReceiver = document.getElementById('inp_receiver');
const suggestBox  = document.getElementById('suggest_box');
const selDept     = document.getElementById('sel_dept');
const hiddenEmpId = document.getElementById('hidden_emp_id');

inpReceiver.addEventListener('input', function() {
    const val = this.value.trim().toLowerCase();
    suggestBox.innerHTML = '';
    if (!val) {
        suggestBox.style.display = 'none';
        return;
    }

    const matched = employees.filter(e => 
        (e.fullname && e.fullname.toLowerCase().includes(val)) || 
        (e.id && e.id.toString() === val) || 
        ('id:' + e.id).includes(val)
    );

    if (matched.length > 0) {
        matched.slice(0, 8).forEach(emp => {
            const div = document.createElement('div');
            div.className = 'suggest-item';
            div.innerHTML = `<strong>#${emp.id} - ${emp.fullname}</strong> (${emp.department || 'Chưa phân bổ'})`;
            div.onclick = function() {
                inpReceiver.value = emp.fullname;
                hiddenEmpId.value = emp.id;
                if (emp.department) {
                    selDept.value = emp.department;
                }
                suggestBox.style.display = 'none';
            };
            suggestBox.appendChild(div);
        });
        suggestBox.style.display = 'block';
    } else {
        suggestBox.style.display = 'none';
        hiddenEmpId.value = '';
    }
});

document.addEventListener('click', function(e) {
    if (!e.target.closest('#inp_receiver') && !e.target.closest('#suggest_box')) {
        suggestBox.style.display = 'none';
    }
});

// 3. Kiểm tra loại bình có cần trả vỏ không & Tự động đồng bộ số vỏ theo số lượng yêu cầu
const selWaterItem     = document.getElementById('sel_water_item');
const boxBottle        = document.getElementById('box_bottle_return');
const inpQuantity      = document.getElementById('inp_quantity');
const inpReturnBottles = document.getElementById('inp_return_bottles');

function isReturnBottleRequired() {
    if (!selWaterItem.value) return false;
    const selectedOpt = selWaterItem.options[selWaterItem.selectedIndex];
    const spec = (selectedOpt.getAttribute('data-spec') || '').toLowerCase();
    const name = (selectedOpt.text || '').toLowerCase();
    return (spec.includes('19') || spec.includes('20') || name.includes('19l') || name.includes('19 lít') || name.includes('bình'));
}

function checkWaterSpec(selectElem) {
    if (isReturnBottleRequired()) {
        boxBottle.style.display = 'block';
        syncReturnBottles();
    } else {
        boxBottle.style.display = 'none';
        inpReturnBottles.value = 0;
    }
}

function syncReturnBottles() {
    if (isReturnBottleRequired()) {
        const qty = parseInt(inpQuantity.value) || 0;
        inpReturnBottles.value = qty;
    }
}

// 4. Bấm nút mũi tên xem / ẩn ghi chú duyệt
function toggleNote(btn, noteBoxId) {
    const box = document.getElementById(noteBoxId);
    if (!box) return;
    if (box.style.display === 'block') {
        box.style.display = 'none';
        btn.classList.remove('active');
    } else {
        box.style.display = 'block';
        btn.classList.add('active');
    }
}

const confirmModal     = document.getElementById('confirmModal');

const btnConfirmSubmit = document.getElementById('btnConfirmSubmit');

formProposal.addEventListener('submit', function(e) {
    if (!isConfirmed) {
        e.preventDefault();
        if (formProposal.checkValidity()) {
            confirmModal.style.display = 'flex';
        } else {
            formProposal.reportValidity();
        }
    }
});

    confirmModal.style.display = 'none';
});

btnConfirmSubmit.addEventListener('click', function() {
    isConfirmed = true;
    confirmModal.style.display = 'none';
    formProposal.submit();
});
