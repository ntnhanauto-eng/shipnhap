function parseExcelDate($dateValue, $defaultDate = null) {
    if (empty($dateValue)) return $defaultDate ?: date('Y-m-d');
    $val = trim((string)$dateValue);

    if (is_numeric($val) && $val > 1000) {
        $unixTimestamp = ($val - 25569) * 86400;
        return gmdate('Y-m-d', $unixTimestamp);
    }

    $val = str_replace('/', '-', $val);
    $ts = strtotime($val);
    if ($ts !== false && $ts > 0) {
        return date('Y-m-d', $ts);
    }
    return $defaultDate ?: date('Y-m-d');
}

function parseXlsxFastStream($filePath) {
    $rows = [];
    $zip = new ZipArchive();
    if ($zip->open($filePath) !== TRUE) {
        return $rows;
    }

    $sharedStrings = [];
    $sharedStringsXml = $zip->getFromName('xl/sharedStrings.xml');
    if ($sharedStringsXml !== false) {
        $xml = simplexml_load_string($sharedStringsXml);
        if ($xml) {
            foreach ($xml->si as $si) {
                if (isset($si->t)) {
                    $sharedStrings[] = (string)$si->t;
                } elseif (isset($si->r)) {
                    $t = '';
                    foreach ($si->r as $r) {
                        $t .= (string)$r->t;
                    }
                    $sharedStrings[] = $t;
                } else {
                    $sharedStrings[] = '';
                }
            }
        }
    }

    $sheetXml = $zip->getFromName('xl/worksheets/sheet1.xml');
    if ($sheetXml !== false) {
        $xml = simplexml_load_string($sheetXml);
        if ($xml && isset($xml->sheetData->row)) {
            foreach ($xml->sheetData->row as $row) {
                $currentRow = [];
                $lastColIndex = 0;
                foreach ($row->c as $cell) {
                    $attr = $cell->attributes();
                    $ref = (string)$attr['r'];
                    preg_match('/^([A-Z]+)(\d+)$/', $ref, $matches);
                    $colLetters = $matches[1] ?? 'A';
                    
                    $colIndex = 0;
                    $len = strlen($colLetters);
                    for ($i = 0; $i < $len; $i++) {
                        $colIndex = $colIndex * 26 + (ord($colLetters[$i]) - 64);
                    }
                    $colIndex -= 1;

                    while ($lastColIndex < $colIndex) {
                        $currentRow[] = '';
                        $lastColIndex++;
                    }

                    $type = (string)$attr['t'];
                    $value = (string)$cell->v;

                    if ($type === 's') {
                        $idx = intval($value);
                        $cellVal = $sharedStrings[$idx] ?? '';
                    } elseif ($type === 'inlineStr' && isset($cell->is->t)) {
                        $cellVal = (string)$cell->is->t;
                    } else {
                        $cellVal = $value;
                    }

                    $currentRow[] = trim($cellVal);
                    $lastColIndex = $colIndex + 1;
                }
                if (!empty(array_filter($currentRow, fn($v) => $v !== ''))) {
                    $rows[] = $currentRow;
                }
            }
        }
    }
    $zip->close();
    return $rows;
}

function renderPaginationBar($currentPage, $totalPages, $paramName) {
    if ($totalPages <= 1) return;
    $queryParams = $_GET;

    echo '<div class="pagination-wrapper">';
    if ($currentPage > 1) {
        $queryParams[$paramName] = $currentPage - 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&laquo;</a>';
    } else {
        echo '<span class="page-link disabled">&laquo;</span>';
    }

    $start = max(1, $currentPage - 1);
    $end = min($totalPages, $currentPage + 1);
    if ($currentPage == 1) $end = min($totalPages, $start + 2);
    elseif ($currentPage == $totalPages) $start = max(1, $end - 2);

    for ($i = $start; $i <= $end; $i++) {
        $queryParams[$paramName] = $i;
        $activeClass = ($i == $currentPage) ? 'active' : '';
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link ' . $activeClass . '">' . $i . '</a>';
    }

    if ($currentPage < $totalPages) {
        $queryParams[$paramName] = $currentPage + 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&raquo;</a>';
    } else {
        echo '<span class="page-link disabled">&raquo;</span>';
    }
    echo '</div>';
}

function generateUniqueVppCode($pdo, $table, $timeStr, $excludeId = 0) {
    $ts = strtotime($timeStr) ?: time();
    $baseCode = date('dmYHis', $ts);

    $sql = "SELECT request_code FROM `{$table}` WHERE request_code LIKE :b AND id != :exId";
    $stmt = $pdo->prepare($sql);
    $stmt->execute([':b' => $baseCode . '%', ':exId' => $excludeId]);
    $existing = $stmt->fetchAll(PDO::FETCH_COLUMN);

    if (empty($existing) || !in_array($baseCode, $existing)) {
        return $baseCode;
    }

    $suffixes = range('A', 'Z');
    foreach ($suffixes as $char) {
        $candidate = $baseCode . $char;
        if (!in_array($candidate, $existing)) {
            return $candidate;
        }
    }
    return $baseCode . '_' . rand(10, 99);
}

function getDisplayCode($row) {
    if (!empty($row['request_code'])) {
        return $row['request_code'];
    }
    return date('dmYHis', strtotime($row['created_at']));
}

function normalizeFastText($str) {
    $str = str_replace("\xC2\xA0", ' ', (string)$str);
    $str = mb_strtolower(trim($str), 'UTF-8');
    $str = preg_replace('/[áàảãạăắằẳẵặâấầẩẫậ]/u', 'a', $str);
    $str = preg_replace('/[éèẻẽẹêếềểễệ]/u', 'e', $str);
    $str = preg_replace('/[íìỉĩị]/u', 'i', $str);
    $str = preg_replace('/[óòỏõọôốồổỗộơớờởỡợ]/u', 'o', $str);
    $str = preg_replace('/[úùủũụưứừửữự]/u', 'u', $str);
    $str = preg_replace('/[ýỳỷỹỵ]/u', 'y', $str);
    $str = preg_replace('/đ/u', 'd', $str);
    return preg_replace('/[^a-z0-9]/', '', $str);
}

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'pre_approve_group_proposals') {
    $req_code       = trim($_POST['request_code'] ?? '');
    $pre_items_qty  = $_POST['pre_approved_quantities'] ?? [];
    $pre_item_notes = $_POST['pre_item_notes'] ?? [];

    try {
        if (empty($req_code)) throw new Exception("Không xác định được mã yêu cầu đề xuất!");
        if (empty($pre_items_qty)) throw new Exception("Không có danh sách mặt hàng để sơ duyệt!");

        $pdo->beginTransaction();
        $upCount = 0;
        foreach ($pre_items_qty as $propId => $qty) {
            $q = max(0, intval($qty));
            $note = trim($pre_item_notes[$propId] ?? '');
            $stmtUp = $pdo->prepare("
                UPDATE vpp_proposals 
                SET status = 'Đã sơ duyệt', pre_approved_quantity = :pre_qty, approval_note = :note 
                WHERE id = :id AND request_code = :code
            ");
            $stmtUp->execute([
                ':pre_qty' => $q,
                ':note'    => $note,
                ':id'      => intval($propId),
                ':code'    => $req_code
            ]);
            $upCount++;
        }
        $pdo->commit();
        $message = "Đã hoàn tất Sơ duyệt cho yêu cầu [{$req_code}]. Bạn có thể in phiếu Trình ký hoặc bấm Phê duyệt!";
    } catch (Exception $e) {
        if ($pdo->inTransaction()) $pdo->rollBack();
        $error = "Lỗi sơ duyệt: " . $e->getMessage();
    }

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'deliver_group_proposals') {
    $req_code      = trim($_POST['request_code'] ?? '');
    $receiver_name = trim($_POST['receiver_name'] ?? '');
    $department    = trim($_POST['department'] ?? '');
    $delivery_loc  = trim($_POST['delivery_location'] ?? '');
    $custom_notes  = trim($_POST['notes'] ?? '');
    $deliver_items = $_POST['deliver_quantities'] ?? [];

    try {
        if (empty($req_code)) throw new Exception("Không xác định được mã yêu cầu đề xuất!");
        $pdo->beginTransaction();

        $stmtProps = $pdo->prepare("
            SELECT p.*, i.item_name, i.stock_qty, i.unit 
            FROM vpp_proposals p 
            JOIN vpp_items i ON p.item_id = i.id 
            WHERE p.request_code = :code AND p.status IN ('Đã duyệt', 'Chờ cấp phát')
            FOR UPDATE
        ");
        $stmtProps->execute([':code' => $req_code]);
        $rows = $stmtProps->fetchAll();

        if (empty($rows)) {
            throw new Exception("Yêu cầu đề xuất chưa được duyệt hoặc đã được cấp phát từ trước!");
        }

        $stmtSub = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty - :qty WHERE id = :id");
        $stmtReq = $pdo->prepare("
            INSERT INTO vpp_requests (request_code, request_date, employee_id, department, receiver_name, item_id, quantity, delivery_location, notes)
            VALUES (:code, :r_date, :empid, :dept, :rec, :item_id, :qty, :loc, :notes)
        ");
        $stmtUpProp = $pdo->prepare("UPDATE vpp_proposals SET status = 'Đã cấp phát', approved_quantity = :appr_q WHERE id = :id");

        $itemDeliveredCount = 0;
        foreach ($rows as $p) {
            $propId = $p['id'];
            $giveQty = isset($deliver_items[$propId]) ? max(1, intval($deliver_items[$propId])) : max(1, intval($p['approved_quantity'] !== null ? $p['approved_quantity'] : $p['quantity']));

            if ($p['stock_qty'] < $giveQty) {
                throw new Exception("Tồn kho không đủ để cấp mặt hàng [{$p['item_name']}]! Tồn kho hiện có: {$p['stock_qty']} {$p['unit']}, cần cấp: {$giveQty} {$p['unit']}.");
            }

            $stmtSub->execute([':qty' => $giveQty, ':id' => $p['item_id']]);
            $stmtReq->execute([
                ':code'    => $req_code,
                ':r_date'  => date('Y-m-d'),
                ':empid'   => !empty($p['employee_id']) ? intval($p['employee_id']) : null,
                ':dept'    => $department ?: $p['department'],
                ':rec'     => $receiver_name ?: $p['receiver_name'],
                ':item_id' => $p['item_id'],
                ':qty'     => $giveQty,
                ':loc'     => $delivery_loc ?: $p['delivery_location'],
                ':notes'   => $custom_notes
            ]);
            $stmtUpProp->execute([':appr_q' => $giveQty, ':id' => $propId]);
            $itemDeliveredCount++;
        }

        $pdo->commit();
        $message = "Đã xác nhận cấp phát thành công {$itemDeliveredCount} mặt hàng cho yêu cầu [{$req_code}]!";
    } catch (Exception $e) {
        if ($pdo->inTransaction()) $pdo->rollBack();
        $error = "Lỗi cấp phát: " . $e->getMessage();
    }
}

$previewImportData = null;
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'preview_import_stock_excel') {
    if (isset($_FILES['excel_file']) && $_FILES['excel_file']['error'] === UPLOAD_ERR_OK) {
        $tmpName  = $_FILES['excel_file']['tmp_name'];
        $fileName = $_FILES['excel_file']['name'];
        $ext      = strtolower(pathinfo($fileName, PATHINFO_EXTENSION));

        $rows = [];
        if ($ext === 'csv') {
            if (($handle = fopen($tmpName, 'r')) !== false) {
                while (($data = fgetcsv($handle, 4000, ',')) !== false) $rows[] = array_map('trim', $data);
                fclose($handle);
            }
        } elseif ($ext === 'xlsx') {
            $rows = parseXlsxFastStream($tmpName);
        } elseif ($ext === 'xls') {
            $content = file_get_contents($tmpName);
            if (strpos($content, '<table') !== false) {
                $dom = new DOMDocument();
                @$dom->loadHTML(mb_convert_encoding($content, 'HTML-ENTITIES', 'UTF-8'));
                $trs = $dom->getElementsByTagName('tr');
                foreach ($trs as $tr) {
                    $rowValues = [];
                    $tds = $tr->getElementsByTagName('td');
                    if ($tds->length === 0) $tds = $tr->getElementsByTagName('th');
                    foreach ($tds as $td) $rowValues[] = trim($td->textContent);
                    if (!empty(array_filter($rowValues, fn($v) => $v !== ''))) $rows[] = $rowValues;
                }
            } else {
                $rows = parseXlsxFastStream($tmpName);
            }
        }

        if (!empty($rows)) {
            $header = array_shift($rows);
            $idxDate = 0; $idxSup = 1; $idxCode = 2; $idxName = 3; $idxSpec = 4; $idxUnit = 5; $idxQty = 6; $idxInv = 7; $idxNote = 8;
            foreach ($header as $colIdx => $colName) {
                $cNorm = normalizeFastText($colName);
                if (strpos($cNorm, 'ngay') !== false) $idxDate = $colIdx;
                elseif (strpos($cNorm, 'ncc') !== false || strpos($cNorm, 'nhacungcap') !== false) $idxSup = $colIdx;
                elseif (strpos($cNorm, 'mavpp') !== false || strpos($cNorm, 'code') !== false) $idxCode = $colIdx;
                elseif (strpos($cNorm, 'ten') !== false) $idxName = $colIdx;
                elseif (strpos($cNorm, 'quycach') !== false) $idxSpec = $colIdx;
                elseif (strpos($cNorm, 'dvt') !== false || strpos($cNorm, 'donvi') !== false) $idxUnit = $colIdx;
                elseif (strpos($cNorm, 'soluong') !== false || strpos($cNorm, 'sl') !== false) $idxQty = $colIdx;
                elseif (strpos($cNorm, 'hoadon') !== false || strpos($cNorm, 'chungtu') !== false) $idxInv = $colIdx;
                elseif (strpos($cNorm, 'ghichu') !== false) $idxNote = $colIdx;
            }

            $allItems = $pdo->query("SELECT id, item_code, item_name, specification, unit, stock_qty FROM vpp_items")->fetchAll();
            $itemMapByCode = []; $itemMapByName = [];
            foreach ($allItems as $it) {
                if (!empty($it['item_code'])) $itemMapByCode[normalizeFastText($it['item_code'])] = $it;
                if (!empty($it['item_name'])) $itemMapByName[normalizeFastText($it['item_name'])] = $it;
            }

            $previewRows = [];
            $validCount = 0;
            $errorCount = 0;
            $totalValidQty = 0;

            foreach ($rows as $rowIndex => $row) {
                $lineNum = $rowIndex + 2;
                $code = trim($row[$idxCode] ?? '');
                $name = trim($row[$idxName] ?? '');
                $rawQty = trim($row[$idxQty] ?? '');
                $qty = intval(preg_replace('/[^0-9]/', '', (string)$rawQty));
                $sup = trim($row[$idxSup] ?? 'Nhà cung cấp tổng hợp') ?: 'Nhà cung cấp tổng hợp';
                $inv = trim($row[$idxInv] ?? '');
                $dateStr = parseExcelDate($row[$idxDate] ?? '', $today);
                $unit = trim($row[$idxUnit] ?? 'Cái') ?: 'Cái';
                $spec = trim($row[$idxSpec] ?? '');
                $note = trim($row[$idxNote] ?? '');

                $errors = [];
                if (empty($name) && empty($code)) {
                    $errors[] = "Thiếu cả Tên mặt hàng và Mã VPP";
                }
                if ($qty <= 0) {
                    $errors[] = "Số lượng nhập không hợp lệ (" . htmlspecialchars($rawQty ?: 'trống') . ")";
                }

                $isValid = empty($errors);
                $isNewItem = false;
                $existingItemName = '';

                if ($isValid) {
                    $cKey = normalizeFastText($code);
                    $nKey = normalizeFastText($name);
                    $target = $itemMapByCode[$cKey] ?? ($itemMapByName[$nKey] ?? null);
                    if ($target) {
                        $existingItemName = $target['item_name'];
                    } else {
                        $isNewItem = true;
                    }
                    $validCount++;
                    $totalValidQty += $qty;
                } else {
                    $errorCount++;
                }

                $previewRows[] = [
                    'line'         => $lineNum,
                    'date'         => $dateStr,
                    'supplier'     => $sup,
                    'invoice_no'   => $inv,
                    'item_code'    => $code,
                    'item_name'    => $name,
                    'spec'         => $spec,
                    'unit'         => $unit,
                    'quantity'     => $qty,
                    'notes'        => $note,
                    'is_valid'     => $isValid,
                    'is_new_item'  => $isNewItem,
                    'exist_name'   => $existingItemName,
                    'error_reason' => implode(', ', $errors)
                ];
            }

            $_SESSION['vpp_pending_import'] = array_values(array_filter($previewRows, fn($r) => $r['is_valid']));
            $previewImportData = [
                'total_rows'      => count($previewRows),
                'valid_count'     => $validCount,
                'error_count'     => $errorCount,
                'total_valid_qty' => $totalValidQty,
                'rows'            => $previewRows
            ];
        } else {
            $error = "File Excel rỗng hoặc không có dữ liệu để đọc!";
        }
    } else {
        $error = "Có lỗi xảy ra khi tải file lên máy chủ!";
    }
}

    $pendingRows = $_SESSION['vpp_pending_import'] ?? [];
    if (!empty($pendingRows)) {
        $allItems = $pdo->query("SELECT id, item_code, item_name, specification, unit, stock_qty FROM vpp_items")->fetchAll();
        $itemMapByCode = []; $itemMapByName = [];
        foreach ($allItems as $it) {
            if (!empty($it['item_code'])) $itemMapByCode[normalizeFastText($it['item_code'])] = $it;
            if (!empty($it['item_name'])) $itemMapByName[normalizeFastText($it['item_name'])] = $it;
        }

        $successCount = 0;
        $totalQtyImported = 0;
        $pdo->beginTransaction();
        try {
            $stmtInsertItem = $pdo->prepare("INSERT INTO vpp_items (item_code, item_name, specification, unit, initial_stock, total_imported, stock_qty) VALUES (:code, :name, :spec, :unit, 0, 0, 0)");
            $stmtImport = $pdo->prepare("INSERT INTO vpp_imports (invoice_no, supplier_name, import_date, item_id, quantity, notes) VALUES (:inv, :sup, :idate, :it_id, :qty, :n)");
            $stmtUpStock = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty + :q1, total_imported = total_imported + :q2 WHERE id = :id");

            foreach ($pendingRows as $r) {
                $code = trim($r['item_code'] ?? '');
                $name = trim($r['item_name'] ?? '');
                $qty  = intval($r['quantity'] ?? 0);
                if ($qty <= 0) continue;

                $cKey = normalizeFastText($code);
                $nKey = normalizeFastText($name);
                $target = $itemMapByCode[$cKey] ?? ($itemMapByName[$nKey] ?? null);

                if (!$target) {
                    $targetCode = $code ?: ('VPP_' . strtoupper(substr(md5(uniqid()), 0, 5)));
                    $targetName = $name ?: ('Mặt hàng mã ' . $targetCode);
                    $targetUnit = trim($r['unit'] ?? 'Cái') ?: 'Cái';
                    $targetSpec = trim($r['spec'] ?? '');
                    $stmtInsertItem->execute([':code' => $targetCode, ':name' => $targetName, ':spec' => $targetSpec, ':unit' => $targetUnit]);
                    $targetId = $pdo->lastInsertId();
                    
                    $newObj = ['id' => $targetId, 'item_code' => $targetCode, 'item_name' => $targetName];
                    $itemMapByCode[normalizeFastText($targetCode)] = $newObj;
                    $itemMapByName[normalizeFastText($targetName)] = $newObj;
                } else {
                    $targetId = $target['id'];

                    ':it_id' => $targetId,

        $stmtIt = $pdo->prepare("SELECT item_name, specification, stock_qty, unit FROM vpp_items WHERE id = :id FOR UPDATE");
        $stmtIt->execute([':id' => $new_item]);
        $it = $stmtIt->fetch();
        if (!$it) throw new Exception("Mặt hàng VPP không tồn tại!");

        if (intval($oldReq['item_id']) === $new_item) {
            $diff = $new_qty - intval($oldReq['quantity']);

                throw new Exception("Không đủ tồn kho mặt hàng mới để cấp! Tồn: " . number_format($it['stock_qty']) . " {$it['unit']}.");

            $stmtTake->execute([':q' => $new_qty, ':id' => $new_item]);
        }

        $stmtUpReq = $pdo->prepare("UPDATE vpp_requests SET request_date = :r_date, department = :dept, receiver_name = :rec, item_id = :item_id, quantity = :qty, delivery_location = :loc, notes = :notes WHERE id = :id");

$week_start_date = date('Y-m-d 00:00:00', strtotime('monday this week'));
$week_end_date   = date('Y-m-d 23:59:59', strtotime('sunday this week'));

$weekPropStmt = $pdo->prepare("
    SELECT 
        COUNT(DISTINCT request_code) AS count_req_week,
        COUNT(DISTINCT CASE WHEN status IN ('Đã duyệt', 'Đã cấp phát') THEN request_code ELSE NULL END) AS count_appr_week,
        COUNT(DISTINCT CASE WHEN status = 'Từ chối' THEN request_code ELSE NULL END) AS count_rej_week
    FROM vpp_proposals 
    WHERE created_at BETWEEN :ws AND :we

const formatNumber = (num) => new Intl.NumberFormat('vi-VN').format(num || 0);

function saveScrollPosition() {
    localStorage.setItem('vpp_scroll_pos', window.scrollY);
}
window.addEventListener('DOMContentLoaded', () => {
    const scrollPos = localStorage.getItem('vpp_scroll_pos');
    if (scrollPos) {
        window.scrollTo(0, parseInt(scrollPos));
        localStorage.removeItem('vpp_scroll_pos');
    }
});

function openModal(id) { document.getElementById(id).style.display = 'flex'; }
function closeModal(id) { document.getElementById(id).style.display = 'none'; }

function openBulkStockModal() {
    openModal('modal-set-stock');
    const input = document.getElementById('bulk_search_input');
    if (input) { input.value = ''; filterBulkStockTable(); }
}

function filterBulkStockTable() {
    const keyword = document.getElementById('bulk_search_input').value.toLowerCase().trim();
    const rows = document.querySelectorAll('.bulk-stock-row');
    rows.forEach(row => {
        const text = row.innerText.toLowerCase();
        if (text.includes(keyword)) {
            row.style.display = '';
        } else {
            row.style.display = 'none';
        }
    });
}

function setAllStockZero() {
    if (confirm('Bạn có chắc chắn muốn đặt số lượng tồn thực tế của tất cả mặt hàng hiển thị thành 0 không?')) {
        const inputs = document.querySelectorAll('#bulk_stock_table tbody tr:not([style*="display: none"]) input[type="number"]');
        inputs.forEach(input => {
            input.value = 0;
        });
    }
}

function toggleDropdownMenu(e, id) {
    if (e) e.stopPropagation();
    const menu = document.getElementById(id);
    const isShown = menu && menu.classList.contains('show');
    closeAllDropdownMenus();
    if (menu && !isShown) menu.classList.add('show');
}

function openPrintDocModal(p) {
    document.getElementById('prn_receiver_name').innerText = p.receiver_name;
    document.getElementById('prn_sign_receiver').innerText = p.receiver_name;
    
    const empInfo = p.employee_id ? `#${p.employee_id} / ${p.department}` : p.department;
    document.getElementById('prn_dept_info').innerText = empInfo;

    const tbody = document.getElementById('prn_items_tbody');
    tbody.innerHTML = '';
    const rawItems = JSON.parse(p.items_json || '[]');
    let totalQty = 0;
    let validRowIndex = 1;

    rawItems.forEach(it => {
        const qtyToIssue = (it.pre_approved_quantity !== null) ? parseInt(it.pre_approved_quantity) : parseInt(it.quantity);
        if (qtyToIssue <= 0) return;

        totalQty += qtyToIssue;
        const stockLeft = Math.max(0, parseInt(it.stock_qty) - qtyToIssue);

        const tr = document.createElement('tr');
        tr.innerHTML = `
            <td>${validRowIndex++}</td>
            <td>${it.item_code || '—'}</td>
            <td class="text-left"><b>${it.item_name}</b> ${it.specification ? '(' + it.specification + ')' : ''}</td>
            <td>${it.unit}</td>
            <td style="font-weight:bold;">${formatNumber(qtyToIssue)}</td>
            <td>${formatNumber(stockLeft)}</td>
            <td></td>
        `;
        tbody.appendChild(tr);
    });

    const currentRowsCount = validRowIndex - 1;
    const emptyRowsCount = Math.max(0, 6 - currentRowsCount);
    for (let i = 0; i < emptyRowsCount; i++) {
        const tr = document.createElement('tr');
        tr.innerHTML = `<td>${currentRowsCount + i + 1}</td><td></td><td></td><td></td><td></td><td></td><td></td>`;
        tbody.appendChild(tr);
    }

    document.getElementById('prn_total_qty').innerText = formatNumber(totalQty);
    openModal('modal-print-view');
}

function openDeliverGroupModal(p) {
    document.getElementById('dlg_request_code').value = p.request_code;
    document.getElementById('dlg_code_text').innerText = p.request_code;
    document.getElementById('dlg_receiver').value = p.receiver_name;
    document.getElementById('dlg_receiver_text').innerText = p.receiver_name + (p.employee_id ? ' (Mã NV: #' + p.employee_id + ')' : '');
    document.getElementById('dlg_department').value = p.department;
    document.getElementById('dlg_dept_text').innerText = p.department;
    if (document.getElementById('dlg_location')) {
        document.getElementById('dlg_location').value = p.delivery_location || '';
    }

    const tbody = document.getElementById('dlg_items_tbody');
    tbody.innerHTML = '';
    const items = JSON.parse(p.items_json || '[]');

    items.forEach((it, idx) => {
        const tr = document.createElement('tr');
        const apprQty = (it.approved_quantity !== null) ? it.approved_quantity : ((it.pre_approved_quantity !== null) ? it.pre_approved_quantity : it.quantity);
        const defaultDeliver = Math.max(1, parseInt(apprQty) || 1);

        tr.innerHTML = `
            <td style="text-align:center; font-weight:bold; color:#64748b;">${idx + 1}</td>
            <td><span style="background:#f1f5f9; padding:2px 5px; border-radius:4px; font-weight:600;">${it.item_code || '—'}</span></td>
            <td>
                <b>${it.item_name}</b>
                ${it.specification ? '<br><span style="font-size:0.75rem; color:#64748b;">(' + it.specification + ')</span>' : ''}
            </td>
            <td style="text-align:center; font-weight:600;">${it.unit}</td>
            <td style="text-align:center; font-weight:bold; ${it.stock_qty >= defaultDeliver ? 'color:#15803d;' : 'color:#dc2626;'}">
                ${formatNumber(it.stock_qty)}
            </td>
            <td style="text-align:center; color:#0284c7; font-weight:bold;">
                ${formatNumber(it.quantity)}
            </td>
            <td style="text-align:center; color:#15803d; font-weight:bold;">
                ${formatNumber(apprQty)}
            </td>
            <td style="text-align:center;">
                <input type="number" name="deliver_quantities[${it.proposal_id}]" min="1" value="${defaultDeliver}" required
                       style="width:80px; padding:4px 6px; border:1.5px solid #10b981; border-radius:4px; text-align:center; font-weight:bold; color:#0f172a;">
            </td>
        `;
        tbody.appendChild(tr);
    });

    openModal('modal-deliver-group');
}
