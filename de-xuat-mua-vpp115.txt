// ==================== HÀM NÉN ẢNH DƯỚI 100KB ====================
function compressAndUploadImage($file, $targetDir = 'uploads/vpp/')
{
    if (!isset($file['tmp_name']) || empty($file['tmp_name'])) return '';

    if (!is_dir($targetDir)) {
        mkdir($targetDir, 0777, true);
    }

    $fileInfo = @getimagesize($file['tmp_name']);

    if ($fileInfo === false) return '';

    $mime = $fileInfo['mime'];
    $srcImg = null;

    if ($mime === 'image/jpeg') {
        $srcImg = imagecreatefromjpeg($file['tmp_name']);
    } elseif ($mime === 'image/png') {
        $srcImg = imagecreatefrompng($file['tmp_name']);
    } elseif ($mime === 'image/webp') {
        $srcImg = imagecreatefromwebp($file['tmp_name']);
    }

    if (!$srcImg) return '';

    $ext = pathinfo($file['name'], PATHINFO_EXTENSION);

    $newFileName = 'vpp_mua_' . uniqid() . '.' . ($ext ?: 'jpg');

    $destination = $targetDir . $newFileName;

    $quality = 85;

    if ($mime === 'image/png') {
        imagepng($srcImg, $destination, 8);
    } else {
        imagejpeg($srcImg, $destination, $quality);
    }

    while (filesize($destination) > 100 * 1024 && $quality > 20) {

        $quality -= 15;

        if ($mime === 'image/png') {
            imagepng($srcImg, $destination, 9);
        } else {
            imagejpeg($srcImg, $destination, $quality);
        }
    }

    imagedestroy($srcImg);

    return $destination;
}

            `request_code` VARCHAR(60) DEFAULT '',
            `employee_id` INT DEFAULT NULL,
            `receiver_name` VARCHAR(150) NOT NULL,

            `quantity` INT NOT NULL DEFAULT 1,

if (
    $_SERVER['REQUEST_METHOD'] === 'POST' &&
    isset($_POST['action']) &&
    $_POST['action'] === 'submit_purchase_proposal'
) {

    $employee_id = !empty($_POST['employee_id'])
        ? intval($_POST['employee_id'])
        : null;

    $receiver_name = trim($_POST['receiver_name'] ?? '');

    $department = trim($_POST['department'] ?? '');

    $items_data = $_POST['items'] ?? [];


    try {

        if (empty($receiver_name) || empty($department)) {

            throw new Exception(
                "Vui lòng điền đầy đủ Họ tên/Mã số người đề xuất và Bộ phận / Phòng ban!"
            );
        }

        if (empty($items_data)) {

            throw new Exception(
                "Vui lòng chọn hoặc điền ít nhất một mặt hàng cần đề xuất mua!"
            );
        }


        /*
         * ======================================================
         * XÁC ĐỊNH MÃ PHIẾU
         *
         * Nếu trong danh sách có ít nhất 1 mặt hàng mới
         * (item_id = 0) thì dùng:
         *
         * MUAMOI-ngày tháng năm giờ phút giây
         *
         * Nếu chỉ có hàng có sẵn thì:
         *
         * MUA-ngày tháng năm giờ phút giây
         * ======================================================
         */

        $hasCustomItem = false;

        foreach ($items_data as $row) {

            $itemIdCheck = intval($row['item_id'] ?? 0);

            if ($itemIdCheck === 0) {

                $customNameCheck = trim(
                    $row['custom_item_name'] ?? ''
                );

                /*
                 * Chỉ xem là hàng mới khi thực sự có
                 * tên mặt hàng mới.
                 */
                if ($customNameCheck !== '') {
                    $hasCustomItem = true;
                    break;
                }
            }
        }


        if ($hasCustomItem) {
            $baseCode = 'MUAMOI-' . date('dmYHis');
        } else {
            $baseCode = 'MUA-' . date('dmYHis');
        }


        $pdo->beginTransaction();


        $stmtIns = $pdo->prepare("
            INSERT INTO vpp_purchase_proposals
            (
                request_code,
                employee_id,
                receiver_name,
                department,
                item_id,
                custom_item_name,
                specification,
                unit,
                custom_image,
                quantity,
                status
            )
            VALUES
            (
                :code,
                :empid,
                :rec,
                :dept,
                :it_id,
                :c_name,
                :spec,
                :unit,
                :img,
                :qty,
                'Chờ duyệt'
            )
        ");


        $insertedCount = 0;


        foreach ($items_data as $idx => $row) {

            $itemId = intval($row['item_id'] ?? 0);

            $qty = intval($row['quantity'] ?? 0);

            if ($qty <= 0) continue;


            $customName = trim(
                $row['custom_item_name'] ?? ''
            );

            $spec = trim(
                $row['specification'] ?? ''
            );

            $unit = trim(
                $row['unit'] ?? 'Cái'
            );

            $imgPath = '';


            // Xử lý nén ảnh nếu là mặt hàng mới ngoài kho
            if (
                $itemId === 0 &&
                isset($_FILES['items']['name'][$idx]['image']) &&
                $_FILES['items']['error'][$idx]['image'] === UPLOAD_ERR_OK
            ) {

                $fileData = [
                    'name' => $_FILES['items']['name'][$idx]['image'],
                    'tmp_name' => $_FILES['items']['tmp_name'][$idx]['image'],
                    'error' => $_FILES['items']['error'][$idx]['image']
                ];

                $imgPath = compressAndUploadImage($fileData);
            }


            $stmtIns->execute([
                ':code' => $baseCode,
                ':empid' => $employee_id,
                ':rec' => $receiver_name,
                ':dept' => $department,
                ':it_id' => ($itemId > 0 ? $itemId : null),
                ':c_name' => ($itemId > 0 ? '' : $customName),
                ':spec' => $spec,
                ':unit' => $unit,
                ':img' => $imgPath,
                ':qty' => $qty
            ]);

            $insertedCount++;
        }


        if ($insertedCount === 0) {

            throw new Exception(
                "Số lượng đề xuất của các mặt hàng không hợp lệ!"
            );
        }


        $pdo->commit();


        $message =
            "Gửi đề xuất mua văn phòng phẩm thành công! " .
            "Mã phiếu của bạn là: <strong>{$baseCode}</strong>";

    } catch (Exception $e) {

        if ($pdo->inTransaction()) {
            $pdo->rollBack();
        }

        $error = "Lỗi gửi đề xuất: " . $e->getMessage();
    }
}

function escapeHtml(value) {

    if (value === null || value === undefined) {
        return '';
    }

    return String(value)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
}

function updateAvailableHiddenInputs() {

    const container =
        document.getElementById('available_hidden_inputs');

    container.innerHTML = '';

    availableItemIndex = 0;


    const selected =
        document.querySelectorAll(
            '#available_tbody .product-check:checked'
        );


    selected.forEach(check => {

        const row = check.closest('tr');

        const qtyInput =
            row.querySelector('.product-qty');


        const itemId =
            check.value;

        const itemName =
            check.getAttribute('data-name') || '';

        const spec =
            check.getAttribute('data-spec') || '';

        const unit =
            check.getAttribute('data-unit') || 'Cái';

        let qty =
            parseInt(qtyInput.value);


        if (isNaN(qty) || qty <= 0) {
            qty = 1;
            qtyInput.value = 1;
        }


        createHiddenInput(
            container,
            `items[${availableItemIndex}][item_id]`,
            itemId
        );


        /*
         * Hàng có sẵn không dùng custom_item_name.
         * Để trống đúng logic PHP hiện tại.
         */
        createHiddenInput(
            container,
            `items[${availableItemIndex}][custom_item_name]`,
            ''
        );


        createHiddenInput(
            container,
            `items[${availableItemIndex}][specification]`,
            spec
        );


        createHiddenInput(
            container,
            `items[${availableItemIndex}][unit]`,
            unit
        );


        createHiddenInput(
            container,
            `items[${availableItemIndex}][quantity]`,
            qty
        );


        availableItemIndex++;

    });


    const selectedInfo =
        document.getElementById('selected_info');


    if (selected.length > 0) {

        selectedInfo.style.display = 'block';

        selectedInfo.textContent =
            'Đã chọn ' +
            selected.length +
            ' mặt hàng. Vui lòng kiểm tra số lượng trước khi gửi.';

    } else {

        selectedInfo.style.display = 'none';

    }

}


function createHiddenInput(container, name, value) {

    const input =
        document.createElement('input');

    input.type = 'hidden';

    input.name = name;

    input.value = value;

    container.appendChild(input);

}

function removeCustomRow(btn) {

    const tbody =
        document.getElementById('custom_items_tbody');

    const rows =
        tbody.querySelectorAll('tr');


    if (rows.length > 1) {

        btn.closest('tr').remove();

    } else {

        /*
         * Không cho xóa dòng cuối.
         */
        const row =
            btn.closest('tr');

        row.querySelectorAll('input').forEach(input => {

            if (
                input.type !== 'hidden' &&
                input.type !== 'file'
            ) {
                input.value =
                    input.type === 'number'
                        ? 1
                        : '';
            }

        });

        const fileInput =
            row.querySelector('input[type="file"]');

        if (fileInput) {
            fileInput.value = '';
        }

        const idx =
            row.getAttribute('data-custom-row');

        removeImage(idx);
    }

}

function handleImageChange(input, idx) {

    if (input.files && input.files[0]) {

        const reader =
            new FileReader();


        reader.onload = function(e) {

            const thumb =
                document.getElementById(
                    'thumb_' + idx
                );


            const btnWrap =
                document.getElementById(
                    'upload_btn_wrap_' + idx
                );


            const previewWrap =
                document.getElementById(
                    'preview_wrap_' + idx
                );


            if (thumb) {
                thumb.src = e.target.result;
            }


            if (btnWrap) {
                btnWrap.style.display = 'none';
            }


            if (previewWrap) {
                previewWrap.style.display = 'flex';
            }

        };


        reader.readAsDataURL(input.files[0]);

    }

}

document
    .getElementById('purchaseForm')
    .addEventListener('submit', function(e) {

        const mode =
            globalModeSelect.value;


        // ----------------------------------------------
        // Chọn hàng có sẵn
        // ----------------------------------------------

        if (mode === 'available') {

            updateAvailableHiddenInputs();


            const selected =
                document.querySelectorAll(
                    '#available_tbody .product-check:checked'
                );


            if (selected.length === 0) {

                e.preventDefault();

                alert(
                    'Vui lòng tick chọn ít nhất một mặt hàng có sẵn trong kho!'
                );

                return false;
            }


            let invalidQty = false;


            selected.forEach(check => {

                const row =
                    check.closest('tr');

                const qty =
                    row.querySelector('.product-qty');


                if (
                    !qty.value ||
                    parseInt(qty.value) <= 0
                ) {

                    invalidQty = true;

                }

            });


            if (invalidQty) {

                e.preventDefault();

                alert(
                    'Vui lòng nhập số lượng yêu cầu hợp lệ cho các mặt hàng đã chọn!'
                );

                return false;
            }

        }


        // ----------------------------------------------
        // Hàng mới
        // ----------------------------------------------

        else if (mode === 'custom') {

            const rows =
                document.querySelectorAll(
                    '#custom_items_tbody tr'
                );


            if (rows.length === 0) {

                e.preventDefault();

                alert(
                    'Vui lòng nhập ít nhất một mặt hàng mới!'
                );

                return false;
            }


            let validRows = 0;


            rows.forEach(row => {

                const name =
                    row.querySelector(
                        'input[name*="[custom_item_name]"]'
                    );


                const qty =
                    row.querySelector(
                        'input[name*="[quantity]"]'
                    );


                if (
                    name &&
                    name.value.trim() !== '' &&
                    qty &&
                    parseInt(qty.value) > 0
                ) {

                    validRows++;

                }

            });


            if (validRows === 0) {

                e.preventDefault();

                alert(
                    'Vui lòng nhập tên mặt hàng và số lượng hợp lệ!'
                );

                return false;
            }

        }


        // ----------------------------------------------
        // Chưa chọn loại
        // ----------------------------------------------

        else {

            e.preventDefault();

            alert(
                'Vui lòng chọn "Mặt hàng có sẵn trong kho" hoặc "Nhập mặt hàng mới bên ngoài"!'
            );

            return false;
        }


        /*
         * Nếu chạy tới đây:
         *
         * Form sẽ submit bình thường.
         *
         * Không AJAX.
         * Không thay đổi backend.
         * Không thay đổi SQL INSERT.
         */

        return true;

    });
