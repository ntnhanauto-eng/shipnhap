// ==================== HÀM NÉN ẢNH DƯỚI 100KB ====================

function compressAndUploadImage($file, $targetDir = 'uploads/vpp/') {

    if (!isset($file['tmp_name']) || empty($file['tmp_name'])) return '';



    if (!is_dir($targetDir)) {

        mkdir($targetDir, 0777, true);

    }



    $fileInfo = @getimagesize($file['tmp_name']);

    if ($fileInfo === false) return '';



    $mime = $fileInfo['mime'];

    $srcImg = null;



    if ($mime === 'image/jpeg') {

        $srcImg = imagecreatefromjpeg($file['tmp_name']);

    } elseif ($mime === 'image/png') {

        $srcImg = imagecreatefrompng($file['tmp_name']);

    } elseif ($mime === 'image/webp') {

        $srcImg = imagecreatefromwebp($file['tmp_name']);

    }



    if (!$srcImg) return '';



    $ext = pathinfo($file['name'], PATHINFO_EXTENSION);

    $newFileName = 'vpp_mua_' . uniqid() . '.' . ($ext ?: 'jpg');

    $destination = $targetDir . $newFileName;



    $quality = 85;



    if ($mime === 'image/png') {

        imagepng($srcImg, $destination, 8);

    } else {

        imagejpeg($srcImg, $destination, $quality);

    }



    while (file_exists($destination) && filesize($destination) > 100 * 1024 && $quality > 20) {

        $quality -= 15;



        if ($mime === 'image/png') {

            imagepng($srcImg, $destination, 9);

        } else {

            imagejpeg($srcImg, $destination, $quality);

        }

    }



    imagedestroy($srcImg);

    return $destination;

}

            `specification` VARCHAR(100) DEFAULT '',

            `quantity` INT NOT NULL DEFAULT 1,

            `approved_quantity` INT DEFAULT NULL,

if (

    $_SERVER['REQUEST_METHOD'] === 'POST'

    && isset($_POST['action'])

    && $_POST['action'] === 'submit_purchase_proposal'

) {



    $employee_id = !empty($_POST['employee_id'])

        ? intval($_POST['employee_id'])

        : null;



    $receiver_name = trim($_POST['receiver_name'] ?? '');

    $department = trim($_POST['department'] ?? '');

    $items_data = $_POST['items'] ?? [];



    try {



        if (empty($receiver_name) || empty($department)) {

            throw new Exception(

                "Vui lòng điền đầy đủ Họ tên/Mã số người đề xuất và Bộ phận / Phòng ban!"

            );

        }



        if (empty($items_data)) {

            throw new Exception(

                "Vui lòng chọn hoặc điền ít nhất một mặt hàng cần đề xuất mua!"

            );

        }



        /*

         * QUAN TRỌNG:

         * Một lần submit chỉ tạo DUY NHẤT một mã phiếu.

         *

         * Ví dụ:

         * MUA-22092026153045

         *

         * Nếu có 5 sản phẩm trong lần gửi này,

         * cả 5 dòng đều dùng chung request_code này.

         *

         * Nhờ vậy các sản phẩm được gom chung thành một phiếu/list.

         */

        $baseCode = 'MUA-' . date('dmYHis');



        $pdo->beginTransaction();



        // GIỮ NGUYÊN CÂU SQL INSERT

        $stmtIns = $pdo->prepare("

            INSERT INTO vpp_purchase_proposals

            (request_code, employee_id, receiver_name, department, item_id, custom_item_name, specification, unit, custom_image, quantity, status)

            VALUES (:code, :empid, :rec, :dept, :it_id, :c_name, :spec, :unit, :img, :qty, 'Chờ duyệt')

        ");



        $insertedCount = 0;



        foreach ($items_data as $idx => $row) {



            $itemId = intval($row['item_id'] ?? 0);

            $qty = intval($row['quantity'] ?? 0);



            if ($qty <= 0) {

                continue;

            }



            $customName = trim($row['custom_item_name'] ?? '');

            $spec = trim($row['specification'] ?? '');

            $unit = trim($row['unit'] ?? 'Cái');

            $imgPath = '';



            // ====================================================

            // XỬ LÝ MẶT HÀNG MỚI NGOÀI KHO

            // ====================================================

            if (

                $itemId === 0

                && isset($_FILES['items']['name'][$idx]['image'])

                && $_FILES['items']['error'][$idx]['image'] === UPLOAD_ERR_OK

            ) {



                $fileData = [

                    'name' => $_FILES['items']['name'][$idx]['image'],

                    'tmp_name' => $_FILES['items']['tmp_name'][$idx]['image'],

                    'error' => $_FILES['items']['error'][$idx]['image']

                ];



                $imgPath = compressAndUploadImage($fileData);

            }



            // ====================================================

            // DÙNG CHUNG $baseCode CHO TẤT CẢ CÁC SẢN PHẨM

            // TRONG CÙNG MỘT LẦN GỬI

            // ====================================================

            $stmtIns->execute([

                ':code' => $baseCode,

                ':empid' => $employee_id,

                ':rec' => $receiver_name,

                ':dept' => $department,

                ':it_id' => ($itemId > 0 ? $itemId : null),

                ':c_name' => ($itemId > 0 ? '' : $customName),

                ':spec' => $spec,

                ':unit' => $unit,

                ':img' => $imgPath,

                ':qty' => $qty

            ]);



            $insertedCount++;

        }



        if ($insertedCount === 0) {

            throw new Exception(

                "Số lượng đề xuất của các mặt hàng không hợp lệ!"

            );

        }



        $pdo->commit();



        $message =

            "Gửi đề xuất mua văn phòng phẩm thành công! " .

            "Mã phiếu của bạn là: <strong>" .

            htmlspecialchars($baseCode) .

            "</strong>";



    } catch (Exception $e) {



        if ($pdo->inTransaction()) {

            $pdo->rollBack();

        }



        $error = "Lỗi gửi đề xuất: " . $e->getMessage();

    }

}

    const availablePanel = document.getElementById('available_panel');

function toggleAvailableProduct(check) {



    const id = check.getAttribute('data-id');



    const qtyInput = document.querySelector(

        `.available-qty[data-id="${CSS.escape(id)}"]`

    );



    if (qtyInput) {

        qtyInput.disabled = !check.checked;



        if (check.checked) {

            qtyInput.value =

                parseInt(qtyInput.value || '1', 10) > 0

                    ? qtyInput.value

                    : '1';

        }

    }



    updateAvailableRows();

}

function handleImageChange(input, idx) {



    if (input.files && input.files[0]) {



        const reader = new FileReader();



        reader.onload = function(e) {



            const thumb = document.getElementById(

                'thumb_' + idx

            );



            const btnWrap = document.getElementById(

                'upload_btn_wrap_' + idx

            );



            const previewWrap = document.getElementById(

                'preview_wrap_' + idx

            );



            if (thumb) {

                thumb.src = e.target.result;

            }



            if (btnWrap) {

                btnWrap.style.display = 'none';

            }



            if (previewWrap) {

                previewWrap.style.display = 'flex';

            }

        };



        reader.readAsDataURL(input.files[0]);

    }

}

function removeImage(idx) {



    const fileInput = document.getElementById(

    }



    if (btnWrap) {

        btnWrap.style.display = 'block';

    }



    if (previewWrap) {

        previewWrap.style.display = 'none';

    }

}

// ESCAPE HTML

// ============================================================



function escapeHtml(value) {



    return String(value ?? '')

        .replace(/&/g, '&amp;')

        .replace(/</g, '&lt;')

        .replace(/>/g, '&gt;')

        .replace(/"/g, '&quot;')

        .replace(/'/g, '&#039;');

}



// ============================================================

// ESCAPE JS

// ============================================================



function escapeJs(value) {



    return String(value ?? '')

        .replace(/\\/g, '\\\\')

        .replace(/'/g, "\\'")

        .replace(/"/g, '\\"');

}

document.querySelector('form').addEventListener('submit', function(e) {



    const mode =

        document.getElementById('global_mode_select').value;



    if (!mode) {



        e.preventDefault();



        alert(

            'Vui lòng chọn "Chọn mặt hàng có sẵn trong kho" hoặc "Nhập mặt hàng mới bên ngoài".'

        );



        return;

    }



    if (mode === 'available') {



        const checked = document.querySelectorAll(

            '.available-product-check:checked'

        );



        if (checked.length === 0) {



            e.preventDefault();



            alert(

                'Vui lòng tick chọn ít nhất một mặt hàng có sẵn trong kho.'

            );



            return;

        }



        // Đảm bảo dữ liệu được dựng lại trước submit.

        updateAvailableRows();



    }



    if (mode === 'custom') {



        const customInputs = document.querySelectorAll(

            '.custom-item-input'

        );



        let hasValidCustom = false;



        customInputs.forEach(input => {



            if (input.value.trim() !== '') {

                hasValidCustom = true;

            }

        });



        if (!hasValidCustom) {



            e.preventDefault();



            alert(

                'Vui lòng nhập ít nhất một mặt hàng mới bên ngoài.'

            );



            return;

        }

    }

});
