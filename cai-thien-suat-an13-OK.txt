<?php
// cai-thien-suat-an.php
require_once 'db_connect.php';

// 8. Trang web yêu cầu đăng nhập mới xem được
if (!isset($_SESSION['user_id'])) {
    header('Location: login.php');
    exit;
}

$message = '';
$error = '';
$today = date('Y-m-d');
$current_week = date('Y-\WW'); // VD: 2026-W37

$check_date = $_GET['check_date'] ?? $today;
$view_mode = $_GET['view_mode'] ?? 'today'; // 'today', 'days_list', 'all_week'

// Tự động khởi tạo cấu trúc CSDL phục vụ quản lý suất ăn
try {
    $pdo->exec("
        CREATE TABLE IF NOT EXISTS `meal_menus` (
            `id` INT AUTO_INCREMENT PRIMARY KEY,
            `week_code` VARCHAR(20) NOT NULL,
            `year` INT NOT NULL,
            `day_of_week` VARCHAR(20) NOT NULL,
            `meal_type` VARCHAR(30) DEFAULT 'Bữa Trưa',
            `dish_name` VARCHAR(150) NOT NULL,
            `description` TEXT DEFAULT NULL,
            `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            INDEX (`week_code`)
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");

    // Bảng lưu thông tin kiểm tra thực tế (Nấu đúng / Tỷ lệ đổi): mỗi món 1 lần/ngày
    $pdo->exec("
        CREATE TABLE IF NOT EXISTS `meal_checks` (
            `id` INT AUTO_INCREMENT PRIMARY KEY,
            `menu_id` INT NOT NULL,
            `check_date` DATE NOT NULL,
            `is_correct` TINYINT(1) DEFAULT 1,
            `adjustment_percent` INT DEFAULT 0,
            `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            INDEX (`check_date`),
            INDEX (`menu_id`)
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");

    // Bảng lưu ý kiến đánh giá chất lượng (Không giới hạn số lần, lưu lịch sử ghi chú)
    $pdo->exec("
        CREATE TABLE IF NOT EXISTS `meal_quality_feedback` (
            `id` INT AUTO_INCREMENT PRIMARY KEY,
            `menu_id` INT NOT NULL,
            `check_date` DATE NOT NULL,
            `quality_rating` VARCHAR(50) DEFAULT 'Tốt',
            `quality_note` TEXT DEFAULT NULL,
            `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            INDEX (`check_date`),
            INDEX (`menu_id`)
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");
} catch (PDOException $e) {
    $error = "Lỗi khởi tạo CSDL: " . $e->getMessage();
}

// ==================== XỬ LÝ CÁC YÊU CẦU POST ====================
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    $act = $_POST['action'];

    // 1. Nhập thủ công thực đơn 1 tuần
    if ($act === 'add_menu_manual') {
        $week_code = trim($_POST['week_code'] ?? $current_week);
        $year      = intval($_POST['year'] ?? date('Y'));
        $day_of_w  = trim($_POST['day_of_week'] ?? 'Thứ 2');
        $meal_type = trim($_POST['meal_type'] ?? 'Bữa Trưa');
        $dish_name = trim($_POST['dish_name'] ?? '');
        $desc      = trim($_POST['description'] ?? '');

        if (!empty($dish_name)) {
            try {
                $stmt = $pdo->prepare("
                    INSERT INTO meal_menus (week_code, year, day_of_week, meal_type, dish_name, description)
                    VALUES (:wcode, :yr, :dow, :mtype, :dname, :desc)
                ");
                $stmt->execute([
                    ':wcode' => $week_code, ':yr' => $year, ':dow' => $day_of_w,
                    ':mtype' => $meal_type, ':dname' => $dish_name, ':desc' => $desc
                ]);
                $message = "Đã thêm món ăn vào thực đơn tuần [{$week_code}] thành công!";
            } catch (PDOException $e) {
                $error = "Lỗi thêm thực đơn: " . $e->getMessage();
            }
        }
    }

    // Sửa món ăn trong thực đơn
    if ($act === 'update_menu_item') {
        $menu_id   = intval($_POST['menu_id']);
        $day_of_w  = trim($_POST['day_of_week'] ?? 'Thứ 2');
        $meal_type = trim($_POST['meal_type'] ?? 'Bữa Trưa');
        $dish_name = trim($_POST['dish_name'] ?? '');
        $desc      = trim($_POST['description'] ?? '');

        if (!empty($dish_name) && $menu_id > 0) {
            try {
                $stmt = $pdo->prepare("
                    UPDATE meal_menus 
                    SET day_of_week = :dow, meal_type = :mtype, dish_name = :dname, description = :desc
                    WHERE id = :id
                ");
                $stmt->execute([
                    ':dow' => $day_of_w, ':mtype' => $meal_type, 
                    ':dname' => $dish_name, ':desc' => $desc, ':id' => $menu_id
                ]);
                $message = "Đã cập nhật thông tin món ăn thành công!";
            } catch (PDOException $e) {
                $error = "Lỗi sửa món ăn: " . $e->getMessage();
            }
        }
    }

    // Xóa món ăn khỏi thực đơn
    if ($act === 'delete_menu_item') {
        $menu_id = intval($_POST['menu_id']);
        if ($menu_id > 0) {
            try {
                $pdo->beginTransaction();
                $pdo->prepare("DELETE FROM meal_checks WHERE menu_id = :mid")->execute([':mid' => $menu_id]);
                $pdo->prepare("DELETE FROM meal_quality_feedback WHERE menu_id = :mid")->execute([':mid' => $menu_id]);
                $pdo->prepare("DELETE FROM meal_menus WHERE id = :id")->execute([':id' => $menu_id]);
                
                $pdo->commit();
                $message = "Đã xóa món ăn khỏi thực đơn!";
            } catch (Exception $e) {
                if ($pdo->inTransaction()) $pdo->rollBack();
                $error = "Lỗi xóa món ăn: " . $e->getMessage();
            }
        }
    }

    // Thêm đánh giá chất lượng không giới hạn số lần
    if ($act === 'add_quality_feedback') {
        $menu_id = intval($_POST['menu_id']);
        $c_date  = $_POST['check_date'] ?? $today;
        $rating  = trim($_POST['quality_rating'] ?? 'Tốt');
        $q_note  = trim($_POST['quality_note'] ?? '');

        if ($menu_id > 0) {
            try {
                $stmt = $pdo->prepare("
                    INSERT INTO meal_quality_feedback (menu_id, check_date, quality_rating, quality_note)
                    VALUES (:mid, :cdate, :rat, :qnote)
                ");
                $stmt->execute([':mid' => $menu_id, ':cdate' => $c_date, ':rat' => $rating, ':qnote' => $q_note]);
                $message = "Đã thêm ý kiến đánh giá chất lượng thành công!";
            } catch (PDOException $e) {
                $error = "Lỗi thêm đánh giá: " . $e->getMessage();
            }
        }
    }

    // Sửa đánh giá chất lượng
    if ($act === 'update_quality_feedback') {
        $feedback_id = intval($_POST['feedback_id']);
        $rating = trim($_POST['quality_rating'] ?? 'Tốt');
        $q_note = trim($_POST['quality_note'] ?? '');

        if ($feedback_id > 0) {
            try {
                $stmt = $pdo->prepare("
                    UPDATE meal_quality_feedback 
                    SET quality_rating = :rat, quality_note = :qnote
                    WHERE id = :id
                ");
                $stmt->execute([':rat' => $rating, ':qnote' => $q_note, ':id' => $feedback_id]);
                $message = "Đã cập nhật ý kiến đánh giá thành công!";
            } catch (PDOException $e) {
                $error = "Lỗi sửa đánh giá: " . $e->getMessage();
            }
        }
    }

    // Xóa đánh giá chất lượng
    if ($act === 'delete_quality_feedback') {
        $feedback_id = intval($_POST['feedback_id']);
        if ($feedback_id > 0) {
            try {
                $pdo->prepare("DELETE FROM meal_quality_feedback WHERE id = :id")->execute([':id' => $feedback_id]);
                $message = "Đã xóa ý kiến đánh giá thành công!";
            } catch (PDOException $e) {
                $error = "Lỗi xóa đánh giá: " . $e->getMessage();
            }
        }
    }

    // Trích xuất thông minh từ File Excel / Hình ảnh
    if ($act === 'smart_import_file') {
        $week_code = trim($_POST['week_code'] ?? $current_week);
        $year      = intval($_POST['year'] ?? date('Y'));
        $raw_text  = trim($_POST['raw_import_text'] ?? '');

        if (!empty($raw_text)) {
            try {
                $lines = explode("\n", $raw_text);
                $pdo->beginTransaction();
                $count = 0;
                
                $currentDay = 'Thứ 2';
                foreach ($lines as $line) {
                    $line = trim($line);
                    if (empty($line)) continue;
                    
                    if (stripos($line, 'thứ') !== false || stripos($line, 'ngày') !== false) {
                        $currentDay = $line;
                        continue;
                    }

                    $stmt = $pdo->prepare("
                        INSERT INTO meal_menus (week_code, year, day_of_week, meal_type, dish_name, description)
                        VALUES (:wcode, :yr, :dow, 'Bữa Trưa', :dname, 'Nhập thông minh tự động')
                    ");
                    $stmt->execute([
                        ':wcode' => $week_code, ':yr' => $year, ':dow' => $currentDay, ':dname' => $line
                    ]);
                    $count++;
                }
                $pdo->commit();
                $message = "Đã trích xuất và nhập thành công {$count} món ăn vào thực đơn tuần [{$week_code}]!";
            } catch (Exception $e) {
                if ($pdo->inTransaction()) $pdo->rollBack();
                $error = "Lỗi phân tích file thông minh: " . $e->getMessage();
            }
        } else {
            $error = "Vui lòng nhập nội dung thực đơn cần trích xuất!";
        }
    }

    // Kiểm tra hàng ngày (Nấu đúng & Tỷ lệ đổi - Mỗi món 1 lần/ngày)
    if ($act === 'save_daily_check') {
        $menu_id    = intval($_POST['menu_id']);
        $is_correct = isset($_POST['is_correct']) ? 1 : 0;
        $adj_pct    = intval($_POST['adjustment_percent'] ?? 0);

        $stmtMenuInfo = $pdo->prepare("SELECT week_code, day_of_week FROM meal_menus WHERE id = :mid");
        $stmtMenuInfo->execute([':mid' => $menu_id]);
        $menuInfo = $stmtMenuInfo->fetch();

        if ($menuInfo) {
            sscanf($menuInfo['week_code'], "%d-W%d", $mYear, $mWeek);
            if (!$mYear) $mYear = date('Y');
            if (!$mWeek) $mWeek = date('W');
            $dMap = [
                'Thứ 2' => 1, 'Thứ 3' => 2, 'Thứ 4' => 3, 'Thứ 5' => 4, 
                'Thứ 6' => 5, 'Thứ 7' => 6, 'Chủ Nhật' => 7
            ];
            $dNum = $dMap[trim($menuInfo['day_of_week'])] ?? 1;
            $dtObj = new DateTime();
            $dtObj->setISODate($mYear, $mWeek, $dNum);
            $c_date = $dtObj->format('Y-m-d');

            if ($c_date > $today) {
                $error = "Không thể đánh giá kiểm tra thực tế cho các món ăn trong tương lai!";
            } else {
                try {
                    $chk = $pdo->prepare("SELECT id FROM meal_checks WHERE menu_id = :mid AND check_date = :cdate");
                    $chk->execute([':mid' => $menu_id, ':cdate' => $c_date]);
                    $exists = $chk->fetch();

                    if ($exists) {
                        $stmt = $pdo->prepare("
                            UPDATE meal_checks 
                            SET is_correct = :corr, adjustment_percent = :apct
                            WHERE id = :id
                        ");
                        $stmt->execute([
                            ':corr' => $is_correct, ':apct' => $is_correct ? 0 : $adj_pct, ':id' => $exists['id']
                        ]);
                    } else {
                        $stmt = $pdo->prepare("
                            INSERT INTO meal_checks (menu_id, check_date, is_correct, adjustment_percent)
                            VALUES (:mid, :cdate, :corr, :apct)
                        ");
                        $stmt->execute([
                            ':mid' => $menu_id, ':cdate' => $c_date, ':corr' => $is_correct, ':apct' => $is_correct ? 0 : $adj_pct
                        ]);
                    }
                    $message = "Đã lưu kết quả kiểm tra thực tế ngày " . date('d/m/Y', strtotime($c_date)) . " thành công!";
                } catch (PDOException $e) {
                    $error = "Lỗi lưu kiểm tra: " . $e->getMessage();
                }
            }
        }
    }

    // Mở khóa kiểm tra thực tế (cho phép đánh giá lại)
    if ($act === 'unlock_daily_check') {
        $menu_id = intval($_POST['menu_id']);
        $stmtMenuInfo = $pdo->prepare("SELECT week_code, day_of_week FROM meal_menus WHERE id = :mid");
        $stmtMenuInfo->execute([':mid' => $menu_id]);
        $menuInfo = $stmtMenuInfo->fetch();
        if ($menuInfo) {
            sscanf($menuInfo['week_code'], "%d-W%d", $mYear, $mWeek);
            if (!$mYear) $mYear = date('Y');
            if (!$mWeek) $mWeek = date('W');
            $dMap = [
                'Thứ 2' => 1, 'Thứ 3' => 2, 'Thứ 4' => 3, 'Thứ 5' => 4, 
                'Thứ 6' => 5, 'Thứ 7' => 6, 'Chủ Nhật' => 7
            ];
            $dNum = $dMap[trim($menuInfo['day_of_week'])] ?? 1;
            $dtObj = new DateTime();
            $dtObj->setISODate($mYear, $mWeek, $dNum);
            $c_date = $dtObj->format('Y-m-d');

            try {
                $stmt = $pdo->prepare("DELETE FROM meal_checks WHERE menu_id = :mid AND check_date = :cdate");
                $stmt->execute([':mid' => $menu_id, ':cdate' => $c_date]);
                $message = "Đã mở khóa kiểm tra thực tế để bạn có thể đánh giá lại!";
            } catch (PDOException $e) {
                $error = "Lỗi mở khóa: " . $e->getMessage();
            }
        }
    }

    // Reset toàn bộ thực đơn trong tuần được chọn
    if ($act === 'reset_week_menu') {
        $target_week = trim($_POST['target_week'] ?? $current_week);
        try {
            $pdo->beginTransaction();
            $stmtGetIds = $pdo->prepare("SELECT id FROM meal_menus WHERE week_code = :wcode");
            $stmtGetIds->execute([':wcode' => $target_week]);
            $menuIds = $stmtGetIds->fetchAll(PDO::FETCH_COLUMN);

            if (!empty($menuIds)) {
                $inQuery = implode(',', array_fill(0, count($menuIds), '?'));
                $stmtDelChk = $pdo->prepare("DELETE FROM meal_checks WHERE menu_id IN ($inQuery)");
                $stmtDelChk->execute($menuIds);
                $stmtDelFb = $pdo->prepare("DELETE FROM meal_quality_feedback WHERE menu_id IN ($inQuery)");
                $stmtDelFb->execute($menuIds);
                $stmtDelMenu = $pdo->prepare("DELETE FROM meal_menus WHERE week_code = ?");
                $stmtDelMenu->execute([$target_week]);
            }

            $pdo->commit();
            $message = "Đã reset và xóa sạch thực đơn của tuần [{$target_week}] thành công!";
        } catch (Exception $e) {
            if ($pdo->inTransaction()) $pdo->rollBack();
            $error = "Lỗi reset thực đơn: " . $e->getMessage();
        }
    }
}

// ==================== BỘ LỌC THỜI GIAN THỐNG KÊ (BẢNG & BÁO CÁO) ====================
$stat_period = $_GET['stat_period'] ?? '1_week';
$intervalSQL = "INTERVAL 7 DAY";
if ($stat_period === '2_weeks') $intervalSQL = "INTERVAL 14 DAY";
elseif ($stat_period === '3_weeks') $intervalSQL = "INTERVAL 21 DAY";
elseif ($stat_period === 'month') $intervalSQL = "INTERVAL 1 MONTH";
elseif ($stat_period === 'year') $intervalSQL = "INTERVAL 1 YEAR";

// ==================== BỘ LỌC THỜI GIAN ĐỘC LẬP CHO PHẦN KẾT LUẬN THÔNG MINH ====================
$insight_period = $_GET['insight_period'] ?? '1_week';
$insightIntervalSQL = "INTERVAL 7 DAY";
if ($insight_period === '1_week') $insightIntervalSQL = "INTERVAL 7 DAY";
elseif ($insight_period === '2_weeks') $insightIntervalSQL = "INTERVAL 14 DAY";
elseif ($insight_period === '3_weeks') $insightIntervalSQL = "INTERVAL 21 DAY";
elseif ($insight_period === '1_month') $insightIntervalSQL = "INTERVAL 1 MONTH";
elseif ($insight_period === '2_months') $insightIntervalSQL = "INTERVAL 2 MONTH";
elseif ($insight_period === '3_months') $insightIntervalSQL = "INTERVAL 3 MONTH";
elseif ($insight_period === '6_months') $insightIntervalSQL = "INTERVAL 6 MONTH";
elseif ($insight_period === '1_year') $insightIntervalSQL = "INTERVAL 1 YEAR";

// ==================== XỬ LÝ XUẤT EXCEL BÁO CÁO ====================
if (isset($_GET['export_excel'])) {
    $date_str = date('d-m-Y');
    header("Content-Type: application/vnd.ms-excel; charset=UTF-8");
    header("Content-Disposition: attachment; filename=Bao_Cao_Cai_Dien_Suat_An_{$stat_period}_{$date_str}.xls");
    header("Pragma: no-cache"); header("Expires: 0"); echo "\xEF\xBB\xBF";

    $stmtExp = $pdo->prepare("
        SELECT m.week_code, m.day_of_week, m.meal_type, m.dish_name, c.check_date, c.is_correct, c.adjustment_percent
        FROM meal_checks c
        JOIN meal_menus m ON c.menu_id = m.id
        WHERE c.check_date >= DATE_SUB(:td, $intervalSQL)
        ORDER BY c.check_date DESC, m.id DESC
    ");
    $stmtExp->execute([':td' => $today]);
    $exportRows = $stmtExp->fetchAll();

    echo '<html xmlns:x="urn:schemas-microsoft-com:office:excel"><head><meta http-equiv="Content-Type" content="text/html; charset=utf-8">
    <style>
        .rpt-title { font-family:"Times New Roman", Arial; font-size:16pt; font-weight:bold; text-align:center; height:35px; } 
        .th-c { background-color:#0284c7; color:#ffffff; font-weight:bold; text-align:center; border:0.5pt solid #000; height:28px; } 
        .td-c { text-align:center; border:0.5pt solid #000; height:25px; } 
        .td-l { text-align:left; border:0.5pt solid #000; padding-left:6px; } 
    </style></head><body>';
    echo '<table border="1">';
    echo '<tr><td colspan="7" class="rpt-title">BÁO CÁO CHI TIẾT CẢI THIỆN CHẤT LƯỢNG SUẤT ĂN (Mốc: ' . strtoupper(str_replace('_', ' ', $stat_period)) . ')</td></tr>';
    echo '<tr><td colspan="7" style="text-align:center; font-style:italic;">Thời gian xuất: ' . date('d/m/Y H:i:s') . '</td></tr><tr><td colspan="7" style="border:none; height:10px;"></td></tr>';

    echo '<tr>
            <th class="th-c" width="45">STT</th>
            <th class="th-c" width="100">Mã Tuần</th>
            <th class="th-c" width="100">Ngày Kiểm Tra</th>
            <th class="th-c" width="90">Thứ / Buổi</th>
            <th class="th-c" width="220">Tên Món Ăn</th>
            <th class="th-c" width="90">Nấu Đúng?</th>
            <th class="th-c" width="90">Tỷ Lệ Đổi</th>
          </tr>';

    $stt = 1;
    if (!empty($exportRows)) {
        foreach ($exportRows as $row) {
            $dungSai = intval($row['is_correct']) === 1 ? 'Đúng thực đơn' : 'Nấu sai/Đổi món';
            $tyle = intval($row['is_correct']) === 1 ? '0%' : intval($row['adjustment_percent']) . '%';
            echo "<tr>
                    <td class='td-c'>{$stt}</td>
                    <td class='td-c'>" . htmlspecialchars($row['week_code']) . "</td>
                    <td class='td-c'>" . date('d/m/Y', strtotime($row['check_date'])) . "</td>
                    <td class='td-c'>" . htmlspecialchars($row['day_of_week']) . " (" . htmlspecialchars($row['meal_type']) . ")</td>
                    <td class='td-l'><b>" . htmlspecialchars($row['dish_name']) . "</b></td>
                    <td class='td-c'>{$dungSai}</td>
                    <td class='td-c'>{$tyle}</td>
                  </tr>";
            $stt++;
        }
    } else {
        echo '<tr><td colspan="7" class="td-c" style="height:35px; font-style:italic;">Không có dữ liệu trong khoảng thời gian này</td></tr>';
    }

    echo '</table></body></html>';
    exit;
}

// ==================== TRUY VẤN DỮ LIỆU HIỂN THỊ ====================
$availableWeeks = [];
try {
    $availableWeeks = $pdo->query("SELECT DISTINCT week_code FROM meal_menus ORDER BY week_code DESC")->fetchAll(PDO::FETCH_COLUMN);
} catch (PDOException $e) {}

$selected_week = $_GET['f_week'] ?? ($availableWeeks[0] ?? $current_week);

$timestampCheck = strtotime($check_date);
$dayOfWeekNum = date('N', $timestampCheck); // 1 (Mon) đến 7 (Sun)
$mapDays = [
    1 => 'Thứ 2', 2 => 'Thứ 3', 3 => 'Thứ 4', 4 => 'Thứ 5', 
    5 => 'Thứ 6', 6 => 'Thứ 7', 7 => 'Chủ Nhật'
];
$currentSelectedDayName = $mapDays[$dayOfWeekNum] ?? 'Thứ 2';

sscanf($selected_week, "%d-W%d", $selYear, $selWeek);
if (!$selYear) $selYear = date('Y');
if (!$selWeek) $selWeek = date('W');

$dowMap = [
    'Thứ 2' => 1, 'Thứ 3' => 2, 'Thứ 4' => 3, 'Thứ 5' => 4, 
    'Thứ 6' => 5, 'Thứ 7' => 6, 'Chủ Nhật' => 7
];

$todayMenus = [];
try {
    $sqlQuery = "
        SELECT m.* 
        FROM meal_menus m 
        WHERE m.week_code = :wcode
    ";
    
    if ($view_mode === 'today') {
        $sqlQuery .= " AND m.day_of_week = :dow";
    }
    
    $sqlQuery .= " ORDER BY FIELD(m.day_of_week, 'Thứ 2', 'Thứ 3', 'Thứ 4', 'Thứ 5', 'Thứ 6', 'Thứ 7', 'Chủ Nhật'), m.id DESC";

    $stmtM = $pdo->prepare($sqlQuery);
    $params = [':wcode' => $selected_week];
    if ($view_mode === 'today') {
        $params[':dow'] = $currentSelectedDayName;
    }
    $stmtM->execute($params);
    $rawMenus = $stmtM->fetchAll();

    foreach ($rawMenus as $m) {
        $dNum = $dowMap[trim($m['day_of_week'])] ?? 1;
        $dto = new DateTime();
        $dto->setISODate($selYear, $selWeek, $dNum);
        $m['item_date'] = $dto->format('Y-m-d');
        $m['is_future'] = ($m['item_date'] > $today);

        $chkStmt = $pdo->prepare("SELECT id, is_correct, adjustment_percent FROM meal_checks WHERE menu_id = :mid AND check_date = :cdate");
        $chkStmt->execute([':mid' => $m['id'], ':cdate' => $m['item_date']]);
        $chkData = $chkStmt->fetch();

        if ($chkData) {
            $m['check_record_id'] = $chkData['id'];
            $m['is_correct'] = $chkData['is_correct'];
            $m['adjustment_percent'] = $chkData['adjustment_percent'];
        } else {
            $m['check_record_id'] = null;
            $m['is_correct'] = 1;

<script>
function openModal(id) { document.getElementById(id).style.display = 'flex'; }
function closeModal(id) { document.getElementById(id).style.display = 'none'; }

function openEditMenuModal(m) {
    document.getElementById('edit_menu_id').value = m.id;
    document.getElementById('edit_day_of_week').value = m.day_of_week;
    document.getElementById('edit_meal_type').value = m.meal_type || 'Bữa Trưa';
    document.getElementById('edit_dish_name').value = m.dish_name;
    document.getElementById('edit_description').value = m.description || '';
    openModal('modal-edit-menu');
}

function openEditFeedbackModal(fb) {
    document.getElementById('edit_feedback_id').value = fb.id;
    document.getElementById('edit_feedback_rating').value = fb.quality_rating;
    document.getElementById('edit_feedback_note').value = fb.quality_note || '';
    openModal('modal-edit-feedback');
}

function simulateSmartOCR(input) {
    if (input.files && input.files[0]) {
        const fileName = input.files[0].name;
        const txtArea = document.getElementById('raw_import_text');
        txtArea.value = `Thứ 2\nThịt kho tàu chuẩn vị\nCanh chua cá lóc\n(Đã quét thông minh từ file: ${fileName})`;
    }
}
window.onclick = function(e) {
    if (e.target.classList.contains('modal-overlay')) {
        e.target.style.display = 'none';
    }
}
</script>
