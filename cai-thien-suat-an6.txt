$message = '';
$error = '';
$today = date('Y-m-d');
$current_week = date('Y-\WW'); // VD: 2026-W37

$check_date = $_GET['check_date'] ?? $today;
$f_week = $_GET['f_week'] ?? 'all'; 

// Khởi tạo CSDL an toàn tuyệt đối
try {
    $pdo->exec("
        CREATE TABLE IF NOT EXISTS `meal_menus` (
            `id` INT AUTO_INCREMENT PRIMARY KEY,
            `week_code` VARCHAR(20) NOT NULL,
            `year` INT NOT NULL,
            `day_of_week` VARCHAR(20) NOT NULL,
            `meal_type` VARCHAR(30) DEFAULT 'Bữa Trưa',
            `dish_name` VARCHAR(150) NOT NULL,
            `description` TEXT DEFAULT NULL,
            `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");

    $pdo->exec("
        CREATE TABLE IF NOT EXISTS `meal_checks` (
            `id` INT AUTO_INCREMENT PRIMARY KEY,
            `menu_id` INT NOT NULL,
            `check_date` DATE NOT NULL,
            `is_correct` TINYINT(1) DEFAULT 1,
            `adjustment_percent` INT DEFAULT 0,
            `is_locked` TINYINT(1) DEFAULT 0,
            `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");

    $pdo->exec("
        CREATE TABLE IF NOT EXISTS `meal_quality_feedback` (
            `id` INT AUTO_INCREMENT PRIMARY KEY,
            `menu_id` INT NOT NULL,
            `check_date` DATE NOT NULL,
            `quality_rating` VARCHAR(50) DEFAULT 'Tốt',
            `quality_note` TEXT DEFAULT NULL,
            `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");
} catch (PDOException $e) {
    $error = "Lỗi khởi tạo CSDL: " . $e->getMessage();
}

// ==================== XỬ LÝ POST ====================
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    $act = $_POST['action'];

    if ($act === 'add_menu_manual') {
        $week_code = trim($_POST['week_code'] ?? $current_week);
        $year      = intval($_POST['year'] ?? date('Y'));
        $day_of_w  = trim($_POST['day_of_week'] ?? 'Thứ 2');
        $meal_type = trim($_POST['meal_type'] ?? 'Bữa Trưa');
        $dish_name = trim($_POST['dish_name'] ?? '');
        $desc      = trim($_POST['description'] ?? '');

        if (!empty($dish_name)) {
            try {
                $stmt = $pdo->prepare("
                    INSERT INTO meal_menus (week_code, year, day_of_week, meal_type, dish_name, description)
                    VALUES (:wcode, :yr, :dow, :mtype, :dname, :desc)
                ");
                $stmt->execute([
                    ':wcode' => $week_code, ':yr' => $year, ':dow' => $day_of_w,
                    ':mtype' => $meal_type, ':dname' => $dish_name, ':desc' => $desc
                ]);
                $message = "Đã thêm món ăn thành công!";
            } catch (PDOException $e) {
                $error = "Lỗi thêm thực đơn: " . $e->getMessage();
            }
        }
    }

    if ($act === 'update_menu_item') {
        $menu_id   = intval($_POST['menu_id']);
        $week_code = trim($_POST['week_code'] ?? $current_week);
        $day_of_w  = trim($_POST['day_of_week'] ?? 'Thứ 2');
        $meal_type = trim($_POST['meal_type'] ?? 'Bữa Trưa');
        $dish_name = trim($_POST['dish_name'] ?? '');
        $desc      = trim($_POST['description'] ?? '');

        if (!empty($dish_name) && $menu_id > 0) {
            try {
                $stmt = $pdo->prepare("
                    UPDATE meal_menus 
                    SET week_code = :wcode, day_of_week = :dow, meal_type = :mtype, dish_name = :dname, description = :desc
                    WHERE id = :id
                ");
                $stmt->execute([
                    ':wcode' => $week_code, ':dow' => $day_of_w, ':mtype' => $meal_type, 
                    ':dname' => $dish_name, ':desc' => $desc, ':id' => $menu_id
                ]);
                $message = "Đã cập nhật món ăn thành công!";
            } catch (PDOException $e) {
                $error = "Lỗi sửa món ăn: " . $e->getMessage();
            }
        }
    }

    if ($act === 'delete_menu_item') {
        $menu_id = intval($_POST['menu_id']);
        if ($menu_id > 0) {
            try {
                $pdo->prepare("DELETE FROM meal_checks WHERE menu_id = ?")->execute([$menu_id]);
                $pdo->prepare("DELETE FROM meal_quality_feedback WHERE menu_id = ?")->execute([$menu_id]);
                $pdo->prepare("DELETE FROM meal_menus WHERE id = ?")->execute([$menu_id]);
                $message = "Đã xóa món ăn!";
            } catch (Exception $e) {
                $error = "Lỗi xóa món ăn: " . $e->getMessage();
            }
        }
    }

    if ($act === 'reset_all_meal_data') {
        try {
            $pdo->exec("TRUNCATE TABLE meal_checks");
            $pdo->exec("TRUNCATE TABLE meal_quality_feedback");
            $pdo->exec("TRUNCATE TABLE meal_menus");
            $message = "Đã reset toàn bộ dữ liệu về 0!";
        } catch (PDOException $e) {
            $error = "Lỗi reset: " . $e->getMessage();
        }
    }

    if ($act === 'add_quality_feedback') {
        $menu_id = intval($_POST['menu_id']);
        $c_date  = $_POST['check_date'] ?? $today;
        $rating  = trim($_POST['quality_rating'] ?? 'Tốt');
        $q_note  = trim($_POST['quality_note'] ?? '');

        if ($menu_id > 0) {
            try {
                $stmt = $pdo->prepare("INSERT INTO meal_quality_feedback (menu_id, check_date, quality_rating, quality_note) VALUES (?, ?, ?, ?)");
                $stmt->execute([$menu_id, $c_date, $rating, $q_note]);
                $message = "Đã lưu đánh giá chất lượng!";
            } catch (PDOException $e) {
                $error = "Lỗi đánh giá: " . $e->getMessage();
            }
        }
    }

    if ($act === 'delete_quality_feedback') {
        $feedback_id = intval($_POST['feedback_id']);
        if ($feedback_id > 0) {
            $pdo->prepare("DELETE FROM meal_quality_feedback WHERE id = ?")->execute([$feedback_id]);
            $message = "Đã xóa đánh giá chất lượng!";
        }
    }

    if ($act === 'save_daily_check') {
        $menu_id    = intval($_POST['menu_id']);
        $c_date     = $_POST['check_date'] ?? $today;
        $is_correct = isset($_POST['is_correct']) ? 1 : 0;
        $adj_pct    = intval($_POST['adjustment_percent'] ?? 0);

        $chk = $pdo->prepare("SELECT id FROM meal_checks WHERE menu_id = ? AND check_date = ?");
        $chk->execute([$menu_id, $c_date]);
        if ($chk->fetch()) {
            $pdo->prepare("UPDATE meal_checks SET is_correct = ?, adjustment_percent = ?, is_locked = 1 WHERE menu_id = ? AND check_date = ?")
                ->execute([$is_correct, $is_correct ? 0 : $adj_pct, $menu_id, $c_date]);
        } else {
            $pdo->prepare("INSERT INTO meal_checks (menu_id, check_date, is_correct, adjustment_percent, is_locked) VALUES (?, ?, ?, ?, 1)")
                ->execute([$menu_id, $c_date, $is_correct, $is_correct ? 0 : $adj_pct]);
        }
        $message = "Đã lưu và khóa kết quả!";
    }

    if ($act === 'unlock_daily_check') {
        $menu_id = intval($_POST['menu_id']);
        $c_date  = $_POST['check_date'] ?? $today;
        $pdo->prepare("UPDATE meal_checks SET is_locked = 0 WHERE menu_id = ? AND check_date = ?")->execute([$menu_id, $c_date]);
        $message = "Đã mở khóa đánh giá!";
    }
}

            LEFT JOIN meal_checks c ON m.id = c.menu_id AND c.check_date = :cdate
