$today = date('Y-m-d');

// 1. Thống kê hợp đồng (Sắp hết hạn trong 30 ngày & Đã quá hạn)
$contract_expiring_count = 0;
$contract_expired_count = 0;
try {
    $contract_expiring_count = $pdo->query("SELECT COUNT(*) FROM contracts WHERE status = 'Hiệu lực' AND contract_type != 'Không xác định thời hạn' AND end_date >= '$today' AND end_date <= DATE_ADD('$today', INTERVAL 30 DAY)")->fetchColumn();
    $contract_expired_count = $pdo->query("SELECT COUNT(*) FROM contracts WHERE status = 'Hiệu lực' AND contract_type != 'Không xác định thời hạn' AND end_date < '$today'")->fetchColumn();
} catch (PDOException $e) {}

// 2. Thống kê chấm công hôm nay (Đi làm, Nghỉ phép, Về sớm, Tăng ca)
$att_working_count = 0;
$att_leave_count = 0;
$att_early_count = 0;
$att_ot_count = 0;
try {
    $att_working_count = $pdo->query("SELECT COUNT(*) FROM attendance_details WHERE work_date = '$today' AND shift_status IN ('1.0', '0.5', 'CT')")->fetchColumn();
    $att_leave_count = $pdo->query("SELECT COUNT(*) FROM attendance_details WHERE work_date = '$today' AND shift_status IN ('PN', 'Ô', 'KP')")->fetchColumn();
    $att_early_count = $pdo->query("SELECT COUNT(*) FROM attendance_details WHERE work_date = '$today' AND early_minutes > 0")->fetchColumn();
    $att_ot_count = $pdo->query("SELECT COUNT(*) FROM attendance_details WHERE work_date = '$today' AND (ot_normal > 0 OR ot_weekend > 0)")->fetchColumn();
} catch (PDOException $e) {}

// 3. Thống kê số lượng suất ăn hôm nay (Đồng bộ chuẩn 100% theo logic của quan-ly-bep.php)
$today_lunch_count = 0;
try {
    $stmtLunch = $pdo->prepare("
        SELECT COUNT(*) FROM employees e
        JOIN attendance_details a ON e.id = a.employee_id
        WHERE a.work_date = :today 
          AND a.shift_status IN ('1.0', '0.5', 'CT')
          AND (a.skip_lunch IS NULL OR a.skip_lunch = 0)
    ");
    $stmtLunch->execute([':today' => $today]);
    $today_lunch_count = $stmtLunch->fetchColumn();
} catch (PDOException $e) {
    $today_lunch_count = 0;
}

// Lấy danh sách việc cần lưu ý gấp từ bộ phận nhân sự (nếu có bảng hr_tasks)
$hr_urgent_tasks = [];
try {
    $stmt = $pdo->query("SELECT task_name, task_url, is_urgent FROM hr_tasks WHERE status = 'pending' AND is_urgent = 1 ORDER BY id DESC LIMIT 3");
    $hr_urgent_tasks = $stmt->fetchAll();
} catch (PDOException $e) {
    $hr_urgent_tasks = [];
