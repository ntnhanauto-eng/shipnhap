function renderPaginationBar($currentPage, $totalPages, $paramName) {
    if ($totalPages <= 1) return;
    $queryParams = $_GET;

    echo '<div class="pagination-wrapper">';
    if ($currentPage > 1) {
        $queryParams[$paramName] = $currentPage - 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&laquo;</a>';
    } else {
        echo '<span class="page-link disabled">&laquo;</span>';
    }

    $start = max(1, $currentPage - 1);
    $end = min($totalPages, $currentPage + 1);
    if ($currentPage == 1) $end = min($totalPages, $start + 2);
    elseif ($currentPage == $totalPages) $start = max(1, $end - 2);

    for ($i = $start; $i <= $end; $i++) {
        $queryParams[$paramName] = $i;
        $activeClass = ($i == $currentPage) ? 'active' : '';
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link ' . $activeClass . '">' . $i . '</a>';
    }

    if ($currentPage < $totalPages) {
        $queryParams[$paramName] = $currentPage + 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&raquo;</a>';
    } else {
        echo '<span class="page-link disabled">&raquo;</span>';
    }
    echo '</div>';
}

function generateUniqueVppCode($pdo, $table, $timeStr, $excludeId = 0) {
    $ts = strtotime($timeStr) ?: time();
    $baseCode = date('dmYHis', $ts);

    $sql = "SELECT request_code FROM `{$table}` WHERE request_code LIKE :b AND id != :exId";
    $stmt = $pdo->prepare($sql);
    $stmt->execute([':b' => $baseCode . '%', ':exId' => $excludeId]);
    $existing = $stmt->fetchAll(PDO::FETCH_COLUMN);

    if (empty($existing) || !in_array($baseCode, $existing)) {
        return $baseCode;
    }

    $suffixes = range('A', 'Z');
    foreach ($suffixes as $char) {
        $candidate = $baseCode . $char;
        if (!in_array($candidate, $existing)) {
            return $candidate;
        }
    }
    return $baseCode . '_' . rand(10, 99);
}

}

function normalizeFastText($str) {
    $str = str_replace("\xC2\xA0", ' ', (string)$str);
    $str = mb_strtolower(trim($str), 'UTF-8');
    $str = preg_replace('/[áàảãạăắằẳẵặâấầẩẫậ]/u', 'a', $str);
    $str = preg_replace('/[éèẻẽẹêếềểễệ]/u', 'e', $str);
    $str = preg_replace('/[íìỉĩị]/u', 'i', $str);
    $str = preg_replace('/[óòỏõọôốồổỗộơớờởỡợ]/u', 'o', $str);
    $str = preg_replace('/[úùủũụưứừửữự]/u', 'u', $str);
    $str = preg_replace('/[ýỳỷỹỵ]/u', 'y', $str);
    $str = preg_replace('/đ/u', 'd', $str);
    return preg_replace('/[^a-z0-9]/', '', $str);
}

function parseExcelDate($rawDate, $defaultToday) {
    $rawDate = trim(str_replace("\xC2\xA0", ' ', (string)$rawDate));
    if (empty($rawDate)) return $defaultToday;

    if (is_numeric($rawDate) && intval($rawDate) > 30000 && intval($rawDate) < 60000) {
        $days = intval($rawDate) - 25569;
        return gmdate('Y-m-d', $days * 86400);
    }

    $rawDate = str_replace('/', '-', $rawDate);
    $ts = strtotime($rawDate);
    if ($ts && date('Y', $ts) > 2000) {
        return date('Y-m-d', $ts);
    }
    return $defaultToday;
}

function parseXlsxFastStream($filepath) {
    if (!class_exists('ZipArchive')) return null;
    $zip = new ZipArchive();
    if ($zip->open($filepath) !== true) return null;

    $strings = [];
    $sharedStrXml = $zip->getFromName('xl/sharedStrings.xml');
    if ($sharedStrXml) {
        $reader = new XMLReader();
        $reader->XML($sharedStrXml);
        $currText = '';
        $inT = false;
        while ($reader->read()) {
            if ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 'si') {
                $currText = '';
            } elseif ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 't') {
                $inT = true;
            } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 't') {
                $inT = false;
            } elseif ($inT && ($reader->nodeType === XMLReader::TEXT || $reader->nodeType === XMLReader::CDATA)) {
                $currText .= $reader->value;
            } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 'si') {
                $strings[] = $currText;
            }
        }
        $reader->close();
    }

    $sheetXml = $zip->getFromName('xl/worksheets/sheet1.xml');
    if (!$sheetXml) {
        $zip->close();
        return null;
    }

    $rows = [];
    $reader = new XMLReader();
    $reader->XML($sheetXml);

    $currentRow = [];
    $currentVal = '';
    $cellType = '';
    $inVal = false;

    while ($reader->read()) {
        if ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 'row') {
            $currentRow = [];
        } elseif ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 'c') {
            $cellType = $reader->getAttribute('t') ?? '';
            $currentVal = '';
        } elseif ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 'v') {
            $inVal = true;
        } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 'v') {
            $inVal = false;
        } elseif ($inVal && ($reader->nodeType === XMLReader::TEXT || $reader->nodeType === XMLReader::CDATA)) {
            $currentVal .= $reader->value;
        } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 'c') {
            $val = trim($currentVal);
            if ($cellType === 's' && isset($strings[intval($val)])) {
                $val = trim($strings[intval($val)]);
            }
            $currentRow[] = $val;
        } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 'row') {
            if (!empty(array_filter($currentRow, fn($v) => $v !== ''))) {
                $rows[] = $currentRow;
            }
        }
    }
    $reader->close();
    $zip->close();
    return $rows;
}

// 1A. Phê duyệt theo mã yêu cầu
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'approve_group_proposals') {
    $req_code       = trim($_POST['request_code'] ?? '');
    $items_approved = $_POST['approved_quantities'] ?? [];
    $approval_note  = trim($_POST['approval_note'] ?? '');

    try {
        if (empty($req_code)) throw new Exception("Không xác định được mã yêu cầu đề xuất!");
        if (empty($items_approved)) throw new Exception("Không có danh sách mặt hàng để phê duyệt!");

        $pdo->beginTransaction();
        $upCount = 0;
        foreach ($items_approved as $propId => $qty) {
            $q = max(1, intval($qty));
            $stmtUp = $pdo->prepare("
                UPDATE vpp_proposals 
                SET status = 'Đã duyệt', approved_quantity = :appr_qty, approval_note = :appr_note 
                WHERE id = :id AND request_code = :code
            ");
            $stmtUp->execute([
                ':appr_qty'  => $q,
                ':appr_note' => $approval_note,
                ':id'        => intval($propId),
                ':code'      => $req_code
            ]);
            $upCount++;
        }
        $pdo->commit();
        $message = "Đã phê duyệt toàn bộ {$upCount} mặt hàng trong yêu cầu [{$req_code}] thành công!";
    } catch (Exception $e) {
        if ($pdo->inTransaction()) $pdo->rollBack();
        $error = "Lỗi duyệt đề xuất: " . $e->getMessage();
    }
}

// 1C. Cấp phát theo mã yêu cầu
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'deliver_group_proposals') {
    $req_code      = trim($_POST['request_code'] ?? '');
    $receiver_name = trim($_POST['receiver_name'] ?? '');
    $department    = trim($_POST['department'] ?? '');
    $delivery_loc  = trim($_POST['delivery_location'] ?? '');
    $custom_notes  = trim($_POST['notes'] ?? '');
    $deliver_items = $_POST['deliver_quantities'] ?? [];

    try {
        if (empty($req_code)) throw new Exception("Không xác định được mã yêu cầu đề xuất!");
        $pdo->beginTransaction();

        $stmtProps = $pdo->prepare("
            SELECT p.*, i.item_name, i.stock_qty, i.unit 
            FROM vpp_proposals p 
            JOIN vpp_items i ON p.item_id = i.id 
            WHERE p.request_code = :code AND p.status IN ('Đã duyệt', 'Chờ cấp phát')
            FOR UPDATE
        ");
        $stmtProps->execute([':code' => $req_code]);
        $rows = $stmtProps->fetchAll();

        if (empty($rows)) {
            throw new Exception("Yêu cầu đề xuất chưa được duyệt hoặc đã được cấp phát từ trước!");
        }

        $stmtSub = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty - :qty WHERE id = :id");
        $stmtReq = $pdo->prepare("
            INSERT INTO vpp_requests (request_code, request_date, department, receiver_name, item_id, quantity, delivery_location, notes)
            VALUES (:code, :r_date, :dept, :rec, :item_id, :qty, :loc, :notes)
        ");
        $stmtUpProp = $pdo->prepare("
            UPDATE vpp_proposals 
            SET status = 'Đã cấp phát', approved_quantity = :appr_q 
            WHERE id = :id
        ");

        $itemDeliveredCount = 0;
        foreach ($rows as $p) {
            $propId = $p['id'];
            $giveQty = isset($deliver_items[$propId]) ? max(1, intval($deliver_items[$propId])) : intval($p['approved_quantity'] ?: $p['quantity']);

            if ($p['stock_qty'] < $giveQty) {
                throw new Exception("Tồn kho không đủ để cấp mặt hàng [{$p['item_name']}]! Tồn kho hiện có: {$p['stock_qty']} {$p['unit']}, cần cấp: {$giveQty} {$p['unit']}.");
            }

            $stmtSub->execute([':qty' => $giveQty, ':id' => $p['item_id']]);

            $stmtReq->execute([
                ':code'    => $req_code,
                ':r_date'  => date('Y-m-d'),
                ':dept'    => $department ?: $p['department'],
                ':rec'     => $receiver_name ?: $p['receiver_name'],
                ':item_id' => $p['item_id'],
                ':qty'     => $giveQty,
                ':loc'     => $delivery_loc ?: $p['delivery_location'],
                ':notes'   => $custom_notes
            ]);

            $stmtUpProp->execute([':appr_q' => $giveQty, ':id' => $propId]);
            $itemDeliveredCount++;
        }

        $pdo->commit();
        $message = "Đã xác nhận cấp phát thành công {$itemDeliveredCount} mặt hàng cho yêu cầu [{$req_code}]!";
    } catch (Exception $e) {
        if ($pdo->inTransaction()) $pdo->rollBack();
        $error = "Lỗi cấp phát: " . $e->getMessage();
    }
}

// 2B. Xử lý Excel nhập kho VPP: Đọc & Xem trước
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'import_stock_excel') {
    if (isset($_FILES['excel_file']) && $_FILES['excel_file']['error'] === UPLOAD_ERR_OK) {
        $tmpName  = $_FILES['excel_file']['tmp_name'];
        $fileName = $_FILES['excel_file']['name'];
        $ext      = strtolower(pathinfo($fileName, PATHINFO_EXTENSION));

        $rows = [];
        if ($ext === 'csv') {
            if (($handle = fopen($tmpName, 'r')) !== false) {
                while (($data = fgetcsv($handle, 4000, ',')) !== false) {
                    $rows[] = array_map('trim', $data);
                }
                fclose($handle);
            }
        } elseif ($ext === 'xlsx') {
            $rows = parseXlsxFastStream($tmpName);
            if ($rows === null) {
                $error = "Máy chủ PHP cần bật tiện ích ZipArchive để đọc file .xlsx nhanh hoặc hãy lưu file dưới dạng CSV / XLS!";
            }
        } elseif ($ext === 'xls') {
            $content = file_get_contents($tmpName);
            if (strpos($content, '<table') !== false) {
                $dom = new DOMDocument();
                @$dom->loadHTML(mb_convert_encoding($content, 'HTML-ENTITIES', 'UTF-8'));
                $trs = $dom->getElementsByTagName('tr');
                foreach ($trs as $tr) {
                    $rowValues = [];
                    $tds = $tr->getElementsByTagName('td');
                    if ($tds->length === 0) $tds = $tr->getElementsByTagName('th');
                    foreach ($tds as $td) {
                        $rowValues[] = trim($td->textContent);
                    }
                    if (!empty(array_filter($rowValues, fn($v) => $v !== ''))) {
                        $rows[] = $rowValues;
                    }
                }
            } else {
                $rows = parseXlsxFastStream($tmpName);
            }
        }

        if (!empty($rows)) {
            $header = array_shift($rows);
            $idxDate = -1; $idxSup = -1; $idxCode = -1; $idxName = -1; 
            $idxSpec = -1; $idxUnit = -1; $idxQty = -1; $idxInv = -1; $idxNote = -1;

            foreach ($header as $colIdx => $colName) {
                $cNorm = normalizeFastText($colName);
                if ($idxDate === -1 && (strpos($cNorm, 'ngay') !== false || strpos($cNorm, 'date') !== false)) {
                    $idxDate = $colIdx;
                } elseif ($idxSup === -1 && (strpos($cNorm, 'ncc') !== false || strpos($cNorm, 'nhacungcap') !== false || strpos($cNorm, 'supplier') !== false)) {
                    $idxSup = $colIdx;
                } elseif ($idxCode === -1 && (strpos($cNorm, 'mavpp') !== false || strpos($cNorm, 'masp') !== false || strpos($cNorm, 'code') !== false || $cNorm === 'ma')) {
                    $idxCode = $colIdx;
                } elseif ($idxName === -1 && (strpos($cNorm, 'ten') !== false || strpos($cNorm, 'mathang') !== false || strpos($cNorm, 'item') !== false)) {
                    $idxName = $colIdx;
                } elseif ($idxSpec === -1 && (strpos($cNorm, 'quycach') !== false || strpos($cNorm, 'donggoi') !== false || strpos($cNorm, 'spec') !== false)) {
                    $idxSpec = $colIdx;
                } elseif ($idxUnit === -1 && (strpos($cNorm, 'dvt') !== false || strpos($cNorm, 'donvi') !== false || strpos($cNorm, 'unit') !== false)) {
                    $idxUnit = $colIdx;
                } elseif ($idxQty === -1 && (strpos($cNorm, 'soluong') !== false || strpos($cNorm, 'sl') !== false || strpos($cNorm, 'qty') !== false || strpos($cNorm, 'quantity') !== false)) {
                    $idxQty = $colIdx;
                } elseif ($idxInv === -1 && (strpos($cNorm, 'hoadon') !== false || strpos($cNorm, 'chungtu') !== false || strpos($cNorm, 'invoice') !== false || strpos($cNorm, 'sohd') !== false)) {
                    $idxInv = $colIdx;
                } elseif ($idxNote === -1 && (strpos($cNorm, 'ghichu') !== false || strpos($cNorm, 'note') !== false)) {
                    $idxNote = $colIdx;
                }
            }

            if ($idxDate === -1) $idxDate = 0;
            if ($idxSup  === -1) $idxSup  = 1;
            if ($idxCode === -1) $idxCode = 2;
            if ($idxName === -1) $idxName = 3;
            if ($idxSpec === -1) $idxSpec = 4;
            if ($idxUnit === -1) $idxUnit = 5;
            if ($idxQty  === -1) $idxQty  = 6;
            if ($idxInv  === -1) $idxInv  = 7;
            if ($idxNote === -1) $idxNote = 8;

            $allItems = $pdo->query("SELECT id, item_code, item_name, specification, unit, stock_qty FROM vpp_items")->fetchAll();
            $itemMapByCode = [];
            $itemMapByName = [];
            $simulatedStockMap = [];

            foreach ($allItems as $it) {
                if (!empty($it['item_code'])) {
                    $itemMapByCode[normalizeFastText($it['item_code'])] = $it;
                }
                if (!empty($it['item_name'])) {
                    $itemMapByName[normalizeFastText($it['item_name'])] = $it;
                }
                $simulatedStockMap[$it['id']] = intval($it['stock_qty']);
            }

            $validPreview = [];
            $failedPreview = [];

            foreach ($rows as $index => $row) {
                $rowNum   = $index + 2;
                $rawDate  = trim($row[$idxDate] ?? '');
                $supplier = trim($row[$idxSup]  ?? '');
                $code     = trim($row[$idxCode] ?? '');
                $name     = trim($row[$idxName] ?? '');
                $spec     = ($idxSpec !== -1) ? trim($row[$idxSpec] ?? '') : '';
                $unit     = ($idxUnit !== -1) ? trim($row[$idxUnit] ?? '') : '';
                $rawQty   = $row[$idxQty] ?? '';
                $invoice  = ($idxInv !== -1)  ? trim($row[$idxInv]  ?? '') : '';
                $note     = ($idxNote !== -1) ? trim($row[$idxNote] ?? '') : '';

                if (empty($code) && empty($name)) {
                    $failedPreview[] = [
                        'row' => $rowNum, 'code' => $code, 'name' => $name, 'qty' => $rawQty,
                        'reason' => 'Thiếu cả Mã VPP và Tên Mặt Hàng'
                    ];
                    continue;
                }

                $cleanQtyStr = preg_replace('/[^0-9]/', '', (string)$rawQty);
                $qty = intval($cleanQtyStr);
                if ($qty <= 0) {
                    $failedPreview[] = [
                        'row' => $rowNum, 'code' => $code, 'name' => $name, 'qty' => $rawQty,
                        'reason' => "Số lượng nhập '{$rawQty}' không hợp lệ (phải là số nguyên > 0)"
                    ];
                    continue;
                }

                if (empty($supplier)) $supplier = 'Nhà cung cấp tổng hợp';
                $parsedDate = parseExcelDate($rawDate, $today);

                $matched = null;
                $cKey = !empty($code) ? normalizeFastText($code) : '';
                $nKey = !empty($name) ? normalizeFastText($name) : '';

                if (!empty($cKey) && isset($itemMapByCode[$cKey])) {
                    $matched = $itemMapByCode[$cKey];
                } elseif (!empty($nKey) && isset($itemMapByName[$nKey])) {
                    $matched = $itemMapByName[$nKey];
                }

                $isNewItem = false;
                if (!$matched) {
                    $isNewItem = true;
                    $calcCode = $code ?: ('VPP_' . strtoupper(substr(md5(uniqid()), 0, 6)));
                    $calcName = $name ?: ('Mặt hàng mã ' . $calcCode);
                    $calcUnit = $unit ?: 'Cái';
                    $prevStock = 0;
                    $newStock = $qty;

                    $dummyId = 'temp_' . uniqid();
                    $matched = [
                        'id' => $dummyId,
                        'item_code' => $calcCode,
                        'item_name' => $calcName,
                        'specification' => $spec,
                        'unit' => $calcUnit,
                        'stock_qty' => 0
                    ];
                    $simulatedStockMap[$dummyId] = $newStock;
                    $itemMapByCode[normalizeFastText($calcCode)] = $matched;
                    $itemMapByName[normalizeFastText($calcName)] = $matched;
                } else {
                    $prevStock = $simulatedStockMap[$matched['id']];
                    $newStock = $prevStock + $qty;
                    $simulatedStockMap[$matched['id']] = $newStock;
                }

                $validPreview[] = [
                    'row'           => $rowNum,
                    'import_date'   => $parsedDate,
                    'supplier_name' => $supplier,
                    'invoice_no'    => $invoice,
                    'item_code'     => $matched['item_code'],
                    'item_name'     => $matched['item_name'],
                    'specification' => $matched['specification'] ?: $spec,
                    'unit'          => $matched['unit'] ?: ($unit ?: 'Cái'),
                    'quantity'      => $qty,
                    'prev_stock'    => $prevStock,
                    'new_stock'     => $newStock,
                    'notes'         => $note,
                    'is_new'        => $isNewItem
                ];
            }

            if (!empty($validPreview) || !empty($failedPreview)) {
                $_SESSION['preview_import_stock_vpp'] = [
                    'valid'  => $validPreview,
                    'failed' => $failedPreview,
                    'source' => $fileName
                ];
            } else {
                $error = "Tệp Excel không có dòng dữ liệu hợp lệ nào!";
            }
        } elseif (empty($error)) {
            $error = "Tệp tải lên rỗng hoặc cấu trúc dữ liệu không hợp lệ!";
        }
    } else {
        $error = "Vui lòng chọn file Excel hợp lệ để tải lên!";
    }
}

// 2C. Xác nhận hoặc hủy nhập kho VPP
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'confirm_import_stock_excel') {
    if (isset($_SESSION['preview_import_stock_vpp']['valid']) && !empty($_SESSION['preview_import_stock_vpp']['valid'])) {
        $validItems = $_SESSION['preview_import_stock_vpp']['valid'];
        try {
            $pdo->beginTransaction();

            $stmtCheckCode = $pdo->prepare("SELECT id, stock_qty, total_imported FROM vpp_items WHERE item_code = :c LIMIT 1");
            $stmtCheckName = $pdo->prepare("SELECT id, stock_qty, total_imported FROM vpp_items WHERE item_name = :n LIMIT 1");

            $stmtInsertItem = $pdo->prepare("
                INSERT INTO vpp_items (item_code, item_name, specification, unit, initial_stock, total_imported, stock_qty)
                VALUES (:code, :name, :spec, :unit, 0, 0, 0)
            ");

            $stmtImport = $pdo->prepare("
                INSERT INTO vpp_imports (invoice_no, supplier_name, import_date, item_id, quantity, notes)
                VALUES (:inv, :sup, :idate, :it_id, :qty, :n)
            ");

            $stmtUpStock = $pdo->prepare("
                UPDATE vpp_items 
                SET stock_qty = stock_qty + :q1, total_imported = total_imported + :q2 
                WHERE id = :id
            ");

            $successCount = 0;
            $autoCreatedCount = 0;

            foreach ($validItems as $v) {
                $targetId = null;
                if (!empty($v['item_code'])) {
                    $stmtCheckCode->execute([':c' => $v['item_code']]);
                    $f = $stmtCheckCode->fetch();
                    if ($f) $targetId = $f['id'];
                }
                if (!$targetId && !empty($v['item_name'])) {
                    $stmtCheckName->execute([':n' => $v['item_name']]);
                    $f = $stmtCheckName->fetch();
                    if ($f) $targetId = $f['id'];
                }

                if (!$targetId) {
                    $stmtInsertItem->execute([
                        ':code' => $v['item_code'],
                        ':name' => $v['item_name'],
                        ':spec' => $v['specification'],
                        ':unit' => $v['unit']
                    ]);
                    $targetId = $pdo->lastInsertId();
                    $autoCreatedCount++;
                }

                $stmtImport->execute([
                    ':inv'   => $v['invoice_no'],
                    ':sup'   => $v['supplier_name'],
                    ':idate' => $v['import_date'],
                    ':it_id' => $targetId,
                    ':qty'   => $v['quantity'],
                    ':n'     => $v['notes']
                ]);

                $stmtUpStock->execute([
                    ':q1' => $v['quantity'],
                    ':q2' => $v['quantity'],
                    ':id' => $targetId
                ]);
                $successCount++;
            }

            $pdo->commit();
            unset($_SESSION['preview_import_stock_vpp']);
            $message = "🎉 <strong>Thành công:</strong> Đã nhập kho chính thức {$successCount} bản ghi VPP" . ($autoCreatedCount > 0 ? " (và tạo mới {$autoCreatedCount} mặt hàng vào danh mục)" : "") . "!";
        } catch (Exception $e) {
            if ($pdo->inTransaction()) $pdo->rollBack();
            $error = "Lỗi khi lưu dữ liệu nhập kho: " . $e->getMessage();
        }
    } else {
        unset($_SESSION['preview_import_stock_vpp']);
        $error = "Không tìm thấy dữ liệu phiên nhập kho hợp lệ để xác nhận!";
    }
}

// 2D. Cập nhật / Xóa phiếu nhập
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'update_import') {

    $new_inv    = trim($_POST['invoice_no'] ?? '');
    $new_sup    = trim($_POST['supplier_name'] ?? '');
    $new_item   = intval($_POST['item_id']);
    $new_qty    = intval($_POST['quantity']);
    $new_notes  = trim($_POST['notes'] ?? '');

            $stmtAdd = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty + :q1, total_imported = total_imported + :q2 WHERE id = :id");

            WHERE id = :id
        ");

            ':inv' => $new_inv, ':sup' => $new_sup, ':idate' => $new_date,

// ==================== 3. SỬA / XÓA LỊCH SỬ CẤP PHÁT ====================
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'update_request') {
    $req_id   = intval($_POST['request_id']);
    $new_date = $_POST['request_date'];
    $new_dept = trim($_POST['department']);
    $new_rec  = trim($_POST['receiver_name'] ?? '');
    $new_item = intval($_POST['item_id']);
    $new_qty  = intval($_POST['quantity']);
    $new_loc  = trim($_POST['delivery_location'] ?? '');
    $new_notes= trim($_POST['notes'] ?? '');

    try {
        $pdo->beginTransaction();
        $stmtOld = $pdo->prepare("SELECT * FROM vpp_requests WHERE id = :id FOR UPDATE");
        $stmtOld->execute([':id' => $req_id]);
        $oldReq = $stmtOld->fetch();

        if (!$oldReq) throw new Exception("Bản ghi cấp phát không tồn tại!");

        $stmtIt = $pdo->prepare("SELECT item_name, specification, stock_qty, unit FROM vpp_items WHERE id = :id FOR UPDATE");
        $stmtIt->execute([':id' => $new_item]);
        $it = $stmtIt->fetch();
        if (!$it) throw new Exception("Mặt hàng VPP không tồn tại!");

        if (intval($oldReq['item_id']) === $new_item) {
            $diff = $new_qty - intval($oldReq['quantity']);
            if ($diff > 0 && $diff > $it['stock_qty']) {
                throw new Exception("Không đủ hàng trong kho để tăng số lượng cấp! Còn thiếu {$diff} {$it['unit']}.");
            }
            $stmtUpStock = $pdo->prepare("UPDATE vpp_items SET stock_qty = GREATEST(0, stock_qty - :diff) WHERE id = :id");
            $stmtUpStock->execute([':diff' => $diff, ':id' => $new_item]);
        } else {
            $stmtBack = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty + :q WHERE id = :id");
            $stmtBack->execute([':q' => $oldReq['quantity'], ':id' => $oldReq['item_id']]);

            if ($new_qty > $it['stock_qty']) {
                throw new Exception("Không đủ tồn kho mặt hàng mới để cấp! Tồn: {$it['stock_qty']} {$it['unit']}.");
            }
            $stmtTake = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty - :q WHERE id = :id");
            $stmtTake->execute([':q' => $new_qty, ':id' => $new_item]);
        }

        $stmtUpReq = $pdo->prepare("
            UPDATE vpp_requests 
            SET request_date = :r_date, department = :dept, receiver_name = :rec, item_id = :item_id, 
                quantity = :qty, delivery_location = :loc, notes = :notes 
            WHERE id = :id
        ");
        $stmtUpReq->execute([
            ':r_date'  => $new_date, ':dept'    => $new_dept, ':rec'     => $new_rec,
            ':item_id' => $new_item, ':qty'     => $new_qty,
            ':loc'     => $new_loc,  ':notes'   => $new_notes, ':id'     => $req_id
        ]);

        $pdo->commit();
        $message = "Đã cập nhật bản ghi lịch sử cấp phát VPP thành công!";
    } catch (Exception $e) {
        if ($pdo->inTransaction()) $pdo->rollBack();
        $error = "Lỗi cập nhật: " . $e->getMessage();
    }
}

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'delete_request') {
    $req_id = intval($_POST['request_id']);
    try {
        $pdo->beginTransaction();
        $stmtGet = $pdo->prepare("SELECT * FROM vpp_requests WHERE id = :id FOR UPDATE");
        $stmtGet->execute([':id' => $req_id]);
        $req = $stmtGet->fetch();

        if ($req) {
            $stmtRet = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty + :qty WHERE id = :item_id");
            $stmtRet->execute([':qty' => $req['quantity'], ':item_id' => $req['item_id']]);

            $stmtDel = $pdo->prepare("DELETE FROM vpp_requests WHERE id = :id");

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {

            $ext      = strtolower(pathinfo($fileName, PATHINFO_EXTENSION));

            $rows = [];

                if (($handle = fopen($tmpName, 'r')) !== false) {
                    while (($data = fgetcsv($handle, 2000, ',')) !== false) {
                        $rows[] = array_map('trim', $data);
                    }
                    fclose($handle);
                }
            } elseif ($ext === 'xlsx') {
                $rows = parseXlsxFastStream($tmpName);
                if ($rows === null) {
                    $error = "Máy chủ PHP cần bật ZipArchive để đọc file .xlsx hoặc bạn có thể lưu file dạng .XLS / .CSV để tải lên!";
                }
            } elseif ($ext === 'xls') {
                $content = file_get_contents($tmpName);
                if (strpos($content, '<table') !== false) {
                    $dom = new DOMDocument();
                    @$dom->loadHTML(mb_convert_encoding($content, 'HTML-ENTITIES', 'UTF-8'));
                    $trs = $dom->getElementsByTagName('tr');
                    foreach ($trs as $tr) {
                        $rowValues = [];
                        $tds = $tr->getElementsByTagName('td');
                        if ($tds->length === 0) $tds = $tr->getElementsByTagName('th');
                        foreach ($tds as $td) {
                            $rowValues[] = trim($td->textContent);

            if (!empty($rows)) {
                $header = array_shift($rows);
                $idxCode = -1; $idxName = -1; $idxSpec = -1; $idxUnit = -1;

                foreach ($header as $colIdx => $colName) {
                    $cNorm = normalizeFastText($colName);
                    if ($idxCode === -1 && (strpos($cNorm, 'mavpp') !== false || strpos($cNorm, 'masp') !== false || strpos($cNorm, 'code') !== false || $cNorm === 'ma')) {
                        $idxCode = $colIdx;
                    } elseif ($idxName === -1 && (strpos($cNorm, 'ten') !== false || strpos($cNorm, 'mathang') !== false || strpos($cNorm, 'name') !== false)) {
                        $idxName = $colIdx;
                    } elseif ($idxSpec === -1 && (strpos($cNorm, 'quycach') !== false || strpos($cNorm, 'donggoi') !== false || strpos($cNorm, 'spec') !== false)) {
                        $idxSpec = $colIdx;
                    } elseif ($idxUnit === -1 && (strpos($cNorm, 'dvt') !== false || strpos($cNorm, 'donvi') !== false || strpos($cNorm, 'unit') !== false)) {

                if ($idxName === -1) $idxName = 1;
                if ($idxSpec === -1) $idxSpec = 2;
                if ($idxUnit === -1) $idxUnit = 3;

                $mapExisting = [];
                foreach ($existingItems as $ex) {

                    $spec = trim($row[$idxSpec] ?? '');
                    $unit = trim($row[$idxUnit] ?? 'Cái');

                            'name'   => $name,
                            'reason' => 'Thiếu Tên Mặt Hàng VPP (Cột bắt buộc)'
                        ];
                        continue;
                    }

                    $codeKey = normalizeFastText($code);
                    $nameKey = normalizeFastText($name);

                    $match = null;
                    if (!empty($code) && isset($mapExisting['c_' . $codeKey])) {
                        $match = $mapExisting['c_' . $codeKey];
                    } elseif (isset($mapExisting['n_' . $nameKey])) {
                        $match = $mapExisting['n_' . $nameKey];
                    }

                    if ($match) {
                        $validItemsPreview[] = [
                            'row'          => $rowNum,
                            'action_type'  => 'update',
                            'item_id'      => $match['id'],
                            'item_code'    => $code ?: $match['item_code'],
                            'item_name'    => $match['item_name'],
                            'old_spec'     => $match['specification'],
                            'new_spec'     => $spec,
                            'old_unit'     => $match['unit'],
                            'new_unit'     => $unit ?: 'Cái',
                            'current_stock'=> $match['stock_qty']

                            'row'          => $rowNum,
                            'action_type'  => 'insert',
                            'item_id'      => null,
                            'item_code'    => $code,
                            'item_name'    => $name,
                            'old_spec'     => '—',
                            'new_spec'     => $spec,
                            'old_unit'     => '—',
                            'new_unit'     => $unit ?: 'Cái',
                            'current_stock'=> 0
                        ];

                            'id' => 'dummy', 'item_code' => $code, 'item_name' => $name, 'specification' => $spec, 'unit' => $unit ?: 'Cái', 'stock_qty' => 0

                        'valid'  => $validItemsPreview,
                        'failed' => $failedItemsPreview,
                        'source' => $fileName
                    ];
                } else {
                    $error = "File tải lên không có dữ liệu phù hợp!";
                }
            } else {
                $error = "File tải lên không có dữ liệu hoặc không đúng định dạng!";
            }
        } else {
            $error = "Vui lòng chọn file Excel hợp lệ để tải lên!";
        }
    }

    // 4C. Xác nhận hoặc hủy nhập danh mục VPP
    if ($act === 'confirm_import_item_excel') {
        if (isset($_SESSION['preview_import_item_vpp']['valid']) && !empty($_SESSION['preview_import_item_vpp']['valid'])) {
            $previewList = $_SESSION['preview_import_item_vpp']['valid'];
            try {
                $pdo->beginTransaction();

                $stmtInsert = $pdo->prepare("
                    INSERT INTO vpp_items (item_code, item_name, specification, unit, initial_stock, total_imported, stock_qty)
                    VALUES (:code, :name, :spec, :unit, 0, 0, 0)
                ");

                $stmtUpdate = $pdo->prepare("
                    UPDATE vpp_items 
                    SET item_code = :code, specification = :spec, unit = :unit 
                    WHERE id = :id
                ");

                $stmtFindByName = $pdo->prepare("SELECT id FROM vpp_items WHERE item_name = :n LIMIT 1");
                $stmtFindByCode = $pdo->prepare("SELECT id FROM vpp_items WHERE item_code = :c LIMIT 1");

                $insCount = 0; $upCount = 0;

                            ':unit' => $pItem['new_unit'],

                        ]);

                    } else {

                            $stmtFindByCode->execute([':c' => $pItem['item_code']]);
                            $existingId = $stmtFindByCode->fetchColumn();

                        if (!$existingId) {
                            $stmtFindByName->execute([':n' => $pItem['item_name']]);

                                ':name' => $pItem['item_name'],
                                ':spec' => $pItem['new_spec'],

        $pdo->exec("TRUNCATE TABLE vpp_requests");
        $pdo->exec("TRUNCATE TABLE vpp_proposals");
        $pdo->exec("TRUNCATE TABLE vpp_imports");
        $message = "Đã reset toàn bộ dữ liệu văn phòng phẩm về 0!";
    }

<script>
function openModal(id) { document.getElementById(id).style.display = 'flex'; }
function closeModal(id) { document.getElementById(id).style.display = 'none'; }
function toggleMoreMenu(e) { if(e) e.stopPropagation(); closeAllMenus(); document.getElementById('more_dropdown_menu').classList.toggle('show'); }

function toggleImportMenu(e) {
    if (e) e.stopPropagation();
    const btn = e.currentTarget;
    const parent = btn.closest('.import-btn-group');
    const menu = parent ? parent.querySelector('.import-menu') : null;
    const isShown = menu && menu.classList.contains('show');
    closeAllMenus();
    if (menu && !isShown) {
        menu.classList.add('show');
    }
}

function toggleAddItemMenu(e) {
    if (e) e.stopPropagation();
    const btn = e.currentTarget;
    const parent = btn.closest('.add-item-btn-group');
    const menu = parent ? parent.querySelector('.add-item-menu') : null;
    const isShown = menu && menu.classList.contains('show');
    closeAllMenus();
    if (menu && !isShown) {
        menu.classList.add('show');
    }
}

function closeAllMenus() {
    document.querySelectorAll('.import-menu').forEach(el => el.classList.remove('show'));
    document.querySelectorAll('.add-item-menu').forEach(el => el.classList.remove('show'));
    const moreMenu = document.getElementById('more_dropdown_menu');
    if (moreMenu) moreMenu.classList.remove('show');
}

// Xử lý mở/đóng Dropdown Chờ Duyệt (Màu cam)
function togglePropDropdown(e, code) {
    if (e) e.stopPropagation();
    const currentMenu = document.getElementById('prop_dropdown_' + code);
    if (!currentMenu) return;
    const isShown = currentMenu.classList.contains('show');
    closeAllPropDropdowns();
    closeAllPencilDropdowns();
    if (!isShown) {
        currentMenu.classList.add('show');
    }
}

function closeAllPropDropdowns() {
    document.querySelectorAll('.prop-dropdown-menu').forEach(el => el.classList.remove('show'));
}

// Xử lý mở/đóng Dropdown Cây Viết (Sửa / Xóa)
function togglePropPencilDropdown(e, code) {
    if (e) e.stopPropagation();
    const currentMenu = document.getElementById('prop_pencil_menu_' + code);
    if (!currentMenu) return;
    const isShown = currentMenu.classList.contains('show');
    closeAllPropDropdowns();
    closeAllPencilDropdowns();
    if (!isShown) {
        currentMenu.classList.add('show');
    }
}

function closeAllPencilDropdowns() {
    document.querySelectorAll('.prop-pencil-menu').forEach(el => el.classList.remove('show'));
}

// Popover hiển thị ghi chú của S.L Duyệt
function toggleNotePopover(e, btn) {
    if (e) e.stopPropagation();
    const container = btn.closest('.note-interactive-container');
    if (!container) return;
    const pop = container.querySelector('.note-popover');
    if (!pop) return;
    const isShown = pop.classList.contains('show');
    closeAllNotePopovers();
    if (!isShown) {
        pop.classList.add('show');
    }
}

function closeAllNotePopovers() {
    document.querySelectorAll('.note-popover').forEach(el => el.classList.remove('show'));
}

window.onclick = function(e) {
    if (e.target.classList.contains('modal-overlay')) e.target.style.display = 'none';
    if (!e.target.closest('.import-btn-group')) {
        document.querySelectorAll('.import-menu').forEach(el => el.classList.remove('show'));
    }
    if (!e.target.closest('.add-item-btn-group')) {
        document.querySelectorAll('.add-item-menu').forEach(el => el.classList.remove('show'));
    }
    if (!e.target.closest('.more-dropdown')) {
        const moreMenu = document.getElementById('more_dropdown_menu');
        if (moreMenu) moreMenu.classList.remove('show');
    }
    if (!e.target.closest('.prop-dropdown')) closeAllPropDropdowns();
    if (!e.target.closest('.prop-pencil-dropdown')) closeAllPencilDropdowns();
    if (!e.target.closest('.note-interactive-container')) closeAllNotePopovers();
}

function togglePropInputs(mode) {
    document.getElementById('prop_input_day').style.display   = (mode === 'day') ? 'inline-block' : 'none';
    document.getElementById('prop_input_week').style.display  = (mode === 'week') ? 'inline-block' : 'none';
    document.getElementById('prop_input_month').style.display = (mode === 'month') ? 'inline-block' : 'none';
    document.getElementById('prop_input_year').style.display  = (mode === 'year') ? 'inline-block' : 'none';
}

function toggleImpInputs(mode) {

// Kiểm tra bắt buộc có ghi chú nếu số lượng duyệt khác số lượng yêu cầu ban đầu
function validateApprovalNote() {
    let hasChangedQty = false;
    document.querySelectorAll('.appr-qty-input').forEach(inp => {
        const orig = parseInt(inp.getAttribute('data-original-qty') || 0);
        const curr = parseInt(inp.value || 0);
        if (orig !== curr) {
            hasChangedQty = true;
        }
    });

    const note = document.getElementById('apg_approval_note').value.trim();
    if (hasChangedQty && note === '') {
        alert('Cảnh báo: Bạn đã thay đổi số lượng duyệt so với yêu cầu ban đầu. Vui lòng nhập lý do/ghi chú duyệt!');
        document.getElementById('apg_approval_note').focus();
        return false;
    }
    return true;
}
