            INDEX (`created_at`)

    $return_bottles = intval($_POST['return_bottle_count'] ?? 0);
    $delivery_loc   = trim($_POST['delivery_location'] ?? '');
    $employee_id    = !empty($_POST['employee_id']) ? intval($_POST['employee_id']) : null;

    if (empty($receiver_name) || empty($department) || $item_id <= 0 || $quantity <= 0) {
        $error = "Vui lòng điền đầy đủ họ tên, phòng ban và chọn loại nước cần đề xuất!";

        try {
            // Mã yêu cầu gửi tới hệ thống: ghép từ ngày tháng năm giờ phút giây (YmdHis)
            $requestCode = date('YmdHis');

            $cols = $pdo->query("SHOW COLUMNS FROM water_proposals")->fetchAll(PDO::FETCH_COLUMN);
            $codeCol = in_array('request_code', $cols) ? 'request_code' : (in_array('code', $cols) ? 'code' : null);

            if ($codeCol) {
                $stmt = $pdo->prepare("
                    INSERT INTO water_proposals ({$codeCol}, employee_id, receiver_name, department, item_id, quantity, return_bottle_count, delivery_location, status)
                    VALUES (:req_code, :emp_id, :rec_name, :dept, :item_id, :qty, :ret_b, :loc, 'Chờ cấp phát')
                ");
                $stmt->execute([
                    ':req_code' => $requestCode,
                    ':emp_id'   => $employee_id,
                    ':rec_name' => $receiver_name,
                    ':dept'     => $department,
                    ':item_id'  => $item_id,
                    ':qty'      => $quantity,
                    ':ret_b'    => max(0, $return_bottles),
                    ':loc'      => $delivery_loc
                ]);
            } else {
                $stmt = $pdo->prepare("
                    INSERT INTO water_proposals (employee_id, receiver_name, department, item_id, quantity, return_bottle_count, delivery_location, status)
                    VALUES (:emp_id, :rec_name, :dept, :item_id, :qty, :ret_b, :loc, 'Chờ cấp phát')

                    ':emp_id'   => $employee_id,
                    ':rec_name' => $receiver_name,
                    ':dept'     => $department,
                    ':item_id'  => $item_id,

                    ':ret_b'    => max(0, $return_bottles),
                    ':loc'      => $delivery_loc

            $error = "Lỗi gửi yêu cầu: " . $e->getMessage();

// 2. Gợi ý nhân viên
const employees   = <?= json_encode($employees) ?>;
const inpReceiver = document.getElementById('inp_receiver');
const suggestBox  = document.getElementById('suggest_box');
const selDept     = document.getElementById('sel_dept');
const hiddenEmpId = document.getElementById('hidden_emp_id');

inpReceiver.addEventListener('input', function() {
    const val = this.value.trim().toLowerCase();
    suggestBox.innerHTML = '';
    if (!val) {
        suggestBox.style.display = 'none';
        return;
    }

    const matched = employees.filter(e => 
        (e.fullname && e.fullname.toLowerCase().includes(val)) || 
        (e.id && e.id.toString() === val) || 
        ('id:' + e.id).includes(val)
    );

    if (matched.length > 0) {
        matched.slice(0, 8).forEach(emp => {
            const div = document.createElement('div');
            div.className = 'suggest-item';
            div.innerHTML = `<strong>#${emp.id} - ${emp.fullname}</strong> (${emp.department || 'Chưa phân bổ'})`;
            div.onclick = function() {
                inpReceiver.value = emp.fullname;
                hiddenEmpId.value = emp.id;
                if (emp.department) {
                    selDept.value = emp.department;
                }
                suggestBox.style.display = 'none';
            };
            suggestBox.appendChild(div);
        });
        suggestBox.style.display = 'block';
    } else {
        suggestBox.style.display = 'none';
        hiddenEmpId.value = '';
    }
});

document.addEventListener('click', function(e) {
    if (!e.target.closest('#inp_receiver') && !e.target.closest('#suggest_box')) {
        suggestBox.style.display = 'none';
    }
});

// 3. Tự động đồng bộ số vỏ theo số lượng yêu cầu
const selWaterItem     = document.getElementById('sel_water_item');
const boxBottle        = document.getElementById('box_bottle_return');
const inpQuantity      = document.getElementById('inp_quantity');
const inpReturnBottles = document.getElementById('inp_return_bottles');

function isReturnBottleRequired() {
    if (!selWaterItem.value) return false;
    const selectedOpt = selWaterItem.options[selWaterItem.selectedIndex];
    const spec = (selectedOpt.getAttribute('data-spec') || '').toLowerCase();
    const name = (selectedOpt.text || '').toLowerCase();
    return (spec.includes('19') || spec.includes('20') || name.includes('19l') || name.includes('19 lít') || name.includes('bình'));
}

function checkWaterSpec(selectElem) {
    if (isReturnBottleRequired()) {
        boxBottle.style.display = 'block';
        syncReturnBottles();
    } else {
        boxBottle.style.display = 'none';
        inpReturnBottles.value = 0;
    }
}

function syncReturnBottles() {
    if (isReturnBottleRequired()) {
        const qty = parseInt(inpQuantity.value) || 0;
        inpReturnBottles.value = qty;
    }
}

// 4. Bật tắt Tooltip Ghi chú duyệt
function toggleTooltip(event, btn) {
    event.stopPropagation();
    const tooltip = btn.nextElementSibling;
    const isCurrentlyVisible = tooltip.style.display === 'block';

    // Đóng tất cả các tooltip khác trước
    document.querySelectorAll('.note-tooltip').forEach(tt => tt.style.display = 'none');
    document.querySelectorAll('.btn-note-popup').forEach(b => b.classList.remove('active'));

    if (!isCurrentlyVisible) {
        tooltip.style.display = 'block';
        btn.classList.add('active');
    }
}

// Bấm ra ngoài thì đóng tooltip
document.addEventListener('click', function(e) {
    if (!e.target.closest('.sl-duyet-container')) {
        document.querySelectorAll('.note-tooltip').forEach(tt => tt.style.display = 'none');
        document.querySelectorAll('.btn-note-popup').forEach(b => b.classList.remove('active'));
    }
});

        e.preventDefault();
        if (formProposal.checkValidity()) {
            confirmModal.style.display = 'flex';
        } else {
            formProposal.reportValidity();

});

btnConfirmSubmit.addEventListener('click', function() {
    isConfirmed = true;
    confirmModal.style.display = 'none';
    formProposal.submit();
});
