<?php
// Tọa độ mặc định: Ninh Hiệp, Ninh Hòa
$default_lat = 12.6075;
$default_lon = 109.1235;

$result = null;

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $lat = filter_input(INPUT_POST, 'lat', FILTER_VALIDATE_FLOAT) ?: $default_lat;
    $lon = filter_input(INPUT_POST, 'lon', FILTER_VALIDATE_FLOAT) ?: $default_lon;
    $target_datetime = $_POST['target_time'] ?? date('Y-m-d\TH:00');

    // Chuyển mốc giờ về định dạng chuỗi tròn giờ: YYYY-MM-DDTHH:00
    $target_timestamp = strtotime($target_datetime);
    $formatted_time = date('Y-m-d\TH:00', $target_timestamp);

    // 1. Gọi API lấy dữ liệu mô hình ECMWF (Châu Âu)
    $url_ecmwf = "https://api.open-meteo.com/v1/ecmwf?latitude={$lat}&longitude={$lon}&hourly=temperature_2m,precipitation,surface_pressure,wind_speed_10m,wind_direction_10m&timezone=Asia%2FBangkok";
    
    // 2. Gọi API lấy dữ liệu mô hình ICON (Đức - DWD)
    $url_icon = "https://api.open-meteo.com/v1/dwd-icon?latitude={$lat}&longitude={$lon}&hourly=temperature_2m,precipitation,surface_pressure,wind_speed_10m,wind_direction_10m&timezone=Asia%2FBangkok";

    $data_ecmwf = json_decode(@file_get_contents($url_ecmwf), true);
    $data_icon  = json_decode(@file_get_contents($url_icon), true);

    if ($data_ecmwf && $data_icon) {
        $idx_ecmwf = array_search($formatted_time, $data_ecmwf['hourly']['time'] ?? []);
        $idx_icon  = array_search($formatted_time, $data_icon['hourly']['time'] ?? []);

        if ($idx_ecmwf !== false && $idx_icon !== false) {
            // Trích xuất số liệu
            $rain_ecmwf = $data_ecmwf['hourly']['precipitation'][$idx_ecmwf];
            $temp_ecmwf = $data_ecmwf['hourly']['temperature_2m'][$idx_ecmwf];
            $wind_deg_ecmwf = $data_ecmwf['hourly']['wind_direction_10m'][$idx_ecmwf];

            $rain_icon = $data_icon['hourly']['precipitation'][$idx_icon];
            $temp_icon = $data_icon['hourly']['temperature_2m'][$idx_icon];
            $wind_speed_icon = $data_icon['hourly']['wind_speed_10m'][$idx_icon];
            $wind_deg_icon = $data_icon['hourly']['wind_direction_10m'][$idx_icon];
            $pressure_icon = $data_icon['hourly']['surface_pressure'][$idx_icon];

            // 3. Áp dụng công thức chuyên gia để phân tích
            $probability = 0;
            $status = "Tạnh ráo";
            $notes = [];

            // Kiểm tra đồng thuận mưa
            if ($rain_ecmwf >= 1.0 && $rain_icon >= 1.0) {
                $probability = 85;
                $status = "Khả năng mưa rất cao";
                $notes[] = "Cả ECMWF ({$rain_ecmwf} mm) và ICON ({$rain_icon} mm) đều đồng thuận báo mưa.";
            } elseif ($rain_ecmwf >= 1.5 || $rain_icon >= 1.5) {
                $probability = 60;
                $status = "Có thể có mưa";
                $notes[] = "Chỉ có 1 mô hình báo mưa lớn (ECMWF: {$rain_ecmwf}mm | ICON: {$rain_icon}mm).";
            } else {
                $probability = 15;
                $status = "Tạnh ráo / Không mưa";
                $notes[] = "Cả 2 mô hình đều không ghi nhận lượng mưa đáng kể.";
            }

            // Kiểm tra hướng gió (Ninh Hòa: gió Đông/Đông Nam từ 60° - 150° là gió mang ẩm từ biển)
            $is_sea_breeze = ($wind_deg_icon >= 60 && $wind_deg_icon <= 150);
            if ($is_sea_breeze) {
                $probability += 10;
                $notes[] = "Gió Đông/Đông Nam từ biển đang thốc ẩm vào đất liền ({$wind_speed_icon} km/h), tăng nguy cơ mưa.";
            } else {
                $notes[] = "Gió lục địa/núi, không có nguồn ẩm biển mạnh.";
            }

            // Kiểm tra dông nhiệt buổi chiều (13h - 17h)
            $hour = (int)date('H', $target_timestamp);
            if ($hour >= 13 && $hour <= 17 && $temp_icon >= 32 && $is_sea_breeze) {
                $status .= " (Cảnh báo: Dông sét nhiệt cục bộ)";
                $notes[] = "Nhiệt độ buổi trưa cao ({$temp_icon}°C) kết hợp gió biển tạo đối lưu mạnh, dễ có dông sét.";
            }

            $probability = min(98, max(5, $probability)); // Giới hạn từ 5% đến 98%

            $result = [
                'time' => $formatted_time,
                'status' => $status,
                'probability' => $probability,
                'rain_ecmwf' => $rain_ecmwf,
                'rain_icon' => $rain_icon,
                'temp' => $temp_icon,
                'wind_speed' => $wind_speed_icon,
                'pressure' => $pressure_icon,
                'notes' => $notes
            ];
        } else {
            $error = "Mốc thời gian chọn nằm ngoài phạm vi dữ liệu dự báo (tối đa 7 ngày).";
        }
    } else {
        $error = "Không thể kết nối máy chủ dữ liệu khí tượng.";
    }
}

<script>
function updateCoords(select) {
    var coords = select.value.split(',');
    document.getElementById('lat').value = coords[0];
    document.getElementById('lon').value = coords[1];
}
