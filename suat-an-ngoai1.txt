$message = '';
$error = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $action = $_POST['action'] ?? '';

    // 1. Nhập danh sách từ Excel sau khi duyệt Preview
    if ($action === 'confirm_import_excel') {
        $importDate = $_POST['import_date'] ?? $today;
        $jsonData = $_POST['excel_rows_json'] ?? '';
        $rows = json_decode($jsonData, true);

        if (!empty($rows) && is_array($rows)) {
            try {
                // Tối ưu hóa: Lấy trước toàn bộ nhân viên đã có trong ngày để tránh query lồng trong vòng lặp
                $stmtExisting = $pdo->prepare("SELECT id, employee_code FROM meal_attendance WHERE record_date = :d");
                $stmtExisting->execute([':d' => $importDate]);
                $existingRows = $stmtExisting->fetchAll(PDO::FETCH_ASSOC);
                $stmtExisting->closeCursor();

                $existingMap = [];
                foreach ($existingRows as $ex) {
                    $existingMap[trim((string)$ex['employee_code'])] = $ex['id'];
                }

                $pdo->beginTransaction();

                $stmtInsert = $pdo->prepare("
                    INSERT INTO meal_attendance (record_date, employee_code, full_name, department, check_time, is_canceled, cancel_reason)
                    VALUES (:d, :c, :n, :dept, :t, 0, '')
                ");
                $stmtUpdate = $pdo->prepare("
                    UPDATE meal_attendance 
                    SET full_name = :n, department = :dept, check_time = :t 
                    WHERE id = :id
                ");

                $countInserted = 0;
                $countUpdated = 0;

                foreach ($rows as $r) {
                    $code = trim((string)($r['code'] ?? ''));
                    $name = trim((string)($r['name'] ?? ''));
                    $dept = trim((string)($r['dept'] ?? 'Khác')) ?: 'Khác';
                    $time = trim((string)($r['time'] ?? ''));

                    if (empty($code) || empty($name)) continue;

                    if (isset($existingMap[$code])) {
                        $targetId = $existingMap[$code];
                        $stmtUpdate->execute([
                            ':n' => $name,
                            ':dept' => $dept,
                            ':t' => $time,
                            ':id' => $targetId
                        ]);
                        $stmtUpdate->closeCursor();
                        $countUpdated++;
                    } else {
                        $stmtInsert->execute([
                            ':d' => $importDate,
                            ':c' => $code,
                            ':n' => $name,
                            ':dept' => $dept,
                            ':t' => $time
                        ]);
                        $newId = $pdo->lastInsertId();
                        $stmtInsert->closeCursor();
                        $existingMap[$code] = $newId;
                        $countInserted++;
                    }
                }

                $pdo->commit();
                $message = "Đã nhập thành công dữ liệu ngày [{$importDate}]: Thêm mới {$countInserted} người, cập nhật {$countUpdated} người có đi làm!";
            } catch (Exception $e) {
                if ($pdo->inTransaction()) {
                    $pdo->rollBack();
                }
                $error = "Lỗi khi lưu dữ liệu: " . $e->getMessage();
            }
        } else {
            $error = "Không có dữ liệu hợp lệ để nhập!";
        }
    }

    // 2. Cập nhật thủ công 1 nhân viên
    if ($action === 'update_row') {
        $id = intval($_POST['id'] ?? 0);
        $name = trim($_POST['full_name'] ?? '');
        $code = trim($_POST['employee_code'] ?? '');
        $dept = trim($_POST['department'] ?? 'Khác');
        $time = trim($_POST['check_time'] ?? '');
        $is_cancel = isset($_POST['is_canceled']) ? 1 : 0;
        $reason = trim($_POST['cancel_reason'] ?? '');

        if ($id > 0 && !empty($name)) {
            $stmt = $pdo->prepare("
                UPDATE meal_attendance 
                SET employee_code = :c, full_name = :n, department = :dept, check_time = :t, is_canceled = :ic, cancel_reason = :cr
                WHERE id = :id
            ");
            $stmt->execute([':c' => $code, ':n' => $name, ':dept' => $dept, ':t' => $time, ':ic' => $is_cancel, ':cr' => $reason, ':id' => $id]);
            $stmt->closeCursor();
            $message = "Đã cập nhật thông tin nhân viên [{$name}] thành công!";
        }
    }

    // 3. Xóa nhân viên khỏi danh sách
    if ($action === 'delete_row') {
        $id = intval($_POST['id'] ?? 0);
        if ($id > 0) {
            $stmt = $pdo->prepare("DELETE FROM meal_attendance WHERE id = :id");
            $stmt->execute([':id' => $id]);
            $stmt->closeCursor();
            $message = "Đã xóa bản ghi chấm công suất ăn!";
        }
    }
}

// Xử lý đọc file Excel và mở Modal Xem trước
function handleExcelUpload(input) {
    const file = input.files[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = function(e) {
        try {
            const data = new Uint8Array(e.target.result);
            const workbook = XLSX.read(data, { type: 'array' });
            const firstSheetName = workbook.SheetNames[0];
            const worksheet = workbook.Sheets[firstSheetName];
            const jsonSheet = XLSX.utils.sheet_to_json(worksheet, { header: 1, defval: '' });

            if (!jsonSheet || jsonSheet.length < 2) {
                alert('File Excel không có dữ liệu hợp lệ!');
                return;
            }

            // Tìm hàng Header và xác định vị trí các cột
            let headerRowIndex = 0;
            let colCode = -1, colName = -1, colDept = -1, colDate = -1, colTime = -1;

            for (let r = 0; r < Math.min(15, jsonSheet.length); r++) {
                const row = jsonSheet[r];
                for (let c = 0; c < row.length; c++) {
                    const val = String(row[c]).toLowerCase().trim();
                    if (val.includes('mã nv') || val.includes('ac-no') || val.includes('user id') || val.includes('id')) {
                        if (colCode === -1) colCode = c;
                    }
                    if (val.includes('tên') || val.includes('name') || val.includes('họ tên')) {
                        if (colName === -1) colName = c;
                    }
                    if (val.includes('phòng') || val.includes('bộ phận') || val.includes('dept') || val.includes('department')) {
                        if (colDept === -1) colDept = c;
                    }
                    if (val.includes('ngày') || val.includes('date')) {
                        if (colDate === -1) colDate = c;
                    }
                    if (val.includes('giờ') || val.includes('time') || val.includes('check')) {
                        if (colTime === -1) colTime = c;
                    }
                }
                if (colName !== -1 && (colCode !== -1 || colTime !== -1)) {
                    headerRowIndex = r;
                    break;
                }
            }

            // Fallback nếu tiêu đề cột bị ẩn
            if (colCode === -1) colCode = 2;
            if (colName === -1) colName = 3;
            if (colDept === -1) colDept = 4;
            if (colDate === -1) colDate = 5;
            if (colTime === -1) colTime = 6;

            const validRows = [];
            let detectedDate = '<?= $f_date ?>';

            for (let r = headerRowIndex + 1; r < jsonSheet.length; r++) {
                const row = jsonSheet[r];
                if (!row || row.length === 0) continue;

                const code = String(row[colCode] || '').trim();
                const name = String(row[colName] || '').trim();
                const dept = String(row[colDept] || 'Khác').trim() || 'Khác';
                const dateVal = String(row[colDate] || '').trim();
                const timeVal = String(row[colTime] || '').trim();

                // ĐIỀU KIỆN QUAN TRỌNG: Cột Time có giờ thì mới tính người đó đi làm
                if (timeVal !== '' && name !== '') {
                    validRows.push({
                        code: code,
                        name: name,
                        dept: dept,
                        time: timeVal
                    });

                    // Tự động nhận diện ngày từ cột Date nếu đúng định dạng
                    if (dateVal && dateVal.includes('-') && dateVal.length >= 8) {
                        detectedDate = dateVal;
                    }
                }
            }

            if (validRows.length === 0) {
                alert('Không tìm thấy nhân viên nào có giờ chấm công (Cột Time có dữ liệu)!');
                input.value = '';
                return;
            }

            // Điền dữ liệu vào Modal Xem trước
            document.getElementById('preview_import_date_val').value = detectedDate;
            document.getElementById('preview_date_text').innerText = detectedDate;
            document.getElementById('preview_count_text').innerText = validRows.length;
            document.getElementById('preview_json_val').value = JSON.stringify(validRows);

            const tbody = document.getElementById('preview_tbody');
            tbody.innerHTML = '';
            validRows.forEach((item, index) => {
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td style="text-align:center; font-weight:bold; color:#64748b;">${index + 1}</td>
                    <td><b>${item.code}</b></td>
                    <td><strong>${item.name}</strong></td>
                    <td>${item.dept}</td>
                    <td style="text-align:center; color:#0284c7; font-weight:600;">${item.time}</td>
                `;
                tbody.appendChild(tr);
            });

            openModal('modal-preview-excel');
            input.value = '';
        } catch (err) {
            alert('Lỗi khi đọc file Excel: ' + err.message);
            input.value = '';
        }
    };
    reader.readAsArrayBuffer(file);
}

// Tick cắt cơm qua AJAX
function toggleCancelMeal(id, checkbox) {
    const isChecked = checkbox.checked ? 1 : 0;
    let reason = '';
    
    if (isChecked) {
        reason = prompt('Nhập lý do cắt cơm (nếu có, có thể để trống):', '') || '';
    }

    const formData = new FormData();
    formData.append('ajax_action', 'toggle_cancel');
    formData.append('id', id);
    formData.append('status', isChecked);
    formData.append('reason', reason);

    fetch('suat-an-ngoai.php', {
        method: 'POST',
        body: formData
    })
    .then(res => res.json())
    .then(data => {
        if (data.success) {
            const row = document.getElementById('row_' + id);
            const statusBadge = document.getElementById('status_badge_' + id);
            const reasonText = document.getElementById('reason_text_' + id);

            if (isChecked) {
                row.classList.add('row-canceled');
                statusBadge.innerHTML = '<span class="badge-status badge-cancel">Cắt Cơm</span>';
                reasonText.innerText = reason;
            } else {
                row.classList.remove('row-canceled');
                statusBadge.innerHTML = '<span class="badge-status badge-eat">Có Ăn Cơm</span>';
                reasonText.innerText = '';
            }
            // Tải lại trang sau 300ms để khối 2 tự động cập nhật số liệu mới nhất
            setTimeout(() => { location.reload(); }, 350);
        } else {
            alert('Có lỗi xảy ra: ' + data.message);
            checkbox.checked = !isChecked;
        }
    })
    .catch(err => {
        alert('Lỗi kết nối máy chủ!');
        checkbox.checked = !isChecked;
    });
}

// Ẩn / Hiện cột cắt cơm ở Khối 2
let isCancelColVisible = false;
function toggleCancelColumn() {
    isCancelColVisible = !isCancelColVisible;
    const cols = document.querySelectorAll('.col-cancel-stat');
    cols.forEach(col => {
        col.style.display = isCancelColVisible ? '' : 'none';
    });
    document.getElementById('btn_toggle_cancel_text').innerText = isCancelColVisible ? 'Ẩn Cột Cắt Cơm' : 'Hiện Cột Cắt Cơm';
}

}
