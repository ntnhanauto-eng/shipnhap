// ==================== HÀM HỖ TRỢ PHÂN TRANG ====================
function renderPaginationBar($currentPage, $totalPages, $paramName) {
    if ($totalPages <= 1) return;
    $queryParams = $_GET;
    echo '<div class="pagination-wrapper">';
    if ($currentPage > 1) {
        $queryParams[$paramName] = $currentPage - 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&laquo;</a>';
    } else {
        echo '<span class="page-link disabled">&laquo;</span>';
    }
    $start = max(1, $currentPage - 1);
    $end = min($totalPages, $currentPage + 1);
    for ($i = $start; $i <= $end; $i++) {
        $queryParams[$paramName] = $i;
        $activeClass = ($i == $currentPage) ? 'active' : '';
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link ' . $activeClass . '">' . $i . '</a>';
    }
    if ($currentPage < $totalPages) {
        $queryParams[$paramName] = $currentPage + 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&raquo;</a>';
    } else {
        echo '<span class="page-link disabled">&raquo;</span>';
    }
    echo '</div>';
}

// ==================== 1. XỬ LÝ HÀNH ĐỘNG ĐỀ XUẤT MUA ====================
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    $act = $_POST['action'];

    if ($act === 'approve_and_push_next_period') {
        $req_code = trim($_POST['request_code'] ?? '');
        $appr_qtys = $_POST['approved_quantities'] ?? [];
        $appr_notes = $_POST['item_notes'] ?? [];
        try {
            if (empty($req_code)) throw new Exception("Mã yêu cầu không hợp lệ!");
            $pdo->beginTransaction();

            $stmtProps = $pdo->prepare("
                SELECT p.*, 
                COALESCE(i.item_name, p.custom_item_name) AS final_item_name,
                COALESCE(i.specification, p.specification) AS final_spec,
                COALESCE(i.unit, p.unit) AS final_unit
                FROM vpp_purchase_proposals p
                LEFT JOIN vpp_items i ON p.item_id = i.id
                WHERE p.request_code = :code
            ");
            $stmtProps->execute([':code' => $req_code]);
            $rows = $stmtProps->fetchAll();

            foreach ($rows as $r) {
                $propId = $r['id'];
                $apprQty = isset($appr_qtys[$propId]) ? max(0, intval($appr_qtys[$propId])) : max(0, intval($r['quantity']));
                $note = trim($appr_notes[$propId] ?? '');

                $stmtUp = $pdo->prepare("UPDATE vpp_purchase_proposals SET status = 'Đã duyệt', approved_quantity = :aq, approval_note = :an WHERE id = :id");
                $stmtUp->execute([':aq' => $apprQty, ':an' => $note, ':id' => $propId]);

                if ($apprQty > 0) {
                    $itemKey = md5(mb_strtolower(trim($r['final_item_name'] . '|' . $r['final_spec'] . '|' . $r['final_unit'])));
                    
                    $stmtCheck = $pdo->prepare("SELECT * FROM vpp_purchase_next_period WHERE item_key = :k");
                    $stmtCheck->execute([':k' => $itemKey]);
                    $existingNext = $stmtCheck->fetch();

                    $sourceInfo = "{$req_code} ({$r['department']} - {$r['receiver_name']}: {$apprQty} {$r['final_unit']})";

                    if ($existingNext) {
                        $newTotal = intval($existingNext['total_approved_qty']) + $apprQty;
                        $sources = trim($existingNext['source_requests'] . ' | ' . $sourceInfo);
                        $stmtUpNext = $pdo->prepare("UPDATE vpp_purchase_next_period SET total_approved_qty = :tot, source_requests = :src WHERE item_key = :k");
                        $stmtUpNext->execute([':tot' => $newTotal, ':src' => $sources, ':k' => $itemKey]);
                    } else {
                        $stmtInsNext = $pdo->prepare("INSERT INTO vpp_purchase_next_period (item_key, item_id, item_name, specification, unit, total_approved_qty, source_requests, status) VALUES (:k, :it_id, :name, :spec, :unit, :tot, :src, 'Chờ mua')");
                        $stmtInsNext->execute([
                            ':k' => $itemKey,
                            ':it_id' => $r['item_id'],
                            ':name' => $r['final_item_name'],
                            ':spec' => $r['final_spec'],
                            ':unit' => $r['final_unit'],
                            ':tot' => $apprQty,
                            ':src' => $sourceInfo
                        ]);
                    }
                }
            }

            $pdo->commit();
            $message = "Đã phê duyệt và đẩy thành công các mặt hàng của yêu cầu [{$req_code}] vào danh sách cần mua cho kỳ tới!";
        } catch (Exception $e) {
            if ($pdo->inTransaction()) $pdo->rollBack();
            $error = "Lỗi phê duyệt: " . $e->getMessage();
        }
    }

    if ($act === 'edit_purchase_proposal') {
        $req_code = trim($_POST['request_code'] ?? '');
        $receiver_name = trim($_POST['receiver_name'] ?? '');
        $department = trim($_POST['department'] ?? '');
        $edit_items = $_POST['items_qty'] ?? [];
        try {
            if (empty($req_code)) throw new Exception("Mã yêu cầu không hợp lệ!");
            $pdo->beginTransaction();
            $stmtUpInfo = $pdo->prepare("UPDATE vpp_purchase_proposals SET receiver_name = :rec, department = :dept WHERE request_code = :code");
            $stmtUpInfo->execute([':rec' => $receiver_name, ':dept' => $department, ':code' => $req_code]);

            $stmtUpQty = $pdo->prepare("UPDATE vpp_purchase_proposals SET quantity = :qty WHERE id = :id AND request_code = :code");
            foreach ($edit_items as $propId => $qty) {
                $q = max(1, intval($qty));
                $stmtUpQty->execute([':qty' => $q, ':id' => intval($propId), ':code' => $req_code]);
            }
            $pdo->commit();
            $message = "Đã cập nhật thông tin đợt đề xuất mua [{$req_code}] thành công!";
        } catch (Exception $e) {
            if ($pdo->inTransaction()) $pdo->rollBack();
            $error = "Lỗi sửa đề xuất: " . $e->getMessage();
        }
    }

    if ($act === 'edit_next_item') {
        $next_id = intval($_POST['next_id'] ?? 0);
        $new_qty = intval($_POST['total_approved_qty'] ?? 0);
        try {
            $stmt = $pdo->prepare("UPDATE vpp_purchase_next_period SET total_approved_qty = :qty WHERE id = :id");
            $stmt->execute([':qty' => max(1, $new_qty), ':id' => $next_id]);
            $message = "Đã cập nhật số lượng mặt hàng kỳ tới thành công!";
        } catch (Exception $e) { $error = "Lỗi cập nhật: " . $e->getMessage(); }
    }

    if ($act === 'reject_purchase_group') {
        $req_code = trim($_POST['request_code'] ?? '');
        $reason = trim($_POST['reject_reason'] ?? 'Không duyệt mua');
        try {
            $stmt = $pdo->prepare("UPDATE vpp_purchase_proposals SET status = 'Từ chối', reject_reason = :r WHERE request_code = :code");
            $stmt->execute([':r' => $reason, ':code' => $req_code]);
            $message = "Đã từ chối toàn bộ yêu cầu đề xuất mua [{$req_code}]!";
        } catch (PDOException $e) { $error = "Lỗi: " . $e->getMessage(); }
    }

    if ($act === 'delete_purchase_proposal') {
        $req_code = trim($_POST['request_code'] ?? '');
        try {
            $stmt = $pdo->prepare("DELETE FROM vpp_purchase_proposals WHERE request_code = :code");
            $stmt->execute([':code' => $req_code]);
            $message = "Đã xóa đợt đề xuất mua [{$req_code}] thành công!";
        } catch (PDOException $e) { $error = "Lỗi xóa: " . $e->getMessage(); }
    }

    if ($act === 'mark_purchased') {
        $next_id = intval($_POST['next_id'] ?? 0);
        try {
            $stmt = $pdo->prepare("UPDATE vpp_purchase_next_period SET status = 'Đã mua' WHERE id = :id");
            $stmt->execute([':id' => $next_id]);
            $message = "Đã cập nhật trạng thái mặt hàng thành 'Đã mua'!";
        } catch (PDOException $e) { $error = "Lỗi: " . $e->getMessage(); }
    }

    if ($act === 'delete_next_item') {
        $next_id = intval($_POST['next_id'] ?? 0);
        try {
            $stmt = $pdo->prepare("DELETE FROM vpp_purchase_next_period WHERE id = :id");
            $stmt->execute([':id' => $next_id]);
            $message = "Đã xóa mặt hàng khỏi danh sách kỳ tới!";
        } catch (PDOException $e) { $error = "Lỗi: " . $e->getMessage(); }
    }
}

// ==================== 2. TRUY VẤN DỮ LIỆU ĐỀ XUẤT MUA (CÓ BỘ LỌC THỜI GIAN TỰ CHỌN) ====================
$f_prop_search = trim($_GET['f_prop_search'] ?? '');
$f_prop_dept = trim($_GET['f_prop_dept'] ?? '');
$f_prop_status = trim($_GET['f_prop_status'] ?? '');
$f_prop_mode = trim($_GET['f_prop_mode'] ?? 'month');
$f_prop_date = trim($_GET['f_prop_date'] ?? $today);
$f_prop_week = trim($_GET['f_prop_week'] ?? date('Y-\WW'));
$f_prop_month = trim($_GET['f_prop_month'] ?? date('Y-m'));
$f_prop_year = trim($_GET['f_prop_year'] ?? date('Y'));
$f_prop_from = trim($_GET['f_prop_from'] ?? date('Y-m-01'));
$f_prop_to = trim($_GET['f_prop_to'] ?? date('Y-m-d'));

$whereP = ["1=1"];
$paramsP = [];
if (!empty($f_prop_search)) {
    $whereP[] = "(p.request_code LIKE :ps1 OR p.receiver_name LIKE :ps2 OR p.custom_item_name LIKE :ps3)";
    $paramsP[':ps1'] = "%$f_prop_search%";
    $paramsP[':ps2'] = "%$f_prop_search%";
    $paramsP[':ps3'] = "%$f_prop_search%";
}
if (!empty($f_prop_dept)) {
    $whereP[] = "p.department = :pdept";
    $paramsP[':pdept'] = $f_prop_dept;
}
if (!empty($f_prop_status)) {
    $whereP[] = "p.status = :pstat";
    $paramsP[':pstat'] = $f_prop_status;
}

if ($f_prop_mode === 'day') {
    $whereP[] = "DATE(p.created_at) = :pd";
    $paramsP[':pd'] = $f_prop_date;
} elseif ($f_prop_mode === 'week') {
    $yw = substr($f_prop_week, 0, 4); $ww = substr($f_prop_week, 6);
    $dto = new DateTime(); $dto->setISODate(intval($yw), intval($ww));
    $ws = $dto->format('Y-m-d 00:00:00'); $dto->modify('+6 days'); $we = $dto->format('Y-m-d 23:59:59');
    $whereP[] = "p.created_at BETWEEN :pws AND :pwe";
    $paramsP[':pws'] = $ws; $paramsP[':pwe'] = $we;
} elseif ($f_prop_mode === 'year') {
    $whereP[] = "DATE_FORMAT(p.created_at, '%Y') = :py";
    $paramsP[':py'] = $f_prop_year;
} elseif ($f_prop_mode === 'custom') {
    $whereP[] = "DATE(p.created_at) BETWEEN :pfrom AND :pto";
    $paramsP[':pfrom'] = !empty($f_prop_from) ? $f_prop_from : date('Y-m-01');
    $paramsP[':pto'] = !empty($f_prop_to) ? $f_prop_to : date('Y-m-d');
} else {
    $whereP[] = "DATE_FORMAT(p.created_at, '%Y-%m') = :pm";
    $paramsP[':pm'] = $f_prop_month;
}

$countPropStmt = $pdo->prepare("SELECT COUNT(DISTINCT p.request_code) FROM vpp_purchase_proposals p WHERE " . implode(' AND ', $whereP));
$countPropStmt->execute($paramsP);
$total_proposals = intval($countPropStmt->fetchColumn() ?? 0);
$pages_prop = ceil($total_proposals / $limit) ?: 1;
if ($p_prop > $pages_prop) $p_prop = $pages_prop;
$offset_prop = max(0, ($p_prop - 1) * $limit);

$sqlProposals = "
    SELECT 
        p.request_code, p.employee_id, p.receiver_name, p.department, p.status, p.reject_reason, p.created_at,
        COUNT(p.id) AS item_count,
        SUM(p.quantity) AS total_qty_requested,
        SUM(COALESCE(p.approved_quantity, p.quantity)) AS total_qty_approved,
        JSON_ARRAYAGG(
            JSON_OBJECT(
                'proposal_id', p.id,
                'item_name', COALESCE(i.item_name, p.custom_item_name),
                'specification', COALESCE(i.specification, p.specification),
                'unit', COALESCE(i.unit, p.unit),
                'quantity', p.quantity,
                'approved_quantity', p.approved_quantity,
                'approval_note', p.approval_note
            )
        ) AS items_json
    FROM vpp_purchase_proposals p
    LEFT JOIN vpp_items i ON p.item_id = i.id
    WHERE " . implode(' AND ', $whereP) . "
    GROUP BY p.request_code, p.employee_id, p.receiver_name, p.department, p.status, p.reject_reason, p.created_at
    ORDER BY (p.status = 'Chờ duyệt') DESC, MAX(p.id) DESC
    LIMIT " . intval($limit) . " OFFSET " . intval($offset_prop) . "
";
$stmtPropList = $pdo->prepare($sqlProposals);
$stmtPropList->execute($paramsP);
$proposals = $stmtPropList->fetchAll();

// Tổng theo bộ lọc hiện tại (Đề xuất)
$sumPropFilter = $pdo->prepare("
    SELECT 
        COALESCE(SUM(p.quantity), 0) AS total_req_filter,
        COALESCE(SUM(CASE WHEN p.status = 'Đã duyệt' THEN COALESCE(p.approved_quantity, p.quantity) ELSE 0 END), 0) AS total_appr_filter
    FROM vpp_purchase_proposals p 
    WHERE " . implode(' AND ', $whereP)
);
$sumPropFilter->execute($paramsP);
$totalsPropFilter = $sumPropFilter->fetch();

$f_next_year = trim($_GET['f_next_year'] ?? date('Y'));
$f_next_from = trim($_GET['f_next_from'] ?? date('Y-m-01'));
$f_next_to = trim($_GET['f_next_to'] ?? date('Y-m-d'));

$whereNext = ["1=1"];
$paramsNext = [];
if (!empty($f_next_search)) {
    $whereNext[] = "(item_name LIKE :ns1 OR source_requests LIKE :ns2)";
    $paramsNext[':ns1'] = "%$f_next_search%";
    $paramsNext[':ns2'] = "%$f_next_search%";
}
if ($f_next_mode === 'day') {
    $whereNext[] = "DATE(updated_at) = :nd";
    $paramsNext[':nd'] = $f_next_date;
} elseif ($f_next_mode === 'week') {
    $yw = substr($f_next_week, 0, 4); $ww = substr($f_next_week, 6);
    $dto = new DateTime(); $dto->setISODate(intval($yw), intval($ww));
    $ws = $dto->format('Y-m-d 00:00:00'); $dto->modify('+6 days'); $we = $dto->format('Y-m-d 23:59:59');
    $whereNext[] = "updated_at BETWEEN :nws AND :nwe";
    $paramsNext[':nws'] = $ws; $paramsNext[':nwe'] = $we;
} elseif ($f_next_mode === 'year') {
    $whereNext[] = "DATE_FORMAT(updated_at, '%Y') = :ny";
    $paramsNext[':ny'] = $f_next_year;
} elseif ($f_next_mode === 'custom') {
    $whereNext[] = "DATE(updated_at) BETWEEN :nfrom AND :nto";
    $paramsNext[':nfrom'] = !empty($f_next_from) ? $f_next_from : date('Y-m-01');
    $paramsNext[':nto'] = !empty($f_next_to) ? $f_next_to : date('Y-m-d');
} else {
    $whereNext[] = "DATE_FORMAT(updated_at, '%Y-%m') = :nm";
    $paramsNext[':nm'] = $f_next_month;
}

$countNextStmt = $pdo->prepare("SELECT COUNT(*) FROM vpp_purchase_next_period WHERE " . implode(' AND ', $whereNext));
$countNextStmt->execute($paramsNext);
$total_next = intval($countNextStmt->fetchColumn() ?? 0);
$pages_next = ceil($total_next / $limit) ?: 1;
if ($p_next > $pages_next) $p_next = $pages_next;
$offset_next = max(0, ($p_next - 1) * $limit);

$sqlNext = "SELECT * FROM vpp_purchase_next_period WHERE " . implode(' AND ', $whereNext) . " ORDER BY (status = 'Chờ mua') DESC, id DESC LIMIT " . intval($limit) . " OFFSET " . intval($offset_next);
$stmtNextList = $pdo->prepare($sqlNext);
$stmtNextList->execute($paramsNext);
$nextPeriodItems = $stmtNextList->fetchAll();

function closeModal(id) { document.getElementById(id).style.display = 'none'; }

    e.stopPropagation();
    closeAllPencilMenus();
    document.getElementById('pencil_menu_' + id).classList.toggle('show');
}
function closeAllPencilMenus() {
    document.querySelectorAll('.pencil-menu').forEach(m => m.classList.remove('show'));
}
window.onclick = function(e) {
    if (e.target.classList.contains('modal-overlay')) e.target.style.display = 'none';
    if (!e.target.closest('.pencil-dropdown')) closeAllPencilMenus();
}

function togglePropFilterInputs(mode) {
    document.getElementById('prop_input_day').style.display = (mode === 'day') ? 'inline-block' : 'none';
    document.getElementById('prop_input_week').style.display = (mode === 'week') ? 'inline-block' : 'none';
    document.getElementById('prop_input_month').style.display = (mode === 'month') ? 'inline-block' : 'none';
    document.getElementById('prop_input_year').style.display = (mode === 'year') ? 'inline-block' : 'none';
    document.getElementById('prop_input_custom').style.display = (mode === 'custom') ? 'inline-flex' : 'none';
}

}

function openDetailsModal(p) {
    document.getElementById('dt_code').innerText = p.request_code;
    document.getElementById('dt_receiver').innerText = p.receiver_name + (p.employee_id ? ' (#' + p.employee_id + ')' : '');
    document.getElementById('dt_dept').innerText = p.department;
    const tbody = document.getElementById('dt_tbody');
    tbody.innerHTML = '';
    const items = JSON.parse(p.items_json || '[]');
    items.forEach((it, idx) => {
        const appr = it.approved_quantity !== null ? it.approved_quantity : '--';
        tbody.innerHTML += `<tr><td>${idx + 1}</td><td><b>${it.item_name}</b> ${it.specification ? '('+it.specification+')' : ''}</td><td>${it.unit}</td><td style="text-align:center; font-weight:bold; color:#0284c7;">${it.quantity}</td><td style="text-align:center; font-weight:bold; color:#15803d;">${appr}</td><td>${it.approval_note || '—'}</td></tr>`;
    });
    openModal('modal-details');
}

    document.getElementById('appr_code').value = p.request_code;
    document.getElementById('appr_code_text').innerText = p.request_code;

}
