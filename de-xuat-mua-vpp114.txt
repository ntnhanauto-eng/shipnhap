// ==================== HÀM NÉN ẢNH DƯỚI 100KB ====================

function compressAndUploadImage($file, $targetDir = 'uploads/vpp/') {

if (!isset($file['tmp_name']) || empty($file['tmp_name'])) return '';

if (!is_dir($targetDir)) {

mkdir($targetDir, 0777, true);

}



$fileInfo = @getimagesize($file['tmp_name']);

if ($fileInfo === false) return '';



$mime = $fileInfo['mime'];

$srcImg = null;

if ($mime === 'image/jpeg') {

$srcImg = imagecreatefromjpeg($file['tmp_name']);

} elseif ($mime === 'image/png') {

$srcImg = imagecreatefrompng($file['tmp_name']);

} elseif ($mime === 'image/webp') {

$srcImg = imagecreatefromwebp($file['tmp_name']);

}

if (!$srcImg) return '';



$ext = pathinfo($file['name'], PATHINFO_EXTENSION);

$newFileName = 'vpp_mua_' . uniqid() . '.' . ($ext ?: 'jpg');

$destination = $targetDir . $newFileName;



$quality = 85;

if ($mime === 'image/png') {

imagepng($srcImg, $destination, 8);

} else {

imagejpeg($srcImg, $destination, $quality);

}



while (filesize($destination) > 100 * 1024 && $quality > 20) {

$quality -= 15;

if ($mime === 'image/png') {

imagepng($srcImg, $destination, 9);

} else {

imagejpeg($srcImg, $destination, $quality);

}

}



imagedestroy($srcImg);

return $destination;

}

try {

$pdo->exec("

CREATE TABLE IF NOT EXISTS `vpp_purchase_proposals` (

`id` INT AUTO_INCREMENT PRIMARY KEY,

`request_code` VARCHAR(60) DEFAULT '',

`employee_id` INT DEFAULT NULL,

`receiver_name` VARCHAR(150) NOT NULL,

`department` VARCHAR(100) NOT NULL,

`item_id` INT DEFAULT NULL,

`custom_item_name` VARCHAR(150) DEFAULT '',

`specification` VARCHAR(100) DEFAULT '',

`unit` VARCHAR(50) DEFAULT 'Cái',

`custom_image` VARCHAR(255) DEFAULT '',

`quantity` INT NOT NULL DEFAULT 1,

`pre_approved_quantity` INT DEFAULT NULL,

`approved_quantity` INT DEFAULT NULL,

`approval_note` VARCHAR(255) DEFAULT '',

`status` VARCHAR(50) DEFAULT 'Chờ duyệt',

`reject_reason` VARCHAR(255) DEFAULT '',

`created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

INDEX (`item_id`),

INDEX (`status`),

INDEX (`request_code`)

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

");



$colsProp = $pdo->query("SHOW COLUMNS FROM `vpp_purchase_proposals`")->fetchAll(PDO::FETCH_COLUMN);

if (!in_array('custom_image', $colsProp)) {

$pdo->exec("ALTER TABLE `vpp_purchase_proposals` ADD COLUMN `custom_image` VARCHAR(255) DEFAULT '' AFTER `unit`");

}

} catch (PDOException $e) {}



$allItems = $pdo->query("SELECT * FROM vpp_items ORDER BY item_name ASC")->fetchAll();



// Truy vấn danh sách nhân viên để hỗ trợ gợi ý ô nhập họ tên / mã số

$employees = [];

try {

$employees = $pdo->query("SELECT id, fullname, department FROM employees WHERE status = 'Đang làm việc' ORDER BY fullname ASC")->fetchAll();

} catch (PDOException $e) {}



if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'submit_purchase_proposal') {

$employee_id = !empty($_POST['employee_id']) ? intval($_POST['employee_id']) : null;

$receiver_name = trim($_POST['receiver_name'] ?? '');

$department = trim($_POST['department'] ?? '');

$items_data = $_POST['items'] ?? [];



try {

if (empty($receiver_name) || empty($department)) {

throw new Exception("Vui lòng điền đầy đủ Họ tên/Mã số người đề xuất và Bộ phận / Phòng ban!");

}

if (empty($items_data)) {

throw new Exception("Vui lòng chọn hoặc điền ít nhất một mặt hàng cần đề xuất mua!");

}



$baseCode = 'MUA-' . date('dmYHis');

$pdo->beginTransaction();



$stmtIns = $pdo->prepare("

INSERT INTO vpp_purchase_proposals

(request_code, employee_id, receiver_name, department, item_id, custom_item_name, specification, unit, custom_image, quantity, status)

VALUES (:code, :empid, :rec, :dept, :it_id, :c_name, :spec, :unit, :img, :qty, 'Chờ duyệt')

");



$insertedCount = 0;

foreach ($items_data as $idx => $row) {

$itemId = intval($row['item_id'] ?? 0);

$qty = intval($row['quantity'] ?? 0);

if ($qty <= 0) continue;



$customName = trim($row['custom_item_name'] ?? '');

$spec = trim($row['specification'] ?? '');

$unit = trim($row['unit'] ?? 'Cái');

$imgPath = '';



// Xử lý nén ảnh nếu là mặt hàng mới ngoài kho và có upload file

if ($itemId === 0 && isset($_FILES['items']['name'][$idx]['image']) && $_FILES['items']['error'][$idx]['image'] === UPLOAD_ERR_OK) {

$fileData = [

'name' => $_FILES['items']['name'][$idx]['image'],

'tmp_name' => $_FILES['items']['tmp_name'][$idx]['image'],

'error' => $_FILES['items']['error'][$idx]['image']

];

$imgPath = compressAndUploadImage($fileData);

}



$stmtIns->execute([

':code' => $baseCode,

':empid' => $employee_id,

':rec' => $receiver_name,

':dept' => $department,

':it_id' => ($itemId > 0 ? $itemId : null),

':c_name' => ($itemId > 0 ? '' : $customName),

':spec' => $spec,

':unit' => $unit,

':img' => $imgPath,

':qty' => $qty

]);

$insertedCount++;

}



if ($insertedCount === 0) {

throw new Exception("Số lượng đề xuất của các mặt hàng không hợp lệ!");

}



$pdo->commit();

$message = "Gửi đề xuất mua văn phòng phẩm thành công! Mã phiếu của bạn là: <strong>{$baseCode}</strong>";

} catch (Exception $e) {

if ($pdo->inTransaction()) $pdo->rollBack();

$error = "Lỗi gửi đề xuất: " . $e->getMessage();

}

}

// CHUYỂN CHẾ ĐỘ ĐỀ XUẤT

function changeProposalMode(mode) {

const availableSection = document.getElementById('available_section');

const customSection = document.getElementById('custom_section');

const description = document.getElementById('mode_description');



if (mode === 'available') {

availableSection.style.display = 'block';

customSection.style.display = 'none';



description.innerHTML =

'Tick vào các mặt hàng có sẵn trong kho và nhập số lượng yêu cầu ở cột cuối.';



/*
 * Khi chuyển sang sản phẩm có sẵn,
 * tắt toàn bộ field custom để không gửi nhầm dữ liệu.
 */

disableCustomFields();

updateSelectedCounter();



} else if (mode === 'custom') {

availableSection.style.display = 'none';

customSection.style.display = 'block';



description.innerHTML =

'Nhập tên, quy cách, đơn vị, hình ảnh và số lượng của mặt hàng cần mua bên ngoài.';



/*
 * Khi chuyển sang custom,
 * tắt toàn bộ field của danh sách sản phẩm có sẵn.
 */

disableAvailableFields();



} else {

availableSection.style.display = 'none';

customSection.style.display = 'none';

description.innerHTML =

'Chọn một hình thức để bắt đầu nhập danh sách mặt hàng cần mua.';

}

}



// =========================================================
// TẮT FIELD CUSTOM
// =========================================================

function disableCustomFields() {

const customSection = document.getElementById('custom_section');



if (!customSection) return;



customSection.querySelectorAll('input, select, textarea').forEach(function(input) {

input.disabled = true;

});

}



// =========================================================
// TẮT FIELD SẢN PHẨM CÓ SẴN
// =========================================================

function disableAvailableFields() {

const availableSection = document.getElementById('available_section');



if (!availableSection) return;



availableSection.querySelectorAll('input, select, textarea').forEach(function(input) {

input.disabled = true;

});

}



// =========================================================
// CHECK / UNCHECK SẢN PHẨM
// =========================================================

function toggleProductRow(checkbox) {

const row = checkbox.closest('tr');



if (!row) return;



const inputs = row.querySelectorAll('input');



inputs.forEach(function(input) {

if (input === checkbox) return;



/*
 * Checkbox được tick:
 * bật item_id, specification, unit và quantity.
 */

input.disabled = !checkbox.checked;

});



if (checkbox.checked) {

const qty = row.querySelector('.qty-input');



if (qty) {

qty.focus();

qty.select();

}

}



updateSelectedCounter();

}



// =========================================================
// ĐẾM SẢN PHẨM ĐÃ CHỌN
// =========================================================

function updateSelectedCounter() {

const counter = document.getElementById('selected_counter');



if (!counter) return;



const selected = document.querySelectorAll(

'#available_tbody .product-check:checked'

);



counter.textContent = 'Đã chọn: ' + selected.length;

}



// =========================================================
// CHỌN TẤT CẢ SẢN PHẨM ĐANG HIỂN THỊ
// =========================================================

function selectAllVisible() {

const rows = document.querySelectorAll(

'#available_tbody .product-row:not(.hidden-row)'

);



rows.forEach(function(row) {

const checkbox = row.querySelector('.product-check');



if (checkbox && !checkbox.checked) {

checkbox.checked = true;

toggleProductRow(checkbox);

}

});



updateSelectedCounter();

}



// =========================================================
// BỎ CHỌN TẤT CẢ
// =========================================================

function clearAllProducts() {

const checkboxes = document.querySelectorAll(

'#available_tbody .product-check'

);



checkboxes.forEach(function(checkbox) {

if (checkbox.checked) {

checkbox.checked = false;

toggleProductRow(checkbox);

}

});



updateSelectedCounter();

}



// =========================================================
// TÌM KIẾM SẢN PHẨM
// =========================================================

function filterProducts(input) {

const keyword = input.value.trim().toLowerCase();



const rows = document.querySelectorAll(

'#available_tbody .product-row'

);



rows.forEach(function(row) {

const name = row.getAttribute('data-name') || '';

const spec = row.getAttribute('data-spec') || '';



if (!keyword ||

name.includes(keyword) ||

spec.includes(keyword)) {

row.classList.remove('hidden-row');

} else {

row.classList.add('hidden-row');

}

});

}



// =========================================================
// UPLOAD ẢNH
// =========================================================

function handleImageChange(input, idx) {

if (input.files && input.files[0]) {

const reader = new FileReader();



reader.onload = function(e) {

const thumb = document.getElementById('thumb_' + idx);

const btnWrap = document.getElementById('upload_btn_wrap_' + idx);

const previewWrap = document.getElementById('preview_wrap_' + idx);



if (thumb) {

thumb.src = e.target.result;

}



if (btnWrap) {

btnWrap.style.display = 'none';

}



if (previewWrap) {

previewWrap.style.display = 'flex';

}

};



reader.readAsDataURL(input.files[0]);

}

}



// =========================================================
// XÓA ẢNH
// =========================================================

function removeImage(idx) {

const fileInput = document.getElementById('file_' + idx);

const btnWrap = document.getElementById('upload_btn_wrap_' + idx);

const previewWrap = document.getElementById('preview_wrap_' + idx);



if (fileInput) {

fileInput.value = '';

}



if (btnWrap) {

btnWrap.style.display = 'block';

}



if (previewWrap) {

previewWrap.style.display = 'none';

}

}



// =========================================================
// RESET MẶT HÀNG CUSTOM
// =========================================================

function resetCustomItem() {

const customName = document.getElementById('custom_item_name');



if (customName) {

customName.value = '';

}



const customSection = document.getElementById('custom_section');



if (customSection) {

const inputs = customSection.querySelectorAll('input, select, textarea');



inputs.forEach(function(input) {

if (input.type === 'file') {

input.value = '';

} else if (input.type === 'number') {

input.value = '1';

} else if (input.name.includes('[unit]')) {

input.value = 'Cái';

} else {

input.value = '';

}

});

}



removeImage(0);

}



// =========================================================
// VALIDATE TRƯỚC KHI SUBMIT
// =========================================================

function validateBeforeSubmit() {

const mode = document.getElementById('proposal_mode').value;



if (!mode) {

alert('Vui lòng chọn loại đề xuất mua hàng!');

return false;

}



if (mode === 'available') {

const selected = document.querySelectorAll(

'#available_tbody .product-check:checked'

);



if (selected.length === 0) {

alert('Vui lòng tick chọn ít nhất một mặt hàng cần đề xuất!');

return false;

}



let invalidQuantity = false;



selected.forEach(function(checkbox) {

const row = checkbox.closest('tr');

const qty = row.querySelector('.qty-input');



if (!qty || parseInt(qty.value, 10) <= 0) {

invalidQuantity = true;

}

});



if (invalidQuantity) {

alert('Vui lòng nhập số lượng lớn hơn 0 cho tất cả mặt hàng đã chọn!');

return false;

}

}



if (mode === 'custom') {

const customName = document.getElementById('custom_item_name');



if (!customName || !customName.value.trim()) {

alert('Vui lòng nhập tên mặt hàng mới!');

return false;

}

}



return true;

}
