function normalizeFastText($str) {
    $str = str_replace("\xC2\xA0", ' ', (string)$str);
    $str = mb_strtolower(trim($str), 'UTF-8');
    $str = preg_replace('/[áàảãạăắằẳẵặâấầẩẫậ]/u', 'a', $str);
    $str = preg_replace('/[éèẻẽẹêếềểễệ]/u', 'e', $str);
    $str = preg_replace('/[íìỉĩị]/u', 'i', $str);
    $str = preg_replace('/[óòỏõọôốồổỗộơớờởỡợ]/u', 'o', $str);
    $str = preg_replace('/[úùủũụưứừửữự]/u', 'u', $str);
    $str = preg_replace('/[ýỳỷỹỵ]/u', 'y', $str);
    $str = preg_replace('/đ/u', 'd', $str);
    return preg_replace('/[^a-z0-9]/', '', $str);
}
function cleanDishNameAdvanced($rawName) {
    $name = preg_replace('/^([0-9]+[\.\)]\s*|[\-\*\•]\s*)/u', '', (string)$rawName);
    $name = preg_replace('/\s*[\(\[].*?[\)\]]/u', '', $name);
    return trim($name);
}
function parseDishAndDescriptionFromText($line) {
    $clean = preg_replace('/^([0-9]+[\.\)]\s*|[\-\*\•]\s*)/u', '', (string)$line);
    $clean = trim($clean);
    
    $dishName = $clean;
    $description = '';
    if (preg_match('/^(.*?)\s*[\(\[]([^\)\]]+)[\)\]](.*)$/u', $clean, $matches)) {
        $dishName = trim($matches[1] . ' ' . $matches[3]);
        $description = trim($matches[2]);
    }
    elseif (preg_match('/^(.*?)\s*[:–—]\s*(.+)$/u', $clean, $matches)) {
        $dishName = trim($matches[1]);
        $description = trim($matches[2]);
    }
    return [
        'dish' => trim($dishName),
        'desc' => trim($description)
    ];
}
function normalizeDayOfWeekInput($rawDay) {
    $norm = normalizeFastText($rawDay);
    if (strpos($norm, 'thu2') !== false || $norm === '2' || strpos($norm, 'hai') !== false) return 'Thứ 2';
    if (strpos($norm, 'thu3') !== false || $norm === '3' || strpos($norm, 'ba') !== false)  return 'Thứ 3';
    if (strpos($norm, 'thu4') !== false || $norm === '4' || strpos($norm, 'tu') !== false)  return 'Thứ 4';
    if (strpos($norm, 'thu5') !== false || $norm === '5' || strpos($norm, 'nam') !== false) return 'Thứ 5';
    if (strpos($norm, 'thu6') !== false || $norm === '6' || strpos($norm, 'sau') !== false) return 'Thứ 6';
    if (strpos($norm, 'thu7') !== false || $norm === '7' || strpos($norm, 'bay') !== false) return 'Thứ 7';
    if (strpos($norm, 'chunhat') !== false || $norm === 'cn' || strpos($norm, 'sunday') !== false) return 'Chủ Nhật';
    return 'Thứ 2';
}
function parseXlsxFastStream($filepath) {
    if (!class_exists('ZipArchive')) return null;
    $zip = new ZipArchive();
    if ($zip->open($filepath) !== true) return null;
    $strings = [];
    $sharedStrXml = $zip->getFromName('xl/sharedStrings.xml');
    if ($sharedStrXml) {
        $reader = new XMLReader();
        $reader->XML($sharedStrXml);
        $currText = '';
        $inT = false;
        while ($reader->read()) {
            if ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 'si') {
                $currText = '';
            } elseif ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 't') {
                $inT = true;
            } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 't') {
                $inT = false;
            } elseif ($inT && ($reader->nodeType === XMLReader::TEXT || $reader->nodeType === XMLReader::CDATA)) {
                $currText .= $reader->value;
            } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 'si') {
                $strings[] = $currText;
            }
        }
        $reader->close();
    }
    $sheetXml = $zip->getFromName('xl/worksheets/sheet1.xml');
    if (!$sheetXml) {
        $zip->close();
        return null;
    }
    $rows = [];
    $reader = new XMLReader();
    $reader->XML($sheetXml);
    $currentRow = [];
    $currentVal = '';
    $cellType = '';
    $inVal = false;
    while ($reader->read()) {
        if ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 'row') {
            $currentRow = [];
        } elseif ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 'c') {
            $cellType = $reader->getAttribute('t') ?? '';
            $currentVal = '';
        } elseif ($reader->nodeType === XMLReader::ELEMENT && $reader->name === 'v') {
            $inVal = true;
        } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 'v') {
            $inVal = false;
        } elseif ($inVal && ($reader->nodeType === XMLReader::TEXT || $reader->nodeType === XMLReader::CDATA)) {
            $currentVal .= $reader->value;
        } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 'c') {
            $val = trim($currentVal);
            if ($cellType === 's' && isset($strings[intval($val)])) {
                $val = trim($strings[intval($val)]);
            }
            $currentRow[] = $val;
        } elseif ($reader->nodeType === XMLReader::END_ELEMENT && $reader->name === 'row') {
            if (!empty(array_filter($currentRow, fn($v) => $v !== ''))) {
                $rows[] = $currentRow;
            }
        }
    }
    $reader->close();
    $zip->close();
    return $rows;
}

$smartInsights = [];
try {
    // Truy xuất toàn diện thông tin kiểm tra và phản hồi gộp theo từng món ăn
    $stmtAI = $pdo->prepare("
        SELECT 
            m.dish_name,
            COUNT(DISTINCT c.id) as total_checks,
            SUM(CASE WHEN c.is_correct = 0 THEN 1 ELSE 0 END) as fails,
            IFNULL(AVG(c.adjustment_percent), 0) as avg_adj,
            COUNT(DISTINCT f.id) as total_fb,
            SUM(CASE WHEN f.quality_rating = 'Kém' THEN 1 ELSE 0 END) as bad_fb,
            SUM(CASE WHEN f.quality_rating = 'Trung bình' THEN 1 ELSE 0 END) as neutral_fb,
            SUM(CASE WHEN f.quality_rating = 'Tốt' THEN 1 ELSE 0 END) as good_fb,
            GROUP_CONCAT(LOWER(f.quality_note) SEPARATOR ' | ') as all_notes
        FROM meal_menus m
        LEFT JOIN meal_checks c ON m.id = c.menu_id AND c.check_date >= DATE_SUB(:td, $insightIntervalSQL)
        LEFT JOIN meal_quality_feedback f ON m.id = f.menu_id AND f.check_date >= DATE_SUB(:td, $insightIntervalSQL)
        GROUP BY m.dish_name
        HAVING (total_checks > 0 OR total_fb > 0)
    ");
    $stmtAI->execute([':td' => $today]);
    $aiData = $stmtAI->fetchAll();

    // Từ khóa phân tích cảm xúc & rủi ro thực phẩm (NLP Heuristics)
    $dangerKeywords = ['sống', 'thiu', 'ôi', 'mùi lạ', 'chua', 'đau bụng', 'vật thể lạ', 'ruồi', 'sán', 'tóc', 'chưa chín', 'hôi', 'nhớt', 'ngộ độc', 'kinh khủng'];
    $qualityKeywords = ['mặn', 'nhạt', 'dai', 'cứng', 'bở', 'nguội', 'dở', 'chán', 'ít', 'mỡ', 'cay', 'ngọt quá'];

    $criticalIssues = [];
    $qualityIssues = [];
    $totalGood = 0; $totalBad = 0; $totalFeedbacksGlobal = 0;

    foreach ($aiData as $row) {
        $totalGood += $row['good_fb'];
        $totalBad += $row['bad_fb'];
        $totalFeedbacksGlobal += $row['total_fb'];

        $riskScore = 0;
        $foundHazards = [];
        $foundQualities = [];
        $rootCause = [];

        // Chấm điểm Rủi ro Vận hành (Đổi món nhiều = rủi ro cao)
        $riskScore += ($row['fails'] * 10) + ($row['avg_adj'] * 0.5);
        
        // Chấm điểm Rủi ro Chất lượng (Feedback)
        $riskScore += ($row['bad_fb'] * 15) + ($row['neutral_fb'] * 5) - ($row['good_fb'] * 2);

        // Phân tích văn bản ghi chú (Text Mining)
        if (!empty($row['all_notes'])) {
            foreach ($dangerKeywords as $dk) {
                if (strpos($row['all_notes'], $dk) !== false) {
                    $foundHazards[] = $dk;
                    $riskScore += 50; // Trọng số cực cao cho lỗi an toàn thực phẩm
                }
            }
            foreach ($qualityKeywords as $qk) {
                if (strpos($row['all_notes'], $qk) !== false) {
                    $foundQualities[] = $qk;
                    $riskScore += 10;
                }
            }
        }

        // Suy luận Nguyên nhân cốt lõi (Root Cause Deduction)
        if ($row['fails'] > 0 && $row['bad_fb'] > 0) {
            $rootCause[] = "Nguồn cung nguyên liệu không ổn định hoặc nhà bếp tự ý thay đổi sai quy trình.";
        } elseif ($row['fails'] == 0 && $row['bad_fb'] > 0) {
            $rootCause[] = "Nấu đúng thực đơn nhưng nhận phản hồi kém ➔ Công thức, định lượng gia vị có vấn đề, cần cải tiến công thức.";
        }

        // Phân loại Đánh giá
        if ($riskScore >= 50 || count($foundHazards) > 0) {
            $msg = "Món <b>{$row['dish_name']}</b>: ";
            if (count($foundHazards) > 0) $msg .= "Phát hiện từ khóa nguy hiểm (<i>" . implode(", ", $foundHazards) . "</i>). ";
            if ($row['bad_fb'] > 0) $msg .= "Có {$row['bad_fb']} lượt đánh giá Kém. ";
            if (!empty($rootCause)) $msg .= "Nguyên nhân khả nghi: " . implode(" ", $rootCause);
            
            $criticalIssues[] = $msg;
        } elseif ($riskScore >= 20 && $riskScore < 50) {
            $msg = "Món <b>{$row['dish_name']}</b>: Tỷ lệ sai/đổi ({$row['fails']} lần) hoặc bị phàn nàn về chất lượng (<i>" . implode(", ", $foundQualities) . "</i>). Cần bếp trưởng kiểm tra lại khâu chế biến.";
            $qualityIssues[] = $msg;
        }
    }

    // Đưa ra Kết luận cuối cùng dựa trên phân tích
    $goodRate = $totalFeedbacksGlobal > 0 ? round(($totalGood / $totalFeedbacksGlobal) * 100, 1) : 100;

    if (!empty($criticalIssues)) {
        $smartInsights[] = [
            'type' => 'danger',
            'title' => '🚨 CẢNH BÁO MỨC ĐỘ 1: RỦI RO AN TOÀN THỰC PHẨM & DỊCH VỤ:',
            'desc' => "Hệ thống A.I phát hiện các nguy cơ tiềm ẩn nghiêm trọng cần xử lý lập tức (Đình chỉ món hoặc thanh tra bếp):<br>• " . implode("<br>• ", $criticalIssues)
        ];
    }
    
    if (!empty($qualityIssues)) {
        $smartInsights[] = [
            'type' => 'warning',
            'title' => '⚠️ CẢNH BÁO MỨC ĐỘ 2: KHUYẾN NGHỊ ĐIỀU CHỈNH CÔNG THỨC / VẬN HÀNH:',
            'desc' => "Nhà quản lý cần họp với bộ phận bếp để chấn chỉnh các vấn đề sau nhằm tránh giảm sút sự hài lòng:<br>• " . implode("<br>• ", $qualityIssues)
        ];
    }

    if ($goodRate < 60 && $totalFeedbacksGlobal > 0) {
        $smartInsights[] = [

<script>
function openModal(id) { document.getElementById(id).style.display = 'flex'; }
function closeModal(id) { document.getElementById(id).style.display = 'none'; }
function toggleMenuDropdown(e) {
    if (e) e.stopPropagation();
    const menu = document.getElementById('menu_dropdown_list');
    if (menu) {
        menu.classList.toggle('show');
    }
}
function closeAllDropdowns() {
    const menu = document.getElementById('menu_dropdown_list');
    if (menu) {
        menu.classList.remove('show');
    }
}
function openEditMenuModal(m) {
    document.getElementById('edit_menu_id').value = m.id;
    document.getElementById('edit_day_of_week').value = m.day_of_week;
    document.getElementById('edit_meal_type').value = m.meal_type || 'Bữa Trưa';
    document.getElementById('edit_dish_name').value = m.dish_name;
    document.getElementById('edit_description').value = m.description || '';
    openModal('modal-edit-menu');
}
function openEditFeedbackModal(fb) {
    document.getElementById('edit_feedback_id').value = fb.id;
    document.getElementById('edit_feedback_rating').value = fb.quality_rating;
    document.getElementById('edit_feedback_note').value = fb.quality_note || '';
    openModal('modal-edit-feedback');
}
window.onclick = function(e) {
    if (e.target.classList.contains('modal-overlay')) {
        e.target.style.display = 'none';
    }
    if (!e.target.closest('.menu-btn-group')) {
        closeAllDropdowns();
    }
}
</script>
