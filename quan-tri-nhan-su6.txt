// Hàm tính thâm niên làm việc
function calculateWorkDuration($startDate) {
    if (empty($startDate) || $startDate === '0000-00-00' || strpos($startDate, '0000') !== false) return '—';
    try {
        $time = strtotime($startDate);
        if ($time === false || $time <= 0) return '—';
        $start = new DateTime(date('Y-m-d', $time));
        $now = new DateTime();
        if ($start > $now) return '0 ngày';
        $diff = $start->diff($now);
        $parts = [];
        if ($diff->y > 0) $parts[] = $diff->y . ' năm';
        if ($diff->m > 0) $parts[] = $diff->m . ' tháng';
        if ($diff->d > 0) $parts[] = $diff->d . ' ngày';
        return !empty($parts) ? implode(' ', $parts) : '0 ngày';
    } catch (Exception $e) {
        return '—';
    }
}

    $att_date = $_POST['att_date'];
    $time_in  = !empty($_POST['time_in']) ? $_POST['time_in'] : null;
    $time_out = !empty($_POST['time_out']) ? $_POST['time_out'] : null;
    $note     = trim($_POST['note'] ?? '');

    $status = 'Đúng giờ';
    if (!$time_in && !$time_out) {
        $status = 'Vắng mặt';
    } else {
        $is_late = ($time_in && $time_in > '08:05:00');
        $is_early = ($time_out && $time_out < '17:00:00');
        if ($is_late && $is_early) $status = 'Trễ & Sớm';
        elseif ($is_late) $status = 'Đi trễ';
        elseif ($is_early) $status = 'Về sớm';
    }

    try {
        $chk = $pdo->prepare("SELECT id FROM attendance WHERE emp_id = ? AND date = ?");
        $chk->execute([$emp_id, $att_date]);
        if ($exist = $chk->fetch()) {
            $upd = $pdo->prepare("UPDATE attendance SET time_in = ?, time_out = ?, status = ?, note = ? WHERE id = ?");
            $upd->execute([$time_in, $time_out, $status, $note, $exist['id']]);
        } else {
            $ins = $pdo->prepare("INSERT INTO attendance (emp_id, date, time_in, time_out, status, note) VALUES (?, ?, ?, ?, ?, ?)");
            $ins->execute([$emp_id, $att_date, $time_in, $time_out, $status, $note]);
        }
        $message = "Đã cập nhật chấm công cho nhân viên #$emp_id!";

// Quản lý chuyển tab mượt mà
function changeTab(tabId) {
    document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));
    document.querySelectorAll('.nav-tab-btn').forEach(el => el.classList.remove('active'));
    
    let targetTab = document.getElementById(tabId);
    let targetBtn = document.getElementById('btn-' + tabId);
    if (targetTab && targetBtn) {
        targetTab.classList.add('active');
        targetBtn.classList.add('active');
        localStorage.setItem('hrm_current_tab', tabId);
    }
}

window.addEventListener('DOMContentLoaded', () => {
    let urlParams = new URLSearchParams(window.location.search);
    let tabFromUrl = urlParams.get('tab');
    let tabFromServer = "<?= htmlspecialchars($active_tab) ?>";
    let activeTab = tabFromUrl || tabFromServer || localStorage.getItem('hrm_current_tab') || 'employees-tab';
    changeTab(activeTab);
});

// Dropdown thao tác ẩn trong icon cây bút
function toggleActionMenu(event, btn) {
    event.stopPropagation();
    let parent = btn.closest('.action-dropdown');
    document.querySelectorAll('.action-dropdown').forEach(dd => {
        if (dd !== parent) dd.classList.remove('active');
    });
    parent.classList.toggle('active');
}

document.addEventListener('click', () => {
    document.querySelectorAll('.action-dropdown').forEach(dd => dd.classList.remove('active'));
});
