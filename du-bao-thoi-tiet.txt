<?php
date_default_timezone_set('Asia/Bangkok');

// Danh sách các địa điểm cài sẵn
$locations = [
    'ninh_hiep'  => ['name' => 'Ninh Hiệp, Ninh Hòa', 'lat' => 12.6075, 'lon' => 109.1235],
    'ninh_phuoc'  => ['name' => 'Ninh Phước, Ninh Hòa', 'lat' => 12.485186, 'lon' => 109.283409],
    'hoa_thang'  => ['name' => 'Hòa Thắng, Ninh Hòa',  'lat' => 12.6333, 'lon' => 109.1667],
    'ninh_da'    => ['name' => 'Ninh Đa, Ninh Hòa',    'lat' => 12.6167, 'lon' => 109.1500],
    'nha_trang'  => ['name' => 'TP. Nha Trang',        'lat' => 12.2388, 'lon' => 109.1967],
    'van_ninh'   => ['name' => 'Huyện Vạn Ninh',       'lat' => 12.7000, 'lon' => 109.2333],
];

$selected_loc = $_POST['location'] ?? 'ninh_hiep';
$lat = $locations[$selected_loc]['lat'] ?? 12.6075;
$lon = $locations[$selected_loc]['lon'] ?? 109.1235;

$selected_date = $_POST['target_date'] ?? date('Y-m-d', strtotime('+1 day'));
$time_period   = $_POST['time_period'] ?? 'day'; // 'day' hoặc 'night'

$result = null;
$error  = null;

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    // Quy chuẩn khung giờ: Ban ngày chọn 14h00 (đỉnh dông nhiệt), Ban đêm chọn 21h00
    $hour_slot = ($time_period === 'day') ? '14:00' : '21:00';
    $formatted_time = "{$selected_date}T{$hour_slot}";
    $target_timestamp = strtotime($formatted_time);

    // 1. API mô hình ECMWF (Châu Âu - Windy dùng làm lõi)
    $url_ecmwf = "https://api.open-meteo.com/v1/ecmwf?latitude={$lat}&longitude={$lon}&hourly=temperature_2m,precipitation,surface_pressure,wind_speed_10m,wind_direction_10m&timezone=Asia%2FBangkok";
    
    // 2. API mô hình ICON (Đức - Zoom.earth và Windy dùng bắt vi khí hậu)
    $url_icon = "https://api.open-meteo.com/v1/dwd-icon?latitude={$lat}&longitude={$lon}&hourly=temperature_2m,precipitation,surface_pressure,wind_speed_10m,wind_direction_10m&timezone=Asia%2FBangkok";

    // Khởi tạo cURL lấy dữ liệu an toàn
    $fetch_data = function($url) {
        $ch = curl_init();
        curl_setopt($ch, CURLOPT_URL, $url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_TIMEOUT, 6);
        curl_setopt($ch, CURLOPT_USERAGENT, 'WeatherPHPApp/1.0');
        $resp = curl_exec($ch);
        curl_close($ch);
        return $resp ? json_decode($resp, true) : null;
    };

    $data_ecmwf = $fetch_data($url_ecmwf);
    $data_icon  = $fetch_data($url_icon);

    if ($data_ecmwf && $data_icon) {
        $idx_ecmwf = array_search($formatted_time, $data_ecmwf['hourly']['time'] ?? []);
        $idx_icon  = array_search($formatted_time, $data_icon['hourly']['time'] ?? []);

        if ($idx_ecmwf !== false && $idx_icon !== false) {
            // Lấy các chỉ số khí tượng
            $rain_ecmwf      = (float)$data_ecmwf['hourly']['precipitation'][$idx_ecmwf];
            $temp_ecmwf      = (float)$data_ecmwf['hourly']['temperature_2m'][$idx_ecmwf];
            $wind_deg_ecmwf  = (float)$data_ecmwf['hourly']['wind_direction_10m'][$idx_ecmwf];

            $rain_icon       = (float)$data_icon['hourly']['precipitation'][$idx_icon];
            $temp_icon       = (float)$data_icon['hourly']['temperature_2m'][$idx_icon];
            $wind_speed_icon = (float)$data_icon['hourly']['wind_speed_10m'][$idx_icon];
            $wind_deg_icon   = (float)$data_icon['hourly']['wind_direction_10m'][$idx_icon];
            $pressure_icon   = (float)$data_icon['hourly']['surface_pressure'][$idx_icon];

            // Áp dụng công thức chuyên gia
            $probability = 0;
            $status = "Tạnh ráo";
            $notes = [];

            // Quy tắc 1: Đối chiếu lượng mưa giữa ECMWF và ICON
            if ($rain_ecmwf >= 2.0 && $rain_icon >= 2.0) {
                $probability = 85;
                $status = "Khả năng mưa to rất cao";
                $notes[] = "Cả ECMWF ({$rain_ecmwf} mm) và ICON ({$rain_icon} mm) đều đồng thuận báo mưa rõ rệt.";
            } elseif ($rain_ecmwf >= 0.8 && $rain_icon >= 0.8) {
                $probability = 70;
                $status = "Khả năng có mưa rào";
                $notes[] = "Cả hai mô hình đều báo mưa nhỏ đến mưa vừa (ECMWF: {$rain_ecmwf} mm, ICON: {$rain_icon} mm).";
            } elseif ($rain_ecmwf >= 2.0 && $rain_icon < 0.5) {
                $probability = 60;
                $status = "Có thể có mưa cục bộ";
                $notes[] = "ECMWF báo mưa ({$rain_ecmwf} mm) nhưng ICON báo tạnh. Ưu tiên độ uy tín của ECMWF.";
            } elseif ($rain_icon >= 2.0 && $rain_ecmwf < 0.5) {
                $probability = 50;
                $status = "Mưa dông nhiệt thoáng qua";
                $notes[] = "ICON báo mưa ({$rain_icon} mm) nhưng ECMWF báo 0 mm. Dễ là mưa rào bóng mây ngắn hạn.";
            } else {
                $probability = 15;
                $status = "Tạnh ráo / Khô ráo";
                $notes[] = "Cả ECMWF và ICON đều không ghi nhận mưa đáng kể tại mốc giờ này.";
            }

            // Quy tắc 2: Phân tích hướng gió (Ninh Hòa: góc từ 60° đến 150° là luồng gió biển Đông/Đông Nam)
            $is_sea_breeze = ($wind_deg_icon >= 60 && $wind_deg_icon <= 150);
            if ($is_sea_breeze) {
                $probability += 10;
                $notes[] = "Gió Đông/Đông Nam từ biển đang thốc hơi ẩm vào bờ ({$wind_speed_icon} km/h), xúc tác tụ mây.";
            } else {
                $notes[] = "Gió lục địa hoặc gió từ sườn núi khô, không có luồng ẩm biển trực tiếp.";
            }

            // Quy tắc 3: Bắt dông nhiệt buổi chiều (chỉ xét khung Ban Ngày)
            if ($time_period === 'day') {
                if ($temp_icon >= 32.0 && $is_sea_breeze) {
                    $probability += 10;
                    $status .= " (Cảnh báo: Dông sét buổi chiều)";
                    $notes[] = "Nhiệt độ trưa rất cao ({$temp_icon}°C) kết hợp gió biển tạo xung lực đối lưu cực mạnh, dễ có sấm sét và gió giật.";
                }
                if ($pressure_icon < 1008.0) {
                    $notes[] = "Khí áp tụt thấp ({$pressure_icon} hPa), hỗ trợ việc bốc hơi và hình thành ổ dông.";
                }
            } else {
                // Phân tích ban đêm
                $notes[] = "Thời gian ban đêm: Khối nhiệt mặt đất đã hạ, chủ yếu chịu ảnh hưởng của hoàn lưu gió nền.";
            }

            $probability = min(98, max(5, $probability));

            $result = [
                'time_str'    => ($time_period === 'day' ? 'Ban Ngày (14:00)' : 'Ban Đêm (21:00)') . " - " . date('d/m/Y', $target_timestamp),
                'status'      => $status,
                'probability' => $probability,
                'rain_ecmwf'  => $rain_ecmwf,
                'rain_icon'   => $rain_icon,
                'temp'        => $temp_icon,
                'wind_speed'  => $wind_speed_icon,
                'pressure'    => $pressure_icon,
                'notes'       => $notes
            ];
        } else {
            $error = "Mốc thời gian chọn nằm ngoài phạm vi dự báo của mô hình (tối đa trong vòng 7 ngày tới).";
        }
    } else {
        $error = "Không thể kết nối đến máy chủ dữ liệu Open-Meteo. Vui lòng kiểm tra lại kết nối mạng.";
    }
}
