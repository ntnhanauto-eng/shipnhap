function renderPaginationBar($currentPage, $totalPages, $paramName) {
    if ($totalPages <= 1) return;
    $queryParams = $_GET;

    echo '<div class="pagination-wrapper">';
    if ($currentPage > 1) {
        $queryParams[$paramName] = $currentPage - 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&laquo;</a>';
    } else {
        echo '<span class="page-link disabled">&laquo;</span>';
    }

    $start = max(1, $currentPage - 1);
    $end = min($totalPages, $currentPage + 1);
    if ($currentPage == 1) $end = min($totalPages, $start + 2);
    elseif ($currentPage == $totalPages) $start = max(1, $end - 2);

    for ($i = $start; $i <= $end; $i++) {
        $queryParams[$paramName] = $i;
        $activeClass = ($i == $currentPage) ? 'active' : '';
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link ' . $activeClass . '">' . $i . '</a>';
    }

    if ($currentPage < $totalPages) {
        $queryParams[$paramName] = $currentPage + 1;
        echo '<a href="?' . http_build_query($queryParams) . '" class="page-link">&raquo;</a>';
    } else {
        echo '<span class="page-link disabled">&raquo;</span>';
    }
    echo '</div>';
}

function generateUniqueVppCode($pdo, $table, $timeStr, $excludeId = 0) {
    $ts = strtotime($timeStr) ?: time();
    $baseCode = date('dmYHis', $ts);

    $sql = "SELECT request_code FROM `{$table}` WHERE request_code LIKE :b AND id != :exId";
    $stmt = $pdo->prepare($sql);
    $stmt->execute([':b' => $baseCode . '%', ':exId' => $excludeId]);
    $existing = $stmt->fetchAll(PDO::FETCH_COLUMN);

    if (empty($existing) || !in_array($baseCode, $existing)) {
        return $baseCode;
    }

    $suffixes = range('A', 'Z');
    foreach ($suffixes as $char) {
        $candidate = $baseCode . $char;
        if (!in_array($candidate, $existing)) {
            return $candidate;
        }
    }
    return $baseCode . '_' . rand(10, 99);
}

function getDisplayCode($row) {
    if (!empty($row['request_code'])) {
        return $row['request_code'];
    }
    return date('dmYHis', strtotime($row['created_at']));
}

function normalizeFastText($str) {
    $str = str_replace("\xC2\xA0", ' ', (string)$str);
    $str = mb_strtolower(trim($str), 'UTF-8');
    $str = preg_replace('/[áàảãạăắằẳẵặâấầẩẫậ]/u', 'a', $str);
    $str = preg_replace('/[éèẻẽẹêếềểễệ]/u', 'e', $str);
    $str = preg_replace('/[íìỉĩị]/u', 'i', $str);
    $str = preg_replace('/[óòỏõọôốồổỗộơớờởỡợ]/u', 'o', $str);
    $str = preg_replace('/[úùủũụưứừửữự]/u', 'u', $str);
    $str = preg_replace('/[ýỳỷỹỵ]/u', 'y', $str);
    $str = preg_replace('/đ/u', 'd', $str);
    return preg_replace('/[^a-z0-9]/', '', $str);
}

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'deliver_group_proposals') {
    $req_code      = trim($_POST['request_code'] ?? '');
    $receiver_name = trim($_POST['receiver_name'] ?? '');
    $department    = trim($_POST['department'] ?? '');
    $delivery_loc  = trim($_POST['delivery_location'] ?? '');
    $custom_notes  = trim($_POST['notes'] ?? '');
    $deliver_items = $_POST['deliver_quantities'] ?? [];

    try {
        if (empty($req_code)) throw new Exception("Không xác định được mã yêu cầu đề xuất!");
        $pdo->beginTransaction();

        $stmtProps = $pdo->prepare("
            SELECT p.*, i.item_name, i.stock_qty, i.unit 
            FROM vpp_proposals p 
            JOIN vpp_items i ON p.item_id = i.id 
            WHERE p.request_code = :code AND p.status IN ('Đã duyệt', 'Chờ cấp phát')
            FOR UPDATE
        ");
        $stmtProps->execute([':code' => $req_code]);
        $rows = $stmtProps->fetchAll();

        if (empty($rows)) {
            throw new Exception("Yêu cầu đề xuất chưa được duyệt hoặc đã được cấp phát từ trước!");
        }

        $stmtSub = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty - :qty WHERE id = :id");
        $stmtReq = $pdo->prepare("
            INSERT INTO vpp_requests (request_code, request_date, department, receiver_name, item_id, quantity, delivery_location, notes)
            VALUES (:code, :r_date, :dept, :rec, :item_id, :qty, :loc, :notes)
        ");
        $stmtUpProp = $pdo->prepare("UPDATE vpp_proposals SET status = 'Đã cấp phát', approved_quantity = :appr_q WHERE id = :id");

        $itemDeliveredCount = 0;
        foreach ($rows as $p) {
            $propId = $p['id'];
            $giveQty = isset($deliver_items[$propId]) ? max(1, intval($deliver_items[$propId])) : max(1, intval($p['approved_quantity'] !== null ? $p['approved_quantity'] : $p['quantity']));

            if ($p['stock_qty'] < $giveQty) {
                throw new Exception("Tồn kho không đủ để cấp mặt hàng [{$p['item_name']}]! Tồn kho hiện có: {$p['stock_qty']} {$p['unit']}, cần cấp: {$giveQty} {$p['unit']}.");
            }

            $stmtSub->execute([':qty' => $giveQty, ':id' => $p['item_id']]);
            $stmtReq->execute([
                ':code'    => $req_code,
                ':r_date'  => date('Y-m-d'),
                ':dept'    => $department ?: $p['department'],
                ':rec'     => $receiver_name ?: $p['receiver_name'],
                ':item_id' => $p['item_id'],
                ':qty'     => $giveQty,
                ':loc'     => $delivery_loc ?: $p['delivery_location'],
                ':notes'   => $custom_notes
            ]);
            $stmtUpProp->execute([':appr_q' => $giveQty, ':id' => $propId]);
            $itemDeliveredCount++;
        }

        $pdo->commit();
        $message = "Đã xác nhận cấp phát thành công {$itemDeliveredCount} mặt hàng cho yêu cầu [{$req_code}]!";
    } catch (Exception $e) {
        if ($pdo->inTransaction()) $pdo->rollBack();
        $error = "Lỗi cấp phát: " . $e->getMessage();
    }
}

        $tmpName  = $_FILES['excel_file']['tmp_name'];
        $fileName = $_FILES['excel_file']['name'];
        $ext      = strtolower(pathinfo($fileName, PATHINFO_EXTENSION));

        $rows = [];
        if ($ext === 'csv') {
            if (($handle = fopen($tmpName, 'r')) !== false) {
                while (($data = fgetcsv($handle, 4000, ',')) !== false) $rows[] = array_map('trim', $data);
                fclose($handle);
            }
        } elseif ($ext === 'xlsx') {
            $rows = parseXlsxFastStream($tmpName);
        } elseif ($ext === 'xls') {
            $content = file_get_contents($tmpName);
            if (strpos($content, '<table') !== false) {
                $dom = new DOMDocument();
                @$dom->loadHTML(mb_convert_encoding($content, 'HTML-ENTITIES', 'UTF-8'));
                $trs = $dom->getElementsByTagName('tr');
                foreach ($trs as $tr) {
                    $rowValues = [];
                    $tds = $tr->getElementsByTagName('td');
                    if ($tds->length === 0) $tds = $tr->getElementsByTagName('th');
                    foreach ($tds as $td) $rowValues[] = trim($td->textContent);
                    if (!empty(array_filter($rowValues, fn($v) => $v !== ''))) $rows[] = $rowValues;
                }
            } else {
                $rows = parseXlsxFastStream($tmpName);
            }
        }

        if (!empty($rows)) {
            $header = array_shift($rows);
            $idxDate = 0; $idxSup = 1; $idxCode = 2; $idxName = 3; $idxSpec = 4; $idxUnit = 5; $idxQty = 6; $idxInv = 7; $idxNote = 8;
            foreach ($header as $colIdx => $colName) {
                $cNorm = normalizeFastText($colName);
                if (strpos($cNorm, 'ngay') !== false) $idxDate = $colIdx;
                elseif (strpos($cNorm, 'ncc') !== false || strpos($cNorm, 'nhacungcap') !== false) $idxSup = $colIdx;
                elseif (strpos($cNorm, 'mavpp') !== false || strpos($cNorm, 'code') !== false) $idxCode = $colIdx;
                elseif (strpos($cNorm, 'ten') !== false) $idxName = $colIdx;
                elseif (strpos($cNorm, 'quycach') !== false) $idxSpec = $colIdx;
                elseif (strpos($cNorm, 'dvt') !== false || strpos($cNorm, 'donvi') !== false) $idxUnit = $colIdx;
                elseif (strpos($cNorm, 'soluong') !== false || strpos($cNorm, 'sl') !== false) $idxQty = $colIdx;
                elseif (strpos($cNorm, 'hoadon') !== false || strpos($cNorm, 'chungtu') !== false) $idxInv = $colIdx;
                elseif (strpos($cNorm, 'ghichu') !== false) $idxNote = $colIdx;
            }

            $allItems = $pdo->query("SELECT id, item_code, item_name, specification, unit, stock_qty FROM vpp_items")->fetchAll();
            $itemMapByCode = []; $itemMapByName = [];
            foreach ($allItems as $it) {
                if (!empty($it['item_code'])) $itemMapByCode[normalizeFastText($it['item_code'])] = $it;
                if (!empty($it['item_name'])) $itemMapByName[normalizeFastText($it['item_name'])] = $it;
            }

            $successCount = 0;
            $pdo->beginTransaction();
            try {
                $stmtInsertItem = $pdo->prepare("INSERT INTO vpp_items (item_code, item_name, specification, unit, initial_stock, total_imported, stock_qty) VALUES (:code, :name, :spec, :unit, 0, 0, 0)");
                $stmtImport = $pdo->prepare("INSERT INTO vpp_imports (invoice_no, supplier_name, import_date, item_id, quantity, notes) VALUES (:inv, :sup, :idate, :it_id, :qty, :n)");
                $stmtUpStock = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty + :q1, total_imported = total_imported + :q2 WHERE id = :id");

                foreach ($rows as $row) {
                    $code = trim($row[$idxCode] ?? '');
                    $name = trim($row[$idxName] ?? '');
                    $qty  = intval(preg_replace('/[^0-9]/', '', (string)($row[$idxQty] ?? 0)));
                    if (empty($name) && empty($code)) continue;
                    if ($qty <= 0) continue;

                    $cKey = normalizeFastText($code);
                    $nKey = normalizeFastText($name);
                    $target = $itemMapByCode[$cKey] ?? ($itemMapByName[$nKey] ?? null);

                    if (!$target) {
                        $targetCode = $code ?: ('VPP_' . strtoupper(substr(md5(uniqid()), 0, 5)));
                        $targetName = $name ?: ('Mặt hàng mã ' . $targetCode);
                        $targetUnit = trim($row[$idxUnit] ?? 'Cái') ?: 'Cái';
                        $targetSpec = trim($row[$idxSpec] ?? '');
                        $stmtInsertItem->execute([':code' => $targetCode, ':name' => $targetName, ':spec' => $targetSpec, ':unit' => $targetUnit]);
                        $targetId = $pdo->lastInsertId();
                    } else {
                        $targetId = $target['id'];
                    }

                    $stmtImport->execute([
                        ':inv'   => trim($row[$idxInv] ?? ''),
                        ':sup'   => trim($row[$idxSup] ?? 'Nhà cung cấp tổng hợp') ?: 'Nhà cung cấp tổng hợp',
                        ':idate' => parseExcelDate($row[$idxDate] ?? '', $today),
                        ':it_id' => $targetId,
                        ':qty'   => $qty,
                        ':n'     => trim($row[$idxNote] ?? '')
                    ]);
                    $stmtUpStock->execute([':q1' => $qty, ':q2' => $qty, ':id' => $targetId]);
                    $successCount++;
                }
                $pdo->commit();
                $message = "Đã nhập thành công {$successCount} dòng hàng vào kho!";
            } catch (Exception $e) {
                $pdo->rollBack();
                $error = "Lỗi nhập dữ liệu: " . $e->getMessage();
            }
        }
    }
}

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'update_request') {
    $req_id   = intval($_POST['request_id']);
    $new_date = $_POST['request_date'];
    $new_dept = trim($_POST['department']);
    $new_rec  = trim($_POST['receiver_name'] ?? '');
    $new_item = intval($_POST['item_id']);
    $new_qty  = intval($_POST['quantity']);
    $new_loc  = trim($_POST['delivery_location'] ?? '');
    $new_notes= trim($_POST['notes'] ?? '');

    try {
        $pdo->beginTransaction();
        $stmtOld = $pdo->prepare("SELECT * FROM vpp_requests WHERE id = :id FOR UPDATE");
        $stmtOld->execute([':id' => $req_id]);
        $oldReq = $stmtOld->fetch();
        if (!$oldReq) throw new Exception("Bản ghi cấp phát không tồn tại!");

        $stmtIt = $pdo->prepare("SELECT item_name, specification, stock_qty, unit FROM vpp_items WHERE id = :id FOR UPDATE");
        $stmtIt->execute([':id' => $new_item]);
        $it = $stmtIt->fetch();
        if (!$it) throw new Exception("Mặt hàng VPP không tồn tại!");

        if (intval($oldReq['item_id']) === $new_item) {
            $diff = $new_qty - intval($oldReq['quantity']);
            if ($diff > 0 && $diff > $it['stock_qty']) {
                throw new Exception("Không đủ hàng trong kho để tăng số lượng cấp! Còn thiếu {$diff} {$it['unit']}.");
            }
            $stmtUpStock = $pdo->prepare("UPDATE vpp_items SET stock_qty = GREATEST(0, stock_qty - :diff) WHERE id = :id");
            $stmtUpStock->execute([':diff' => $diff, ':id' => $new_item]);
        } else {
            $stmtBack = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty + :q WHERE id = :id");
            $stmtBack->execute([':q' => $oldReq['quantity'], ':id' => $oldReq['item_id']]);

            if ($new_qty > $it['stock_qty']) {
                throw new Exception("Không đủ tồn kho mặt hàng mới để cấp! Tồn: {$it['stock_qty']} {$it['unit']}.");
            }
            $stmtTake = $pdo->prepare("UPDATE vpp_items SET stock_qty = stock_qty - :q WHERE id = :id");
            $stmtTake->execute([':q' => $new_qty, ':id' => $new_item]);
        }

        $stmtUpReq = $pdo->prepare("UPDATE vpp_requests SET request_date = :r_date, department = :dept, receiver_name = :rec, item_id = :item_id, quantity = :qty, delivery_location = :loc, notes = :notes WHERE id = :id");
        $stmtUpReq->execute([':r_date' => $new_date, ':dept' => $new_dept, ':rec' => $new_rec, ':item_id' => $new_item, ':qty' => $new_qty, ':loc' => $new_loc, ':notes' => $new_notes, ':id' => $req_id]);

        $pdo->commit();
        $message = "Đã cập nhật bản ghi lịch sử cấp phát thành công!";
    } catch (Exception $e) {
        if ($pdo->inTransaction()) $pdo->rollBack();
        $error = "Lỗi cập nhật: " . $e->getMessage();
    }
}

                    fclose($handle);
                }

                $rows = parseXlsxFastStream($tmpName);

                $pdo->beginTransaction();
                try {
                    $stmtCheck = $pdo->prepare("SELECT id FROM vpp_items WHERE item_code = :c OR item_name = :n LIMIT 1");
                    $stmtIns = $pdo->prepare("INSERT INTO vpp_items (item_code, item_name, specification, unit, initial_stock, total_imported, stock_qty) VALUES (:c, :n, :s, :u, 0, 0, 0)");

                        $code = trim($r[0] ?? '');
                        $name = trim($r[1] ?? '');
                        $spec = trim($r[2] ?? '');
                        $unit = trim($r[3] ?? 'Cái') ?: 'Cái';
                        if (empty($name)) continue;

                        $stmtCheck->execute([':c' => $code, ':n' => $name]);

                            $upCount++;

                            $stmtIns->execute([':c' => $code, ':n' => $name, ':s' => $spec, ':u' => $unit]);

    }

    if ($act === 'reset_all_data') {
        $pdo->exec("UPDATE vpp_items SET initial_stock = 0, total_imported = 0, stock_qty = 0");
        $pdo->exec("TRUNCATE TABLE vpp_requests");
        $pdo->exec("TRUNCATE TABLE vpp_proposals");
        $pdo->exec("TRUNCATE TABLE vpp_imports");
        $message = "Đã reset toàn bộ dữ liệu văn phòng phẩm về 0!";
    }

function saveScrollPosition() {
    localStorage.setItem('vpp_scroll_pos', window.scrollY);

    const scrollPos = localStorage.getItem('vpp_scroll_pos');

        localStorage.removeItem('vpp_scroll_pos');
    }

function openModal(id) { document.getElementById(id).style.display = 'flex'; }
function closeModal(id) { document.getElementById(id).style.display = 'none'; }

function toggleDropdownMenu(e, id) {
    if (e) e.stopPropagation();
    const menu = document.getElementById(id);
    const isShown = menu && menu.classList.contains('show');
    closeAllDropdownMenus();
    if (menu && !isShown) menu.classList.add('show');
}
function closeAllDropdownMenus() {
    document.querySelectorAll('.dropdown-action-menu').forEach(el => el.classList.remove('show'));
}

function togglePropDropdown(e, code) {
    if (e) e.stopPropagation();
    const currentMenu = document.getElementById('prop_dropdown_' + code);
    if (!currentMenu) return;
    const isShown = currentMenu.classList.contains('show');
    closeAllPropDropdowns();
    closeAllPencilDropdowns();
    closeAllNotePopovers();
    if (!isShown) currentMenu.classList.add('show');
}
function closeAllPropDropdowns() {
    document.querySelectorAll('.prop-dropdown-menu').forEach(el => el.classList.remove('show'));
}

function togglePropPencilDropdown(e, code) {
    if (e) e.stopPropagation();
    const currentMenu = document.getElementById('prop_pencil_menu_' + code);
    if (!currentMenu) return;
    const isShown = currentMenu.classList.contains('show');
    closeAllPropDropdowns();
    closeAllPencilDropdowns();
    closeAllNotePopovers();
    if (!isShown) currentMenu.classList.add('show');
}
function closeAllPencilDropdowns() {
    document.querySelectorAll('.prop-pencil-menu').forEach(el => el.classList.remove('show'));
}

    const container = btn.closest('.note-interactive-container');
    if (!container) return;
    const pop = container.querySelector('.note-popover');
    if (!pop) return;
    const isShown = pop.classList.contains('show');
    closeAllNotePopovers();
    closeAllPropDropdowns();
    closeAllPencilDropdowns();
    if (!isShown) pop.classList.add('show');
}
function closeAllNotePopovers() {

}

function openPrintDocModal(p) {
    document.getElementById('prn_receiver_name').innerText = p.receiver_name;
    document.getElementById('prn_sign_receiver').innerText = p.receiver_name;
    
    const empInfo = p.employee_id ? `#${p.employee_id} / ${p.department}` : p.department;
    document.getElementById('prn_dept_info').innerText = empInfo;

    const tbody = document.getElementById('prn_items_tbody');
    tbody.innerHTML = '';
    const rawItems = JSON.parse(p.items_json || '[]');
    let totalQty = 0;
    let validRowIndex = 1;

    rawItems.forEach(it => {
        const qtyToIssue = (it.pre_approved_quantity !== null) ? parseInt(it.pre_approved_quantity) : parseInt(it.quantity);
        if (qtyToIssue <= 0) return;

        totalQty += qtyToIssue;
        const stockLeft = Math.max(0, parseInt(it.stock_qty) - qtyToIssue);

        const tr = document.createElement('tr');
        tr.innerHTML = `
            <td>${validRowIndex++}</td>
            <td>${it.item_code || '—'}</td>
            <td class="text-left"><b>${it.item_name}</b> ${it.specification ? '(' + it.specification + ')' : ''}</td>
            <td>${it.unit}</td>
            <td style="font-weight:bold;">${qtyToIssue}</td>
            <td>${stockLeft}</td>
            <td></td>
        `;
        tbody.appendChild(tr);
    });

    const currentRowsCount = validRowIndex - 1;
    const emptyRowsCount = Math.max(0, 6 - currentRowsCount);
    for (let i = 0; i < emptyRowsCount; i++) {
        const tr = document.createElement('tr');
        tr.innerHTML = `<td>${currentRowsCount + i + 1}</td><td></td><td></td><td></td><td></td><td></td><td></td>`;
        tbody.appendChild(tr);
    }

    document.getElementById('prn_total_qty').innerText = totalQty;
    openModal('modal-print-view');
}

    const items = JSON.parse(p.items_json || '[]');
    const isApprovedFinal = (p.status === 'Đã duyệt' || p.status === 'Đã cấp phát');

            <td style="text-align:center; font-weight:bold; color:#0284c7;">${it.quantity}</td>
            <td style="text-align:center; font-weight:bold; color:#15803d;">${apprText}</td>
