// ==================== HÀM NÉN ẢNH DƯỚI 100KB ====================

function compressAndUploadImage($file, $targetDir = 'uploads/vpp/') {

if (!isset($file['tmp_name']) || empty($file['tmp_name'])) return '';

if (!is_dir($targetDir)) {

mkdir($targetDir, 0777, true);

}



$fileInfo = @getimagesize($file['tmp_name']);

if ($fileInfo === false) return '';



$mime = $fileInfo['mime'];

$srcImg = null;

if ($mime === 'image/jpeg') {

$srcImg = imagecreatefromjpeg($file['tmp_name']);

} elseif ($mime === 'image/png') {

$srcImg = imagecreatefrompng($file['tmp_name']);

} elseif ($mime === 'image/webp') {

$srcImg = imagecreatefromwebp($file['tmp_name']);

}

if (!$srcImg) return '';



$ext = pathinfo($file['name'], PATHINFO_EXTENSION);

$newFileName = 'vpp_mua_' . uniqid() . '.' . ($ext ?: 'jpg');

$destination = $targetDir . $newFileName;



$quality = 85;

if ($mime === 'image/png') {

imagepng($srcImg, $destination, 8);

} else {

imagejpeg($srcImg, $destination, $quality);

}



while (filesize($destination) > 100 * 1024 && $quality > 20) {

$quality -= 15;

if ($mime === 'image/png') {

imagepng($srcImg, $destination, 9);

} else {

imagejpeg($srcImg, $destination, $quality);

}

}



imagedestroy($srcImg);

return $destination;

}

try {

$pdo->exec("

CREATE TABLE IF NOT EXISTS `vpp_purchase_proposals` (

`id` INT AUTO_INCREMENT PRIMARY KEY,

`request_code` VARCHAR(60) DEFAULT '',

`employee_id` INT DEFAULT NULL,

`receiver_name` VARCHAR(150) NOT NULL,

`department` VARCHAR(100) NOT NULL,

`item_id` INT DEFAULT NULL,

`custom_item_name` VARCHAR(150) DEFAULT '',

`specification` VARCHAR(100) DEFAULT '',

`unit` VARCHAR(50) DEFAULT 'Cái',

`custom_image` VARCHAR(255) DEFAULT '',

`quantity` INT NOT NULL DEFAULT 1,

`pre_approved_quantity` INT DEFAULT NULL,

`approved_quantity` INT DEFAULT NULL,

`approval_note` VARCHAR(255) DEFAULT '',

`status` VARCHAR(50) DEFAULT 'Chờ duyệt',

`reject_reason` VARCHAR(255) DEFAULT '',

`created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

INDEX (`item_id`),

INDEX (`status`),

INDEX (`request_code`)

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

");



$colsProp = $pdo->query("SHOW COLUMNS FROM `vpp_purchase_proposals`")->fetchAll(PDO::FETCH_COLUMN);

if (!in_array('custom_image', $colsProp)) {

$pdo->exec("ALTER TABLE `vpp_purchase_proposals` ADD COLUMN `custom_image` VARCHAR(255) DEFAULT '' AFTER `unit`");

}

} catch (PDOException $e) {}



$allItems = $pdo->query("SELECT * FROM vpp_items ORDER BY item_name ASC")->fetchAll();



// Truy vấn danh sách nhân viên để hỗ trợ gợi ý ô nhập họ tên / mã số

$employees = [];

try {

$employees = $pdo->query("SELECT id, fullname, department FROM employees WHERE status = 'Đang làm việc' ORDER BY fullname ASC")->fetchAll();

} catch (PDOException $e) {}



if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'submit_purchase_proposal') {

$employee_id = !empty($_POST['employee_id']) ? intval($_POST['employee_id']) : null;

$receiver_name = trim($_POST['receiver_name'] ?? '');

$department = trim($_POST['department'] ?? '');

$items_data = $_POST['items'] ?? [];



try {

if (empty($receiver_name) || empty($department)) {

throw new Exception("Vui lòng điền đầy đủ Họ tên/Mã số người đề xuất và Bộ phận / Phòng ban!");

}

if (empty($items_data)) {

throw new Exception("Vui lòng chọn hoặc điền ít nhất một mặt hàng cần đề xuất mua!");

}



$baseCode = 'MUA-' . date('dmYHis');

$pdo->beginTransaction();



$stmtIns = $pdo->prepare("

INSERT INTO vpp_purchase_proposals

(request_code, employee_id, receiver_name, department, item_id, custom_item_name, specification, unit, custom_image, quantity, status)

VALUES (:code, :empid, :rec, :dept, :it_id, :c_name, :spec, :unit, :img, :qty, 'Chờ duyệt')

");



$insertedCount = 0;

foreach ($items_data as $idx => $row) {

$itemId = intval($row['item_id'] ?? 0);

$qty = intval($row['quantity'] ?? 0);

if ($qty <= 0) continue;



$customName = trim($row['custom_item_name'] ?? '');

$spec = trim($row['specification'] ?? '');

$unit = trim($row['unit'] ?? 'Cái');

$imgPath = '';



// Xử lý nén ảnh nếu là mặt hàng mới ngoài kho và có upload file

if ($itemId === 0 && isset($_FILES['items']['name'][$idx]['image']) && $_FILES['items']['error'][$idx]['image'] === UPLOAD_ERR_OK) {

$fileData = [

'name' => $_FILES['items']['name'][$idx]['image'],

'tmp_name' => $_FILES['items']['tmp_name'][$idx]['image'],

'error' => $_FILES['items']['error'][$idx]['image']

];

$imgPath = compressAndUploadImage($fileData);

}



$stmtIns->execute([

':code' => $baseCode,

':empid' => $employee_id,

':rec' => $receiver_name,

':dept' => $department,

':it_id' => ($itemId > 0 ? $itemId : null),

':c_name' => ($itemId > 0 ? '' : $customName),

':spec' => $spec,

':unit' => $unit,

':img' => $imgPath,

':qty' => $qty

]);

$insertedCount++;

}



if ($insertedCount === 0) {

throw new Exception("Số lượng đề xuất của các mặt hàng không hợp lệ!");

}



$pdo->commit();

$message = "Gửi đề xuất mua văn phòng phẩm thành công! Mã phiếu của bạn là: <strong>{$baseCode}</strong>";

} catch (Exception $e) {

if ($pdo->inTransaction()) $pdo->rollBack();

$error = "Lỗi gửi đề xuất: " . $e->getMessage();

}

}

function handleGlobalModeChange(select) {

const mode = select.value;



const availablePanel = document.getElementById('available_panel');

const customPanel = document.getElementById('custom_panel');

const addCustomButton = document.getElementById('btn_add_custom_row');



if (mode === 'available') {

availablePanel.classList.add('active');

customPanel.classList.remove('active');



addCustomButton.style.display = 'none';



/*
 * Xóa dữ liệu items ẩn của chế độ custom.
 * Không ảnh hưởng PHP/backend.
 */

document.getElementById('available_hidden_inputs').innerHTML = '';



updateSelectedProducts();



} else if (mode === 'custom') {

availablePanel.classList.remove('active');

customPanel.classList.add('active');



addCustomButton.style.display = 'inline-block';



/*
 * Bỏ chọn toàn bộ sản phẩm có sẵn khi chuyển sang
 * chế độ nhập mặt hàng mới.
 */

document.querySelectorAll('.product-checkbox').forEach(function(cb) {

cb.checked = false;

cb.closest('tr').classList.remove('selected-row');

});



document.querySelectorAll('.product-quantity').forEach(function(input) {

input.disabled = true;

});



document.getElementById('available_hidden_inputs').innerHTML = '';



updateSelectedCount();

}

}

// TÌM SẢN PHẨM CÓ SẴN
// =========================================================

function filterAvailableProducts(input) {

const keyword = input.value.trim().toLowerCase();

const rows = document.querySelectorAll('.available-product-row');



let visibleCount = 0;



rows.forEach(function(row) {

const name = row.dataset.name || '';

const spec = row.dataset.spec || '';



const matched =

!keyword ||

name.includes(keyword) ||

spec.includes(keyword);



if (matched) {

row.style.display = '';

visibleCount++;

} else {

row.style.display = 'none';

}

});



const emptyBox = document.getElementById('available_empty');



if (visibleCount === 0) {

emptyBox.style.display = 'block';

} else {

emptyBox.style.display = 'none';

}



const clearButton = document.getElementById('btn_clear_search');



if (clearButton) {

clearButton.style.display = keyword ? 'block' : 'none';

}

}



// =========================================================
// XÓA TÌM KIẾM
// =========================================================

function clearProductSearch() {

const search = document.getElementById('available_search');



search.value = '';



filterAvailableProducts(search);



search.focus();

}



// =========================================================
// TICK SẢN PHẨM
// =========================================================

function handleProductCheck(checkbox) {

const row = checkbox.closest('tr');

const quantityInput = row.querySelector('.product-quantity');



if (checkbox.checked) {

row.classList.add('selected-row');

quantityInput.disabled = false;



if (!quantityInput.value || parseInt(quantityInput.value) <= 0) {

quantityInput.value = 1;

}

} else {

row.classList.remove('selected-row');

quantityInput.disabled = true;

}



updateSelectedProducts();

}



// =========================================================
// THAY ĐỔI SỐ LƯỢNG SẢN PHẨM
// =========================================================

function handleProductQuantity(input) {

if (!input.disabled) {

let value = parseInt(input.value);



if (!value || value < 1) {

input.value = 1;

}

}



updateSelectedProducts();

}

// =========================================================
// TẠO INPUT ẨN ĐỂ GỬI ĐÚNG STRUCTURE CHO PHP
// =========================================================

function updateSelectedProducts() {

const hiddenContainer = document.getElementById(

'available_hidden_inputs'

);



hiddenContainer.innerHTML = '';



const checkedProducts = document.querySelectorAll(

'.product-checkbox:checked'

);



checkedProducts.forEach(function(checkbox, index) {

const row = checkbox.closest('tr');

const quantityInput = row.querySelector('.product-quantity');



const itemId = checkbox.dataset.id || '';

const itemName = checkbox.dataset.name || '';

const specification = checkbox.dataset.spec || '';

const unit = checkbox.dataset.unit || 'Cái';

let quantity = parseInt(quantityInput.value);



if (!quantity || quantity < 1) {

quantity = 1;

quantityInput.value = 1;

}



/*
 * PHP hiện tại đọc:
 *
 * items[index][item_id]
 * items[index][custom_item_name]
 * items[index][specification]
 * items[index][unit]
 * items[index][quantity]
 *
 * Vì vậy giao diện mới vẫn tạo đúng 100% cấu trúc này.
 */

createHiddenInput(

hiddenContainer,

`items[${index}][mode_select]`,

'available'

);



createHiddenInput(

hiddenContainer,

`items[${index}][item_id]`,

itemId

);



createHiddenInput(

hiddenContainer,

`items[${index}][custom_item_name]`,

itemName

);



createHiddenInput(

hiddenContainer,

`items[${index}][specification]`,

specification

);



createHiddenInput(

hiddenContainer,

`items[${index}][unit]`,

unit

);



createHiddenInput(

hiddenContainer,

`items[${index}][quantity]`,

quantity

);

});



updateSelectedCount();

}



// =========================================================
// TẠO INPUT ẨN
// =========================================================

function createHiddenInput(container, name, value) {

const input = document.createElement('input');



input.type = 'hidden';

input.name = name;

input.value = value;



container.appendChild(input);

}



// =========================================================
// ESCAPE HTML
// =========================================================

function escapeHtml(value) {

return String(value ?? '')

.replace(/&/g, '&amp;')

.replace(/</g, '&lt;')

.replace(/>/g, '&gt;')

.replace(/"/g, '&quot;')

.replace(/'/g, '&#039;');

}



// =========================================================
// UPLOAD ẢNH
// =========================================================

function handleImageChange(input, idx) {

if (input.files && input.files[0]) {

const reader = new FileReader();



reader.onload = function(e) {

const thumb = document.getElementById('thumb_' + idx);

const btnWrap = document.getElementById('upload_btn_wrap_' + idx);

const previewWrap = document.getElementById('preview_wrap_' + idx);



if (thumb) {

thumb.src = e.target.result;

}



if (btnWrap) {

btnWrap.style.display = 'none';

}



if (previewWrap) {

previewWrap.style.display = 'flex';

}

};



reader.readAsDataURL(input.files[0]);

}

}

);

}

selected.forEach(function(checkbox) {

if (

!quantityInput ||

!quantityInput.value ||

parseInt(quantityInput.value) <= 0

) {

invalidQuantity = true;

}

});



if (invalidQuantity) {

e.preventDefault();

alert("Vui lòng nhập số lượng hợp lệ cho các mặt hàng đã chọn!");

return false;

}

}



if (mode === 'custom') {

'.custom-item-name'

);



let hasValidName = false;



customNames.forEach(function(input) {

if (input.value.trim() !== '') {

hasValidName = true;

}

});



if (!hasValidName) {

e.preventDefault();

alert("Vui lòng nhập ít nhất một mặt hàng mới cần đề xuất!");

return false;

}

}



if (!mode) {

e.preventDefault();

alert("Vui lòng chọn loại mặt hàng!");

return false;

}

});
