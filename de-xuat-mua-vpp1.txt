// ==================== HÀM NÉN ẢNH DƯỚI 100KB ====================
function compressAndUploadImage($file, $targetDir = 'uploads/vpp/') {
if (!isset($file['tmp_name']) || empty($file['tmp_name'])) return '';
if (!is_dir($targetDir)) {
mkdir($targetDir, 0777, true);
}

$fileInfo = @getimagesize($file['tmp_name']);
if ($fileInfo === false) return '';

$mime = $fileInfo['mime'];
$srcImg = null;
if ($mime === 'image/jpeg') {
$srcImg = imagecreatefromjpeg($file['tmp_name']);
} elseif ($mime === 'image/png') {
$srcImg = imagecreatefrompng($file['tmp_name']);
} elseif ($mime === 'image/webp') {
$srcImg = imagecreatefromwebp($file['tmp_name']);
}
if (!$srcImg) return '';

$ext = pathinfo($file['name'], PATHINFO_EXTENSION);
$newFileName = 'vpp_mua_' . uniqid() . '.' . ($ext ?: 'jpg');
$destination = $targetDir . $newFileName;

$quality = 85;
if ($mime === 'image/png') {
imagepng($srcImg, $destination, 8);
} else {
imagejpeg($srcImg, $destination, $quality);
}

while (filesize($destination) > 100 * 1024 && $quality > 20) {
$quality -= 15;
if ($mime === 'image/png') {
imagepng($srcImg, $destination, 9);
} else {
imagejpeg($srcImg, $destination, $quality);
}
}

imagedestroy($srcImg);
return $destination;
}

try {
$pdo->exec("
CREATE TABLE IF NOT EXISTS `vpp_purchase_proposals` (
`id` INT AUTO_INCREMENT PRIMARY KEY,
`request_code` VARCHAR(60) DEFAULT '',
`employee_id` INT DEFAULT NULL,
`receiver_name` VARCHAR(150) NOT NULL,
`department` VARCHAR(100) NOT NULL,
`item_id` INT DEFAULT NULL,
`custom_item_name` VARCHAR(150) DEFAULT '',
`specification` VARCHAR(100) DEFAULT '',
`unit` VARCHAR(50) DEFAULT 'Cái',
`custom_image` VARCHAR(255) DEFAULT '',
`quantity` INT NOT NULL DEFAULT 1,
`pre_approved_quantity` INT DEFAULT NULL,
`approved_quantity` INT DEFAULT NULL,
`approval_note` VARCHAR(255) DEFAULT '',
`status` VARCHAR(50) DEFAULT 'Chờ duyệt',
`reject_reason` VARCHAR(255) DEFAULT '',
`created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
INDEX (`item_id`),
INDEX (`status`),
INDEX (`request_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
");

$colsProp = $pdo->query("SHOW COLUMNS FROM `vpp_purchase_proposals`")->fetchAll(PDO::FETCH_COLUMN);
if (!in_array('custom_image', $colsProp)) {
$pdo->exec("ALTER TABLE `vpp_purchase_proposals` ADD COLUMN `custom_image` VARCHAR(255) DEFAULT '' AFTER `unit`");
}
} catch (PDOException $e) {}

$allItems = $pdo->query("SELECT * FROM vpp_items ORDER BY item_name ASC")->fetchAll();

// Truy vấn danh sách nhân viên để hỗ trợ gợi ý ô nhập họ tên / mã số
$employees = [];

$employees = $pdo->query("SELECT id, fullname, department FROM employees WHERE status = 'Đang làm việc' ORDER BY fullname ASC")->fetchAll();
} catch (PDOException $e) {}

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'submit_purchase_proposal') {
$employee_id = !empty($_POST['employee_id']) ? intval($_POST['employee_id']) : null;
$receiver_name = trim($_POST['receiver_name'] ?? '');
$department = trim($_POST['department'] ?? '');
$items_data = $_POST['items'] ?? [];

try {
if (empty($receiver_name) || empty($department)) {
throw new Exception("Vui lòng điền đầy đủ Họ tên/Mã số người đề xuất và Bộ phận / Phòng ban!");
}
if (empty($items_data)) {
throw new Exception("Vui lòng chọn hoặc điền ít nhất một mặt hàng cần đề xuất mua!");
}

$baseCode = 'MUA-' . date('dmYHis');
$pdo->beginTransaction();

$stmtIns = $pdo->prepare("
INSERT INTO vpp_purchase_proposals
(request_code, employee_id, receiver_name, department, item_id, custom_item_name, specification, unit, custom_image, quantity, status)
VALUES (:code, :empid, :rec, :dept, :it_id, :c_name, :spec, :unit, :img, :qty, 'Chờ duyệt')
");

$insertedCount = 0;
foreach ($items_data as $idx => $row) {
$itemId = intval($row['item_id'] ?? 0);
$qty = intval($row['quantity'] ?? 0);
if ($qty <= 0) continue;

$customName = trim($row['custom_item_name'] ?? '');
$spec = trim($row['specification'] ?? '');
$unit = trim($row['unit'] ?? 'Cái');
$imgPath = '';

// Xử lý nén ảnh nếu là mặt hàng mới ngoài kho và có upload file
if ($itemId === 0 && isset($_FILES['items']['name'][$idx]['image']) && $_FILES['items']['error'][$idx]['image'] === UPLOAD_ERR_OK) {
$fileData = [
'name' => $_FILES['items']['name'][$idx]['image'],
'tmp_name' => $_FILES['items']['tmp_name'][$idx]['image'],
'error' => $_FILES['items']['error'][$idx]['image']
];
$imgPath = compressAndUploadImage($fileData);
}

$stmtIns->execute([
':code' => $baseCode,
':empid' => $employee_id,
':rec' => $receiver_name,
':dept' => $department,
':it_id' => ($itemId > 0 ? $itemId : null),
':c_name' => ($itemId > 0 ? '' : $customName),
':spec' => $spec,
':unit' => $unit,
':img' => $imgPath,
':qty' => $qty
]);
$insertedCount++;
}

if ($insertedCount === 0) {
throw new Exception("Số lượng đề xuất của các mặt hàng không hợp lệ!");
}

$pdo->commit();
$message = "Gửi đề xuất mua văn phòng phẩm thành công! Mã phiếu của bạn là: <strong>{$baseCode}</strong>";
} catch (Exception $e) {
if ($pdo->inTransaction()) $pdo->rollBack();
$error = "Lỗi gửi đề xuất: " . $e->getMessage();
}
}

// Xử lý hiển thị preview ảnh sau khi chọn file

}
}
