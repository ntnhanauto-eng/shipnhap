// ==================== 1. XỬ LÝ HÀNH ĐỘNG ĐỀ XUẤT MUA ====================
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    $act = $_POST['action'];

    if ($act === 'approve_and_push_next_period') {
        $req_code = trim($_POST['request_code'] ?? '');
        $appr_qtys = $_POST['approved_quantities'] ?? [];
        $appr_notes = $_POST['item_notes'] ?? [];
        try {
            if (empty($req_code)) throw new Exception("Mã yêu cầu không hợp lệ!");
            $pdo->beginTransaction();

            $stmtProps = $pdo->prepare("
                SELECT p.*, 
                COALESCE(i.item_name, p.custom_item_name) AS final_item_name,
                COALESCE(i.specification, p.specification) AS final_spec,
                COALESCE(i.unit, p.unit) AS final_unit
                FROM vpp_purchase_proposals p
                LEFT JOIN vpp_items i ON p.item_id = i.id
                WHERE p.request_code = :code
            ");
            $stmtProps->execute([':code' => $req_code]);
            $rows = $stmtProps->fetchAll();

            foreach ($rows as $r) {
                $propId = $r['id'];
                $apprQty = isset($appr_qtys[$propId]) ? max(0, intval($appr_qtys[$propId])) : max(0, intval($r['quantity']));
                $note = trim($appr_notes[$propId] ?? '');

                $stmtUp = $pdo->prepare("UPDATE vpp_purchase_proposals SET status = 'Đã duyệt', approved_quantity = :aq, approval_note = :an WHERE id = :id");
                $stmtUp->execute([':aq' => $apprQty, ':an' => $note, ':id' => $propId]);

                if ($apprQty > 0) {
                    $itemKey = md5(mb_strtolower(trim($r['final_item_name'] . '|' . $r['final_spec'] . '|' . $r['final_unit'])));
                    
                    $stmtCheck = $pdo->prepare("SELECT * FROM vpp_purchase_next_period WHERE item_key = :k");
                    $stmtCheck->execute([':k' => $itemKey]);
                    $existingNext = $stmtCheck->fetch();

                    $sourceInfo = "{$req_code} ({$r['department']} - {$r['receiver_name']}: {$apprQty} {$r['final_unit']})";

                    if ($existingNext) {
                        $newTotal = intval($existingNext['total_approved_qty']) + $apprQty;
                        $sources = trim($existingNext['source_requests'] . ' | ' . $sourceInfo);
                        $stmtUpNext = $pdo->prepare("UPDATE vpp_purchase_next_period SET total_approved_qty = :tot, source_requests = :src WHERE item_key = :k");
                        $stmtUpNext->execute([':tot' => $newTotal, ':src' => $sources, ':k' => $itemKey]);
                    } else {
                        $stmtInsNext = $pdo->prepare("INSERT INTO vpp_purchase_next_period (item_key, item_id, item_name, specification, unit, total_approved_qty, source_requests, status) VALUES (:k, :it_id, :name, :spec, :unit, :tot, :src, 'Chờ mua')");
                        $stmtInsNext->execute([
                            ':k' => $itemKey,
                            ':it_id' => $r['item_id'],
                            ':name' => $r['final_item_name'],
                            ':spec' => $r['final_spec'],
                            ':unit' => $r['final_unit'],
                            ':tot' => $apprQty,
                            ':src' => $sourceInfo
                        ]);
                    }
                }
            }

            $pdo->commit();
            $message = "Đã phê duyệt và đẩy thành công các mặt hàng của yêu cầu [{$req_code}] vào danh sách cần mua cho kỳ tới!";
        } catch (Exception $e) {
            if ($pdo->inTransaction()) $pdo->rollBack();
            $error = "Lỗi phê duyệt: " . $e->getMessage();
        }
    }

    if ($act === 'reject_purchase_group') {
        $req_code = trim($_POST['request_code'] ?? '');
        $reason = trim($_POST['reject_reason'] ?? 'Không duyệt mua');
        try {
            $stmt = $pdo->prepare("UPDATE vpp_purchase_proposals SET status = 'Từ chối', reject_reason = :r WHERE request_code = :code");
            $stmt->execute([':r' => $reason, ':code' => $req_code]);
            $message = "Đã từ chối toàn bộ yêu cầu đề xuất mua [{$req_code}]!";
        } catch (PDOException $e) { $error = "Lỗi: " . $e->getMessage(); }
    }

    if ($act === 'delete_purchase_proposal') {
        $req_code = trim($_POST['request_code'] ?? '');
        try {
            $stmt = $pdo->prepare("DELETE FROM vpp_purchase_proposals WHERE request_code = :code");
            $stmt->execute([':code' => $req_code]);
            $message = "Đã xóa đợt đề xuất mua [{$req_code}] thành công!";
        } catch (PDOException $e) { $error = "Lỗi xóa: " . $e->getMessage(); }
    }

    if ($act === 'mark_purchased') {
        $next_id = intval($_POST['next_id'] ?? 0);
        try {
            $stmt = $pdo->prepare("UPDATE vpp_purchase_next_period SET status = 'Đã mua' WHERE id = :id");
            $stmt->execute([':id' => $next_id]);
            $message = "Đã cập nhật trạng thái mặt hàng thành 'Đã mua'!";
        } catch (PDOException $e) { $error = "Lỗi: " . $e->getMessage(); }
    }

    if ($act === 'delete_next_item') {
        $next_id = intval($_POST['next_id'] ?? 0);
        try {
            $stmt = $pdo->prepare("DELETE FROM vpp_purchase_next_period WHERE id = :id");
            $stmt->execute([':id' => $next_id]);
            $message = "Đã xóa mặt hàng khỏi danh sách kỳ tới!";
        } catch (PDOException $e) { $error = "Lỗi: " . $e->getMessage(); }
    }
}

// ==================== 2. TRUY VẤN DỮ LIỆU ĐỀ XUẤT MUA ====================
$f_prop_search = trim($_GET['f_prop_search'] ?? '');

$f_prop_status = trim($_GET['f_prop_status'] ?? '');

$whereP = ["1=1"];
$paramsP = [];
if (!empty($f_prop_search)) {
    $whereP[] = "(p.request_code LIKE :ps1 OR p.receiver_name LIKE :ps2 OR p.custom_item_name LIKE :ps3)";
    $paramsP[':ps1'] = "%$f_prop_search%";
    $paramsP[':ps2'] = "%$f_prop_search%";
    $paramsP[':ps3'] = "%$f_prop_search%";
}
if (!empty($f_prop_dept)) {
    $whereP[] = "p.department = :pdept";
    $paramsP[':pdept'] = $f_prop_dept;
}
if (!empty($f_prop_status)) {
    $whereP[] = "p.status = :pstat";
    $paramsP[':pstat'] = $f_prop_status;
}

$countPropStmt = $pdo->prepare("SELECT COUNT(DISTINCT p.request_code) FROM vpp_purchase_proposals p WHERE " . implode(' AND ', $whereP));
$countPropStmt->execute($paramsP);
$total_proposals = intval($countPropStmt->fetchColumn() ?? 0);

$pages_prop = ceil($total_proposals / $limit) ?: 1;
if ($p_prop > $pages_prop) $p_prop = $pages_prop;
$offset_prop = max(0, ($p_prop - 1) * $limit);

$sqlProposals = "
    SELECT 
        p.request_code, p.employee_id, p.receiver_name, p.department, p.status, p.reject_reason, p.created_at,
        COUNT(p.id) AS item_count,
        SUM(p.quantity) AS total_qty_requested,
        SUM(COALESCE(p.approved_quantity, p.quantity)) AS total_qty_approved,
        JSON_ARRAYAGG(
            JSON_OBJECT(
                'proposal_id', p.id,
                'item_name', COALESCE(i.item_name, p.custom_item_name),
                'specification', COALESCE(i.specification, p.specification),
                'unit', COALESCE(i.unit, p.unit),
                'quantity', p.quantity,
                'approved_quantity', p.approved_quantity,
                'approval_note', p.approval_note
            )
        ) AS items_json
    FROM vpp_purchase_proposals p
    LEFT JOIN vpp_items i ON p.item_id = i.id
    WHERE " . implode(' AND ', $whereP) . "
    GROUP BY p.request_code, p.employee_id, p.receiver_name, p.department, p.status, p.reject_reason, p.created_at
    ORDER BY (p.status = 'Chờ duyệt') DESC, MAX(p.id) DESC
    LIMIT " . intval($limit) . " OFFSET " . intval($offset_prop) . "
";
$stmtPropList = $pdo->prepare($sqlProposals);
$stmtPropList->execute($paramsP);
$proposals = $stmtPropList->fetchAll();

<script>
function openModal(id) { document.getElementById(id).style.display = 'flex'; }
function closeModal(id) { document.getElementById(id).style.display = 'none'; }

function openDetailsModal(p) {
document.getElementById('dt_code').innerText = p.request_code;
document.getElementById('dt_receiver').innerText = p.receiver_name + (p.employee_id ? ' (#' + p.employee_id + ')' : '');
document.getElementById('dt_dept').innerText = p.department;

const tbody = document.getElementById('dt_tbody');
tbody.innerHTML = '';
const items = JSON.parse(p.items_json || '[]');
items.forEach((it, idx) => {
const appr = it.approved_quantity !== null ? it.approved_quantity : '--';
tbody.innerHTML += `<tr>
<td>${idx + 1}</td>
<td><b>${it.item_name}</b> ${it.specification ? '('+it.specification+')' : ''}</td>
<td>${it.unit}</td>
<td style="text-align:center; font-weight:bold; color:#0284c7;">${it.quantity}</td>
<td style="text-align:center; font-weight:bold; color:#15803d;">${appr}</td>
<td>${it.approval_note || '—'}</td>
</tr>`;
});
openModal('modal-details');
}

function openApproveModal(p) {
document.getElementById('appr_code').value = p.request_code;
document.getElementById('appr_code_text').innerText = p.request_code;
const tbody = document.getElementById('appr_tbody');
tbody.innerHTML = '';
const items = JSON.parse(p.items_json || '[]');
items.forEach((it, idx) => {
const def = it.quantity;
tbody.innerHTML += `<tr>
<td>${idx + 1}</td>
<td><b>${it.item_name}</b> ${it.specification ? '('+it.specification+')' : ''}</td>
<td>${it.unit}</td>
<td style="text-align:center; font-weight:bold; color:#0284c7;">${def}</td>
<td style="text-align:center;"><input type="number" name="approved_quantities[${it.proposal_id}]" min="0" value="${def}" required style="width:70px; text-align:center; font-weight:bold; color:#15803d;"></td>
<td><input type="text" name="item_notes[${it.proposal_id}]" value="${it.approval_note || ''}" placeholder="Ghi chú..."></td>
</tr>`;
});
openModal('modal-approve');
}

function openRejectModal(code) {
document.getElementById('rej_code').value = code;
openModal('modal-reject');
}

if (e.target.classList.contains('modal-overlay')) e.target.style.display = 'none';
}
